    59→    59→export function PriceChart({

    61→    61→  sma,
    62→    62→  signals,
    63→    63→  papers,

    74→    74→}: Props) {
    75→    75→  const wrapRef = useRef<HTMLDivElement>(null);
    76→    76→  const plotRef = useRef<HTMLDivElement>(null);
    77→    77→  const [width, setWidth] = useState(900);
    78→    78→  const [measuredH, setMeasuredH] = useState(360);
    79→    79→  const [viewEndRaw, setViewEndRaw] = useState<number | null>(null);
    80→    80→  /** user zoom override — visible bar count (null = auto by width) */
    81→    81→  const [bars, setBars] = useState<number | null>(null);
    82→    82→  /** hover crosshair — candle index + cursor y inside the plot */
    83→    83→  const [hover, setHover] = useState<{ i: number; py: number } | null>(null);
    84→    84→  const dragRef = useRef<{ x: number; end: number } | null>(null);
    85→    85→
    86→    86→  useEffect(() => {
    87→    87→    const el = wrapRef.current;
    88→    88→    const plot = plotRef.current;
    89→    89→    if (!el) return;
    90→    90→    const ro = new ResizeObserver((entries) => {
    91→    91→      for (const en of entries) {
    92→    92→        if (en.target === el) {
    93→    93→          setWidth(Math.max(320, Math.floor(en.contentRect.width)));
    94→    94→        } else if (en.target === plot) {
    95→    95→          // floor of 160px keeps the plot usable — going lower would overflow the card
    96→    96→          setMeasuredH(Math.max(160, Math.floor(en.contentRect.height)));
    97→    97→        }
    98→    98→      }
    99→    99→    });
   100→   100→    ro.observe(el);
   101→   101→    if (plot) ro.observe(plot);
   102→   102→    return () => ro.disconnect();
   103→   103→  }, []);
   104→   104→
   105→   105→  // wheel zoom — native listener (passive: false) so preventDefault works
   106→   106→  useEffect(() => {
   107→   107→    const el = wrapRef.current;
   108→   108→    if (!el) return;
   109→   109→    const onWheel = (e: WheelEvent) => {
   110→   110→      if (!e.deltaY) return;
   111→   111→      e.preventDefault();
   112→   112→      setBars((prev) => {
   113→   113→        const base = prev ?? autoBars(width);
   114→   114→        const next = e.deltaY > 0 ? base * 1.18 : base / 1.18;
   115→   115→        return Math.max(24, Math.min(340, Math.round(next)));
   116→   116→      });
   117→   117→    };
   118→   118→    el.addEventListener("wheel", onWheel, { passive: false });
   119→   119→    return () => el.removeEventListener("wheel", onWheel);
   120→   120→  }, [width]);
   121→   121→
   122→   122→  const n = klines.length;
   123→   123→  const winSize = Math.max(20, Math.min(bars ?? autoBars(width), n || 1));
   124→   124→  const viewEnd = Math.min(viewEndRaw ?? n - 1, n - 1);
   125→   125→  const viewStart = Math.max(0, viewEnd - winSize + 1);
   126→   126→
   127→   127→  const zoom = useCallback(
   128→   128→    (dir: 1 | -1) =>
   129→   129→      setBars((prev) => {
   130→   130→        const base = prev ?? autoBars(width);
   131→   131→        const next = dir === 1 ? base / 1.35 : base * 1.35;
   132→   132→        return Math.max(24, Math.min(340, Math.round(next)));
   133→   133→      }),
   134→   134→    [width],
   135→   135→  );
   136→   136→
   137→   137→  const jumpLatest = useCallback(() => setViewEndRaw(null), []);
   138→   138→
   139→   139→  const height = fill ? measuredH : width < 520 ? 280 : 360;
   140→   140→  const plotW = width - PAD_L - PAD_R;
   141→   141→  // ── Equity pane (Đường cong vốn) — a small strip carved out of the plot bottom
   142→   142→  const eq = useMemo(() => equityFrom(papers), [papers]);
   143→   143→  const eqOn = eq.points.length >= 2;
   144→   144→  const EQ_PANE = 58;
   145→   145→  const eqPaneH = eqOn ? EQ_PANE : 0;
   146→   146→  const plotH = height - PAD_T - PAD_B - eqPaneH;
   147→   147→  const cw = plotW / Math.max(1, viewEnd - viewStart + 1);
   148→   148→
   149→   149→  const { minP, maxP } = useMemo(() => {
   150→   150→    let lo = Infinity;
   151→   151→    let hi = -Infinity;
   152→   152→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   153→   153→      const k = klines[i];
   154→   154→      if (k.l < lo) lo = k.l;
   155→   155→      if (k.h > hi) hi = k.h;
   156→   156→      const s = sma[i];
   157→   157→      if (s != null && s < lo) lo = s;
   158→   158→      if (s != null && s > hi) hi = s;
   159→   159→      const st = stop?.[i];
   160→   160→      if (st != null && st < lo) lo = st;
   161→   161→      if (st != null && st > hi) hi = st;
   162→   162→    }
   163→   163→    // NOTE: open SL/TP are deliberately NOT part of the scale — a wide 6% SL
   164→   164→    // would squash the candles. Levels are clipped to the plot instead.
   165→   165→    if (openPaper && openPaper.entry >= lo * 0.98 && openPaper.entry <= hi * 1.02) {
   166→   166→      if (openPaper.entry < lo) lo = openPaper.entry;
   167→   167→      if (openPaper.entry > hi) hi = openPaper.entry;
   168→   168→    }
   169→   169→    if (!isFinite(lo) || !isFinite(hi)) return { minP: 0, maxP: 1 };
   170→   170→    if (hi === lo) {
   171→   171→      hi += hi * 0.001 || 1;
   172→   172→      lo -= lo * 0.001 || 1;
   173→   173→    }
   174→   174→    const pad = (hi - lo) * 0.06;
   175→   175→    return { minP: lo - pad, maxP: hi + pad };
   176→   176→  }, [klines, sma, stop, viewStart, viewEnd, n, openPaper]);
   177→   177→
   178→   178→  const x = (i: number) => PAD_L + (i - viewStart) * cw + cw / 2;
   179→   179→  const y = (p: number) => PAD_T + ((maxP - p) / (maxP - minP)) * plotH;
   180→   180→  /** inverse of y() — price under the crosshair */
   181→   181→  const priceAtY = (py: number) => minP + ((PAD_T + plotH - py) / plotH) * (maxP - minP);
   182→   182→
   183→   183→  const onPointerDown = (e: React.PointerEvent<SVGSVGElement>) => {
   184→   184→    dragRef.current = { x: e.clientX, end: viewEnd };
   185→   185→    (e.target as Element).setPointerCapture?.(e.pointerId);
   186→   186→  };
   187→   187→  const onPointerMove = (e: React.PointerEvent<SVGSVGElement>) => {
   188→   188→    // hover crosshair (TradingView-style) — always tracked, pan continues on top
   189→   189→    const rect = e.currentTarget.getBoundingClientRect();
   190→   190→    const py = Math.max(PAD_T, Math.min(PAD_T + plotH, e.clientY - rect.top));
   191→   191→    const i = Math.max(viewStart, Math.min(viewEnd, viewStart + Math.floor((e.clientX - rect.left - PAD_L) / cw)));
   192→   192→    setHover((prev) => (prev && prev.i === i && Math.abs(prev.py - py) < 0.5 ? prev : { i, py }));
   193→   193→    const d = dragRef.current;
   194→   194→    if (!d) return;
   195→   195→    const shift = Math.round((d.x - e.clientX) / cw);
   196→   196→    const end = Math.min(n - 1, Math.max(winSize - 1, d.end + shift));
   197→   197→    setViewEndRaw(end >= n - 1 ? null : end);
   198→   198→  };
   199→   199→  const onPointerUp = () => {
   200→   200→    dragRef.current = null;
   201→   201→  };
   202→   202→  const onPointerLeave = () => {
   203→   203→    dragRef.current = null;
   204→   204→    setHover(null);
   205→   205→  };
   206→   206→
   207→   207→  // ── Equity pane geometry (Đường cong vốn) ──────────────────────────────
   208→   208→  const eqTop = PAD_T + plotH + 10;
   209→   209→  const eqH = eqPaneH - 16;
   210→   210→  const eqLo = Math.min(eq.trough, 0);
   211→   211→  const eqHi = Math.max(eq.peak, eqLo + 0.5);
   212→   212→  const eqY = (v: number) => eqTop + ((eqHi - v) / (eqHi - eqLo)) * eqH;
   213→   213→  const eqX = (bar: number) => Math.max(PAD_L, Math.min(PAD_L + plotW, x(bar)));
   214→   214→  // trades closed up to the viewed bar — the curve stays honest while panning back
   215→   215→  const eqSeen = eqOn ? eq.points.filter((p) => p.bar <= viewEnd) : [];
   216→   216→  const eqLine = (() => {
   217→   217→    if (!eqOn || eqH <= 4) return null;
   218→   218→    if (eqSeen.length === 0) {
   219→   219→      const d = `M${PAD_L},${eqY(0).toFixed(1)} L${(PAD_L + plotW).toFixed(1)},${eqY(0).toFixed(1)}`;
   220→   220→      return { d, area: `${d} Z`, endV: 0, endY: eqY(0) };
   221→   221→    }
   222→   222→    const pts = eqSeen;
   223→   223→    let d = pts[0].bar > viewStart ? `M${PAD_L},${eqY(0).toFixed(1)} L${eqX(pts[0].bar).toFixed(1)},${eqY(pts[0].cum).toFixed(1)}` : `M${eqX(pts[0].bar).toFixed(1)},${eqY(pts[0].cum).toFixed(1)}`;
   224→   224→    for (let k = 1; k < pts.length; k++) d += ` L${eqX(pts[k].bar).toFixed(1)},${eqY(pts[k].cum).toFixed(1)}`;
   225→   225→    const endY = eqY(pts[pts.length - 1].cum);
   226→   226→    d += ` L${(PAD_L + plotW).toFixed(1)},${endY.toFixed(1)}`; // flat extension to "now"
   227→   227→    const area = `${d} L${(PAD_L + plotW).toFixed(1)},${eqY(0).toFixed(1)} L${PAD_L},${eqY(0).toFixed(1)} Z`;
   228→   228→    return { d, area, endV: pts[pts.length - 1].cum, endY };
   229→   229→  })();
   230→   230→
   231→   231→  const sigInView = signals.filter((s) => s.i >= viewStart && s.i <= viewEnd);
   232→   232→  const eqColor = eqLine ? (eqLine.endV >= 0 ? UP : DOWN) : UP;
   233→   233→  const papersInView = papers.filter(
   234→   234→    (p) => p.exitIndex !== undefined ? p.exitIndex >= viewStart || p.entryIndex <= viewEnd : p.entryIndex <= viewEnd,
   235→   235→  );
   236→   236→
   237→   237→  // ── Volume profile over the visible window (buy/sell split via taker data) ─
   238→   238→  const vp = useMemo(() => volumeProfile(klines, viewStart, viewEnd, 26), [klines, viewStart, viewEnd]);
   239→   239→  const vpW = plotW * 0.16;
   240→   240→
   241→   241→  // ── Whale delta histogram over the visible window ────────────────────────
   242→   242→  const whale = useMemo(() => {
   243→   243→    let maxD = 0;
   244→   244→    const rows: { i: number; d: number }[] = [];
   245→   245→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   246→   246→      const k = klines[i];
   247→   247→      if (k.tb == null || k.v <= 0) continue;
   248→   248→      const d = 2 * k.tb - k.v;
   249→   249→      rows.push({ i, d });
   250→   250→      if (Math.abs(d) > maxD) maxD = Math.abs(d);
   251→   251→    }
   252→   252→    return { rows, maxD };
   253→   253→  }, [klines, viewStart, viewEnd, n]);
   254→   254→  const whaleH = plotH * 0.2;
   255→   255→
   256→   256→  // ── Wyckoff layers (only when enabled) ─────────────────────────────────
   257→   257→  const wyOn = !!wyckoffOn && !!wyckoff;
   258→   258→  const wyBoxes = wyOn
   259→   259→    ? wyckoff!.boxes.filter((b) => !b.deleted && b.right >= viewStart && b.left <= viewEnd)
   260→   260→    : [];
   261→   261→  const wyEvents = wyOn
   262→   262→    ? wyckoff!.events.filter((e) => !e.deleted && e.bar >= viewStart && e.bar <= viewEnd)
   263→   263→    : [];
   264→   264→  const wyPivots = wyOn
   265→   265→    ? wyckoff!.pivots.filter((p) => !p.hidden && p.bar >= viewStart && p.bar <= viewEnd)
   266→   266→    : [];
   267→   267→  const tintOf = (i: number): 0 | 1 | -1 => (wyOn ? (wyckoff!.tints[i] as 0 | 1 | -1) : 0);
   268→   268→
   269→   269→  /** polyline path skipping nulls (EMA arrays) */
   270→   270→  const linePath = (arr: (number | null)[]): string => {
   271→   271→    let d = "";
   272→   272→    let pen = false;
   273→   273→    for (let i = viewStart; i <= viewEnd; i++) {
   274→   274→      const v = arr[i];
   275→   275→      if (v == null) {
   276→   276→        pen = false;
   277→   277→        continue;
   278→   278→      }
   279→   279→      d += `${pen ? "L" : "M"}${x(i)},${y(v)} `;
   280→   280→      pen = true;
   281→   281→    }
   282→   282→    return d.trim();
   283→   283→  };
   284→   284→
   285→   285→  /** SMA ± band filled area — one closed subpath per contiguous SMA run */
   286→   286→  const bandPath = (): string => {
   287→   287→    let d = "";
   288→   288→    let top: string[] = [];
   289→   289→    let bot: string[] = [];
   290→   290→    const flush = () => {
   291→   291→      if (top.length >= 2) {
   292→   292→        d += `M${top.join(" L")} L${bot.reverse().join(" L")} Z `;
   293→   293→      }
   294→   294→      top = [];
   295→   295→      bot = [];
   296→   296→    };
   297→   297→    for (let i = viewStart; i <= viewEnd; i++) {
   298→   298→      const v = sma[i];
   299→   299→      if (v == null) {
   300→   300→        flush();
   301→   301→        continue;
   302→   302→      }
   303→   303→      top.push(`${x(i)},${y(v * (1 + BAND_PCT))}`);
   304→   304→      bot.push(`${x(i)},${y(v * (1 - BAND_PCT))}`);
   305→   305→    }
   306→   306→    flush();
   307→   307→    return d.trim();
   308→   308→  };
   309→   309→
   310→   310→  /** UT trailing-stop segments, colored per trend run */
   311→   311→  const stopSegs = (): { d: string; up: boolean }[] => {
   312→   312→    const segs: { d: string; up: boolean }[] = [];
   313→   313→    let cur: { d: string; up: boolean } | null = null;
   314→   314→    for (let i = viewStart; i <= viewEnd; i++) {
   315→   315→      const v = stop?.[i];
   316→   316→      const tr = trend?.[i] ?? 0;
   317→   317→      if (v == null || tr === 0) {
   318→   318→        cur = null;
   319→   319→        continue;
   320→   320→      }
   321→   321→      const up = tr === 1;
   322→   322→      if (cur && cur.up === up) cur.d += ` L${x(i)},${y(v)}`;
   323→   323→      else {
   324→   324→        cur = { d: `M${x(i)},${y(v)}`, up };
   325→   325→        segs.push(cur);
   326→   326→      }
   327→   327→    }
   328→   328→    return segs;
   329→   329→  };
   330→   330→
   331→   331→  const timeLabel = (t: number) => {
   332→   332→    const d = new Date(t);
   333→   333→    const hh = String(d.getHours()).padStart(2, "0");
   334→   334→    const mm = String(d.getMinutes()).padStart(2, "0");
   335→   335→    return `${d.getDate()}/${d.getMonth() + 1} ${hh}:${mm}`;
   336→   336→  };
   337→   337→
   338→   338→  if (n === 0) {
   339→   339→    return (
   340→   340→      <div
   341→   341→        ref={wrapRef}
   342→   342→        className={`flex items-center justify-center rounded-lg border border-zinc-800 bg-zinc-950 text-sm text-zinc-500 ${fill ? "h-full" : "h-64"}`}
   343→   343→      >
   344→   344→        {t(lang, "chart.empty")}
   345→   345→      </div>
   346→   346→    );
   347→   347→  }
   348→   348→
   349→   349→  const gridPrices = Array.from({ length: 5 }, (_, k) => minP + ((maxP - minP) * k) / 4);
   350→   350→  const openEntryLabel = openPaper?.flippedFrom ? t(lang, "chart.line.flipEntry") : t(lang, "chart.line.entry");
   351→   351→
   352→   352→  return (
   353→   353→    <div ref={wrapRef} className={`relative flex select-none flex-col ${fill ? "h-full" : ""}`}>
   354→   354→      <div ref={plotRef} className={`relative min-h-0 overflow-hidden ${fill ? "flex-1" : ""}`}>
   355→   355→        <svg
   356→   356→          width={width}
   357→   357→          height={height}
   358→   358→          className="touch-pan-y rounded-lg border border-zinc-800 bg-zinc-950"
   359→   359→          onPointerDown={onPointerDown}
   360→   360→          onPointerMove={onPointerMove}
   361→   361→          onPointerUp={onPointerUp}
   362→   362→          onPointerLeave={onPointerLeave}
   363→   363→          role="img"
   364→   364→          aria-label={t(lang, "chart.title")}
   365→   365→        >
   366→   366→          <defs>
   367→   367→            <clipPath id="wy-plot-clip">
   368→   368→              <rect x={PAD_L} y={PAD_T} width={plotW} height={plotH} />
   369→   369→            </clipPath>
   370→   370→          </defs>
   371→   371→          {/* grid + price axis */}
   372→   372→          {gridPrices.map((p, k) => (
   373→   373→            <g key={k}>
   374→   374→              <line x1={PAD_L} x2={PAD_L + plotW} y1={y(p)} y2={y(p)} stroke={GRID} strokeWidth={1} />
   375→   375→              <text x={PAD_L + plotW + 6} y={y(p) + 4} fill={TEXT} fontSize={10} fontFamily="ui-monospace, monospace">
   376→   376→                {fmtPrice(p)}
   377→   377→              </text>
   378→   378→            </g>
   379→   379→          ))}
   380→   380→          {/* time axis — edge labels anchor inward, greedy de-dupe avoids collisions */}
   381→   381→          {(() => {
   382→   382→            const last = width < 520 ? 3 : 5;
   383→   383→            const est = (s: string) => s.length * 6.2; // ~px per char @ 10px mono
   384→   384→            let prevRight = -Infinity;
   385→   385→            const nodes: React.ReactNode[] = [];
   386→   386→            for (let k = 0; k <= last; k++) {
   387→   387→              const idx = Math.round(viewStart + ((viewEnd - viewStart) * k) / last);
   388→   388→              const kk = klines[Math.min(idx, n - 1)];
   389→   389→              const label = timeLabel(kk.t);
   390→   390→              const w = est(label);
   391→   391→              const anchor = k === 0 ? "start" : k === last ? "end" : "middle";
