    58→  [

    74→    `Vốn: ${(r.capitalBefore ?? 0).toFixed(2)} → ${(r.capitalAfter ?? 0).toFixed(2)} USDT`,
    75→    `Tổng: ${closed} lệnh · win ${winRate.toFixed(0)}% · net ROE ${netPnlPct >= 0 ? "+" : ""}${netPnlPct.toFixed(1)}%`,

    78→const msgScan = (rows: ScanRow[], nPairs: number, nTf: number): string => {
    79→  const top3 = rows
    80→    .slice(0, 3)
    81→    .map((r) => `${r.symbol} ${r.interval} (${r.score.toFixed(1)})`)
    82→    .join(" · ");
    83→  const best = rows[0];
    84→  return [
    85→    "📊 AUTO QUÉT XONG",
    86→    `${nPairs} pair × ${nTf} khung = ${nPairs * nTf} combo · ${rows.length} hợp lệ`,
    87→    best
    88→      ? `Tốt nhất: #${best.symbol} ${best.interval} ${best.bias} · AUC ${best.auc.toFixed(2)} · ACC ${best.acc.toFixed(1)}% · Điểm ${best.score.toFixed(1)}`
    89→      : "Không có combo hợp lệ",
    90→    top3 ? `Top 3: ${top3}` : "",
    91→  ]
    92→    .filter(Boolean)
    93→    .join("\n");
    94→};

    99→
   100→export function AutoPanel({ lang, pairs, leverage, tickers, settings, onPickPair, pushToast }: Props) {
   101→  const [cfg, setCfg] = useState<AutoConfig>(loadAutoConfigSafe());
   102→  const [rows, setRows] = useState<ScanRow[]>([]);
   103→  const [scanAt, setScanAt] = useState<number | null>(null);
   104→  const [scanning, setScanning] = useState<{ done: number; total: number; current: string } | null>(null);
   105→  const [auto, setAuto] = useState<AutoState | null>(null);
   106→  const [tgConfigured, setTgConfigured] = useState<boolean | null>(null);
   107→  const [tgState, setTgState] = useState<"idle" | "sending" | "ok" | "fail">("idle");
   108→  const [tgDetail, setTgDetail] = useState<string | null>(null);
   109→  const [tgHelp, setTgHelp] = useState(false); // step-by-step connect guide (chat not found)
   110→  const [tgChatId, setTgChatId] = useState<string | null>(null); // effective channel
   111→  const [tgChats, setTgChats] = useState<TgChat[] | null>(null); // discover result
   112→  const [tgDiscovering, setTgDiscovering] = useState(false);
   113→  const [tgSaving, setTgSaving] = useState(false);
   114→  const [status, setStatus] = useState<string>("");
   115→  const [countdown, setCountdown] = useState<number | null>(null);
   116→
   117→  // refs mirror props/state for the interval callbacks (no stale closures,
   118→  // and the cycle interval is NOT reset every time a prop changes)
   119→  const cfgRef = useRef(cfg);
   120→  const rowsRef = useRef(rows);
   121→  const scanAgeRef = useRef(scanAt);
   122→  const tickersRef = useRef(tickers);
   123→  const pairsRef = useRef(pairs);
   124→  const settingsRef = useRef(settings);
   125→  const leverageRef = useRef(leverage);
   126→  const langRef = useRef(lang);
   127→  const busyRef = useRef(false);
   128→  const scanBusyRef = useRef(false);
   129→  const closingRef = useRef(false);
   130→  const mountedRef = useRef(true);
   131→  const nextAtRef = useRef<number>(0);
   132→
   133→  useEffect(() => {
   134→    cfgRef.current = cfg;
   135→  }, [cfg]);
   136→  useEffect(() => {
   137→    rowsRef.current = rows;
   138→  }, [rows]);
   139→  useEffect(() => {
   140→    scanAgeRef.current = scanAt;
   141→  }, [scanAt]);
   142→  useEffect(() => {
   143→    tickersRef.current = tickers;
   144→  }, [tickers]);
   145→  useEffect(() => {
   146→    pairsRef.current = pairs;
   147→  }, [pairs]);
   148→  useEffect(() => {
   149→    settingsRef.current = settings;
   150→  }, [settings]);
   151→  useEffect(() => {
   152→    leverageRef.current = leverage;
   153→  }, [leverage]);
   154→  useEffect(() => {
   155→    langRef.current = lang;
   156→  }, [lang]);
   157→  useEffect(() => {
   158→    mountedRef.current = true;
   159→    return () => {
   160→      mountedRef.current = false;
   161→    };
   162→  }, []);
   163→
   164→  const refreshAuto = useCallback(async (): Promise<AutoState | null> => {
   165→    try {
   166→      const st = await fetchAutoState();
   167→      if (mountedRef.current) setAuto(st);
   168→      return st;
   169→    } catch {
   170→      return null;
   171→    }
   172→  }, []);
   173→
   174→  // mirror for the 15s watcher (no stale closures)
   175→  const autoRef = useRef<AutoState | null>(null);
   176→  useEffect(() => {
   177→    autoRef.current = auto;
   178→  }, [auto]);
   179→
   180→  /** Close an open dry-run position — shared by the cycle and the intra-cycle watcher. */
   181→  const closeOpen = useCallback(
   182→    async (open: AutoTradeRow, price: number, kind: "sl" | "tp" | "manual"): Promise<AutoTradeRow | null> => {
   183→      if (closingRef.current) return null;
   184→      closingRef.current = true;
   185→      try {
   186→        const closed = await closeAutoTrade(open.id, price, kind);
   187→        if (closed && mountedRef.current) {
   188→          const st = await refreshAuto(); // stats now include this close
   189→          pushToast({
   190→            kind: "flip",
   191→            text: t(langRef.current, "auto.closed", { pair: closed.symbol, tf: closed.interval, pnl: (closed.pnlPct ?? 0).toFixed(2) }),
   192→          });
   193→          void sendTelegram(msgClose(closed, st?.stats.closed ?? 0, st?.stats.winRate ?? 0, st?.stats.netPnlPct ?? 0));
   194→        }
   195→        return closed;
   196→      } finally {
   197→        closingRef.current = false;
   198→      }
   199→    },
   200→    [refreshAuto, pushToast],
   201→  );
   202→
   203→  // initial load — position book + telegram configuration probe
   204→  useEffect(() => {
   205→    void refreshAuto();
   206→    void telegramConfigured().then((ok) => mountedRef.current && setTgConfigured(ok));
   207→    void telegramStatus().then((s) => mountedRef.current && setTgChatId(s.chatId));
   208→  }, [refreshAuto]);
   209→
   210→  const doScan = useCallback(async (): Promise<ScanRow[]> => {
   211→    if (scanBusyRef.current) return rowsRef.current;
   212→    scanBusyRef.current = true;
   213→    setScanning({ done: 0, total: pairsRef.current.length * 5, current: "" });
   214→    try {
   215→      const out = await scanAll({
   216→        pairs: pairsRef.current,
   217→        settings: settingsRef.current,
   218→        onProgress: (p) => mountedRef.current && setScanning(p),
   219→      });
   220→      if (mountedRef.current) {
   221→        setRows(out);
   222→        setScanAt(Date.now());
   223→      }
   224→      if (out.length > 0) {
   225→        void sendTelegram(msgScan(out, pairsRef.current.length, 5));
   226→      }
   227→      return out;
   228→    } finally {
   229→      scanBusyRef.current = false;
   230→      if (mountedRef.current) setScanning(null);
   231→    }
   232→  }, []);
   233→
   234→  // ── one auto cycle: manage the open position, then maybe enter the best pick
   235→  const runCycle = useCallback(async () => {
   236→    if (busyRef.current) return;
   237→    busyRef.current = true;
   238→    const c = cfgRef.current;
   239→    try {
   240→      const st = await refreshAuto();
   241→      if (!st) {
   242→        setStatus("DB error");
   243→        return;
   244→      }
   245→
   246→      // 1) manage the open position against the live price
   247→      if (st.open) {
   248→        const open = st.open;
   249→        const price = tickersRef.current[open.symbol]?.price ?? 0;
   250→        if (price > 0) {
   251→          const kind = slTpHit(open.side, price, open.entryPrice, open.slPct, open.tpPct);
   252→          const liq = liquidationHit(open.side, price, open.entryPrice, open.leverage);
   253→          if (kind || liq) {
   254→            await closeOpen(open, price, liq ? "sl" : (kind ?? "sl"));
   255→          } else {
   256→            const roe = roeOf(open.side, open.entryPrice, price, open.leverage);
   257→            setStatus(`${open.symbol} ${open.interval} ${open.side} · ROE ${roe >= 0 ? "+" : ""}${roe.toFixed(2)}%`);
   258→          }
   259→        }
   260→        return; // one position at a time — no new entry while one is open
   261→      }
   262→
   263→      // 2) no open position → rank the scan and enter the best tradeable row
   264→      let current = rowsRef.current;
   265→      if (current.length === 0 || Date.now() - (scanAgeRef.current ?? 0) > SCAN_TTL) {
   266→        setStatus(t(langRef.current, "auto.scan.busy", { done: 0, total: pairsRef.current.length * 5, current: "…" }));
   267→        current = await doScan();
   268→      }
   269→      if (current.length === 0) {
   270→        setStatus(t(langRef.current, "auto.scan.empty"));
   271→        return;
   272→      }
   273→      const { best, allInCooldown } = bestTradeableAvoid(current, c.minAuc, c.minAcc, st.trades, c.cooldownMin);
   274→      if (!best) {
   275→        setStatus(allInCooldown ? t(langRef.current, "auto.cooldown.all") : t(langRef.current, "auto.rank.hint"));
   276→        return;
   277→      }
   278→      const price = tickersRef.current[best.symbol]?.price ?? 0;
   279→      if (price <= 0) return; // no live price for that pair — wait for the next cycle
   280→      const capitalNow = (await refreshAuto())?.capital ?? c.capital;
   281→      const trade = await openAutoTrade({
   282→        symbol: best.symbol,
   283→        interval: best.interval,
   284→        side: best.bias,
   285→        entryPrice: price,
   286→        slPct: best.slPct,
   287→        tpPct: best.tpPct,
   288→        leverage: leverageRef.current,
   289→        prob: best.prob,
   290→        acc: best.acc,
   291→        auc: best.auc,
   292→        score: best.score,
   293→        capitalBefore: capitalNow,
   294→      });
   295→      if (trade) {
   296→        pushToast({
   297→          kind: "entry",
   298→          text: t(langRef.current, "auto.opened", { side: trade.side, pair: trade.symbol, tf: trade.interval, price: fmtPrice(trade.entryPrice) }),
   299→        });
   300→        void sendTelegram(msgOpen(trade, capitalNow));
   301→        await refreshAuto();
   302→        setStatus(`${trade.symbol} ${trade.interval} ${trade.side} @ ${fmtPrice(trade.entryPrice)}`);
   303→      }
   304→    } finally {
   305→      busyRef.current = false;
   306→      nextAtRef.current = Date.now() + cfgRef.current.cycleMin * 60_000;
   307→    }
   308→  }, [closeOpen, doScan, pushToast, refreshAuto]);
   309→
   310→  // auto cycle — immediate tick on enable, then every cycleMin minutes
   311→  useEffect(() => {
   312→    if (!cfg.on) {
   313→      setCountdown(null);
   314→      return;
   315→    }
   316→    nextAtRef.current = Date.now() + 2_000;
   317→    void runCycle();
   318→    const id = window.setInterval(() => void runCycle(), cfg.cycleMin * 60_000);
   319→    return () => window.clearInterval(id);
   320→  }, [cfg.on, cfg.cycleMin, runCycle]);
   321→
   322→  // intra-cycle watcher — SL/TP/liquidation can hit BETWEEN cycles; this books
   323→  // them within ~15s from the live tickers instead of waiting for the next
   324→  // full cycle, then kicks a fresh entry a few seconds later
   325→  useEffect(() => {
   326→    if (!cfg.on) return;
   327→    const id = window.setInterval(() => {
   328→      const open = autoRef.current?.open;
   329→      if (!open || busyRef.current || closingRef.current) return;
   330→      const price = tickersRef.current[open.symbol]?.price ?? 0;
   331→      if (price <= 0) return;
   332→      const kind = slTpHit(open.side, price, open.entryPrice, open.slPct, open.tpPct);
   333→      const liq = liquidationHit(open.side, price, open.entryPrice, open.leverage);
   334→      if (kind || liq) {
   335→        void closeOpen(open, price, liq ? "sl" : (kind ?? "sl")).then((closed) => {
   336→          if (closed && cfgRef.current.on) {
   337→            window.setTimeout(() => void runCycle(), 3_000);
   338→          }
   339→        });
   340→      }
   341→    }, 15_000);
   342→    return () => window.clearInterval(id);
   343→  }, [cfg.on, closeOpen, runCycle]);
   344→
   345→  // 1 s countdown to the next cycle
   346→  useEffect(() => {
   347→    if (!cfg.on) return;
   348→    const id = window.setInterval(() => {
   349→      const left = Math.max(0, Math.round((nextAtRef.current - Date.now()) / 1000));
   350→      setCountdown(left);
   351→    }, 1000);
   352→    return () => window.clearInterval(id);
   353→  }, [cfg.on]);
   354→
   355→  const updateCfg = (patch: Partial<AutoConfig>) => {
   356→    setCfg((prev) => {
   357→      const next = { ...prev, ...patch };
   358→      saveAutoConfig(next);
   359→      return next;
   360→    });
   361→  };
   362→
   363→  const checkTg = async () => {
   364→    setTgState("sending");
   365→    setTgDetail(null);
   366→    const r = await sendTelegram(TEST_MSG);
   367→    if (r.ok) {
   368→      setTgState("ok");
   369→      setTgHelp(false);
   370→      setTgChats(null);
   371→      pushToast({ kind: "xgb", text: t(lang, "auto.tg.sent") });
   372→    } else {
   373→      setTgState("fail");
   374→      setTgDetail(r.detail ?? null);
   375→      if (/chat not found/i.test(r.detail ?? "")) {
   376→        setTgHelp(true); // persistent in-panel guide — a toast disappears too fast
   377→        void telegramStatus().then((s) => mountedRef.current && setTgChatId(s.chatId)); // refresh possibly-stale channel label
   378→      }
   379→      pushToast({
   380→        kind: "error",
   381→        text: /chat not found/i.test(r.detail ?? "")
   382→          ? t(lang, "auto.err.chat")
   383→          : t(lang, "auto.tg.err", { detail: r.detail ?? "?" }),
   384→      });
   385→    }
   386→  };
   387→
   388→  // ask the bot which chats it can see → user picks the real channel
   389→  const discoverChats = async () => {
   390→    setTgDiscovering(true);
   391→    const d = await telegramDiscover();
   392→    if (mountedRef.current) {
   393→      setTgChats(d.chats);
   394→      if (d.bot) setTgDetail(d.bot);
   395→    }
   396→    setTgDiscovering(false);
   397→  };
   398→
   399→  // persist the picked chat id, then re-test immediately
   400→  const applyChat = async (id: string) => {
   401→    setTgSaving(true);
   402→    const saved = await telegramSaveChat(id);
   403→    if (saved && mountedRef.current) {
   404→      setTgChatId(id);
   405→      pushToast({ kind: "xgb", text: t(lang, "auto.tg.help.saved", { id }) });
   406→      setTgSaving(false);
   407→      void checkTg();
   408→      return;
   409→    }
   410→    setTgSaving(false);
   411→    pushToast({ kind: "error", text: t(lang, "auto.tg.err", { detail: "save failed" }) });
   412→  };
   413→
   414→  const closeNow = async () => {
   415→    const open = auto?.open;
   416→    if (!open || busyRef.current) return;
   417→    const price = tickers[open.symbol]?.price ?? open.entryPrice;
   418→    const closed = await closeAutoTrade(open.id, price, "manual");
   419→    if (closed) {
   420→      pushToast({
   421→        kind: "flip",
   422→        text: t(lang, "auto.closed", { pair: closed.symbol, tf: closed.interval, pnl: (closed.pnlPct ?? 0).toFixed(2) }),
   423→      });
   424→      void sendTelegram(msgClose(closed, auto?.stats.closed ?? 0, auto?.stats.winRate ?? 0, auto?.stats.netPnlPct ?? 0));
   425→      await refreshAuto();
   426→    }
   427→  };
   428→
   429→  const capitalNow = auto?.capital ?? cfg.capital;
   430→  const open = auto?.open ?? null;
   431→  const best = rows.length > 0 ? rows[0] : null;
   432→  const closedHist = (auto?.trades ?? []).filter((x) => x.status === "closed").slice(0, 4);
   433→  const sideClass = (side: string) => (side === "LONG" ? "text-emerald-400" : "text-rose-400");
   434→
   435→  return (
   436→    <Card className="border-zinc-800 bg-zinc-900/60">
   437→      <CardHeader className="pb-2">
   438→        <div className="flex flex-wrap items-center justify-between gap-2">
   439→          <CardTitle className="flex items-center gap-1.5 text-base font-semibold text-zinc-100">
   440→            <Bot className="h-4 w-4 text-teal-400" />
   441→            {t(lang, "auto.title")}
   442→            {cfg.on && <span className="ml-1 inline-block h-2 w-2 animate-pulse rounded-full bg-emerald-400" aria-hidden />}
   443→          </CardTitle>
   444→          <div className="flex items-center gap-2">
   445→            {tgConfigured != null && (
   446→              <span
   447→                className={`h-2 w-2 rounded-full ${tgConfigured ? "bg-teal-400" : "bg-zinc-600"}`}
   448→                title={tgConfigured ? t(lang, "auto.tg.ok") : t(lang, "auto.tg.none")}
   449→                aria-hidden
   450→              />
   451→            )}
   452→            <Switch checked={cfg.on} onCheckedChange={(v) => updateCfg({ on: v })} aria-label={t(lang, "auto.on")} />
   453→          </div>
   454→        </div>
   455→        <p className="mt-0.5 text-[11px] leading-relaxed text-zinc-500">{t(lang, "auto.sub")}</p>
   456→      </CardHeader>
   457→
   458→      <CardContent className="space-y-3 pt-0">
   459→        {/* config */}
   460→        <div className="flex flex-wrap items-center gap-2">
   461→          <label className="flex items-center gap-1.5 text-[11px] text-zinc-500">
   462→            {t(lang, "auto.cfg.capital")}
   463→            <Input
   464→              type="number"
   465→              min={1}
   466→              value={cfg.capital}
   467→              onChange={(e) => updateCfg({ capital: Math.max(1, Number(e.target.value) || 0) })}
   468→              className="h-7 w-24 border-zinc-800 bg-zinc-950/60 px-2 font-mono text-xs text-zinc-100"
   469→            />
   470→          </label>
   471→          <div className="flex items-center gap-1" role="group" aria-label={t(lang, "auto.cfg.cycle")}>
   472→            <span className="mr-0.5 text-[11px] text-zinc-500">{t(lang, "auto.cfg.cycle")}</span>
   473→            {CYCLES.map((m) => (
   474→              <button
   475→                key={m}
   476→                type="button"
   477→                onClick={() => updateCfg({ cycleMin: m })}
   478→                className={`h-7 rounded-md border px-2 font-mono text-[11px] ${
   479→                  cfg.cycleMin === m
   480→                    ? "border-teal-700 bg-teal-950/40 text-teal-300"
   481→                    : "border-zinc-800 bg-zinc-950/40 text-zinc-400 hover:bg-zinc-800"
   482→                }`}
