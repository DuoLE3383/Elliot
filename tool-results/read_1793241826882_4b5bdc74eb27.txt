   605→export function t(lang: Lang, key: string, vars?: Record<string, string | number>): string {
   606→  let s = DICTS[lang][key] ?? DICTS.vi[key] ?? key;
   607→  if (vars) {
   608→    for (const [k, v] of Object.entries(vars)) {
   609→      s = s.replace(`{${k}}`, String(v));
   610→    }
   611→  }
   612→  return s;
   613→}

   615→export function fmtPrice(n: number): string {
   616→  const abs = Math.abs(n);
   617→  // dynamic precision so sub-cent pairs (PEPE, SHIB…) stay readable
   618→  const max = abs >= 100 ? 2 : abs >= 1 ? 4 : abs >= 0.01 ? 5 : abs >= 0.0001 ? 6 : 8;
   619→  return new Intl.NumberFormat("en-US", {
   620→    minimumFractionDigits: 2,
   621→    maximumFractionDigits: max,
   622→  }).format(n);
   623→}

   625→/** "PEPEUSDT" → "PEPE" — base asset label for any pair chip. */
   626→export function symbolLabel(sym: string): string {
   627→  const base = sym.replace(/(USDT|USDC|FDUSD|TUSD|BUSD)$/i, "");
   628→  return base || sym;
   629→}
   630→
   631→export function fmtPct(n: number): string {
   632→  return `${n >= 0 ? "+" : ""}${n.toFixed(2)}%`;
   633→}

   635→export function fmtTime(t: number, lang: Lang): string {
   636→  return new Date(t).toLocaleString(lang === "vi" ? "vi-VN" : "en-US", {
   637→    day: "2-digit",
   638→    month: "2-digit",
   639→    hour: "2-digit",
   640→    minute: "2-digit",
   641→    hour12: false,
   642→  });
   643→}
