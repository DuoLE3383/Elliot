    63→    63→export function PriceChart({
    64→    64→  klines,
    65→    65→  sma,
    66→    66→  signals,
    67→    67→  papers,
    68→    68→  openPaper,
    69→    69→  pending,
    70→    70→  pinnedIndex,
    71→    71→  onPin,
    72→    72→  lang,
    73→    73→  wyckoff,
    74→    74→  wyckoffOn,
    75→    75→  stop,
    76→    76→  trend,
    77→    77→  fill = false,
    78→    78→  candleStyle = "neon",
    79→    79→  onCandleStyle,
    80→    80→}: Props) {
    81→    81→  const wrapRef = useRef<HTMLDivElement>(null);
    82→    82→  const plotRef = useRef<HTMLDivElement>(null);
    83→    83→  const [width, setWidth] = useState(900);
    84→    84→  const [measuredH, setMeasuredH] = useState(360);
    85→    85→  const [viewEndRaw, setViewEndRaw] = useState<number | null>(null);
    86→    86→  /** user zoom override — visible bar count (null = auto by width) */
    87→    87→  const [bars, setBars] = useState<number | null>(null);
    88→    88→  /** hover crosshair — candle index + cursor y inside the plot */
    89→    89→  const [hover, setHover] = useState<{ i: number; py: number } | null>(null);
    90→    90→  const dragRef = useRef<{ x: number; end: number } | null>(null);
    91→    91→
    92→    92→  useEffect(() => {
    93→    93→    const el = wrapRef.current;
    94→    94→    const plot = plotRef.current;
    95→    95→    if (!el) return;
    96→    96→    const ro = new ResizeObserver((entries) => {
    97→    97→      for (const en of entries) {
    98→    98→        if (en.target === el) {
    99→    99→          setWidth(Math.max(320, Math.floor(en.contentRect.width)));
   100→   100→        } else if (en.target === plot) {
   101→   101→          // floor of 160px keeps the plot usable — going lower would overflow the card
   102→   102→          setMeasuredH(Math.max(160, Math.floor(en.contentRect.height)));
   103→   103→        }
   104→   104→      }
   105→   105→    });
   106→   106→    ro.observe(el);
   107→   107→    if (plot) ro.observe(plot);
   108→   108→    return () => ro.disconnect();
   109→   109→  }, []);
   110→   110→
   111→   111→  // wheel zoom — native listener (passive: false) so preventDefault works
   112→   112→  useEffect(() => {
   113→   113→    const el = wrapRef.current;
   114→   114→    if (!el) return;
   115→   115→    const onWheel = (e: WheelEvent) => {
   116→   116→      if (!e.deltaY) return;
   117→   117→      e.preventDefault();
   118→   118→      setBars((prev) => {
   119→   119→        const base = prev ?? autoBars(width);
   120→   120→        const next = e.deltaY > 0 ? base * 1.18 : base / 1.18;
   121→   121→        return Math.max(24, Math.min(340, Math.round(next)));
   122→   122→      });
   123→   123→    };
   124→   124→    el.addEventListener("wheel", onWheel, { passive: false });
   125→   125→    return () => el.removeEventListener("wheel", onWheel);
   126→   126→  }, [width]);
   127→   127→
   128→   128→  const n = klines.length;
   129→   129→  const winSize = Math.max(20, Math.min(bars ?? autoBars(width), n || 1));
   130→   130→  const viewEnd = Math.min(viewEndRaw ?? n - 1, n - 1);
   131→   131→  const viewStart = Math.max(0, viewEnd - winSize + 1);
   132→   132→
   133→   133→  const zoom = useCallback(
   134→   134→    (dir: 1 | -1) =>
   135→   135→      setBars((prev) => {
   136→   136→        const base = prev ?? autoBars(width);
   137→   137→        const next = dir === 1 ? base / 1.35 : base * 1.35;
   138→   138→        return Math.max(24, Math.min(340, Math.round(next)));
   139→   139→      }),
   140→   140→    [width],
   141→   141→  );
   142→   142→
   143→   143→  const jumpLatest = useCallback(() => setViewEndRaw(null), []);
   144→   144→
   145→   145→  const height = fill ? measuredH : width < 520 ? 280 : 360;
   146→   146→  const plotW = width - PAD_L - PAD_R;
   147→   147→  // ── Equity pane (Đường cong vốn) — a small strip carved out of the plot bottom
   148→   148→  const eq = useMemo(() => equityFrom(papers), [papers]);
   149→   149→  const eqOn = eq.points.length >= 2;
   150→   150→  const EQ_PANE = 58;
   151→   151→  const eqPaneH = eqOn ? EQ_PANE : 0;
   152→   152→  const plotH = height - PAD_T - PAD_B - eqPaneH;
   153→   153→  const cw = plotW / Math.max(1, viewEnd - viewStart + 1);
   154→   154→
   155→   155→  const { minP, maxP } = useMemo(() => {
   156→   156→    let lo = Infinity;
   157→   157→    let hi = -Infinity;
   158→   158→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   159→   159→      const k = klines[i];
   160→   160→      if (k.l < lo) lo = k.l;
   161→   161→      if (k.h > hi) hi = k.h;
   162→   162→      const s = sma[i];
   163→   163→      if (s != null && s < lo) lo = s;
   164→   164→      if (s != null && s > hi) hi = s;
   165→   165→      const st = stop?.[i];
   166→   166→      if (st != null && st < lo) lo = st;
   167→   167→      if (st != null && st > hi) hi = st;
   168→   168→    }
   169→   169→    // NOTE: open SL/TP are deliberately NOT part of the scale — a wide 6% SL
   170→   170→    // would squash the candles. Levels are clipped to the plot instead.
   171→   171→    if (openPaper && openPaper.entry >= lo * 0.98 && openPaper.entry <= hi * 1.02) {
   172→   172→      if (openPaper.entry < lo) lo = openPaper.entry;
   173→   173→      if (openPaper.entry > hi) hi = openPaper.entry;
   174→   174→    }
   175→   175→    if (!isFinite(lo) || !isFinite(hi)) return { minP: 0, maxP: 1 };
   176→   176→    if (hi === lo) {
   177→   177→      hi += hi * 0.001 || 1;
   178→   178→      lo -= lo * 0.001 || 1;
   179→   179→    }
   180→   180→    const pad = (hi - lo) * 0.06;
   181→   181→    return { minP: lo - pad, maxP: hi + pad };
   182→   182→  }, [klines, sma, stop, viewStart, viewEnd, n, openPaper]);
   183→   183→
   184→   184→  const x = (i: number) => PAD_L + (i - viewStart) * cw + cw / 2;
   185→   185→  const y = (p: number) => PAD_T + ((maxP - p) / (maxP - minP)) * plotH;
   186→   186→  /** inverse of y() — price under the crosshair */
   187→   187→  const priceAtY = (py: number) => minP + ((PAD_T + plotH - py) / plotH) * (maxP - minP);
   188→   188→
   189→   189→  const onPointerDown = (e: React.PointerEvent<SVGSVGElement>) => {
   190→   190→    dragRef.current = { x: e.clientX, end: viewEnd };
   191→   191→    (e.target as Element).setPointerCapture?.(e.pointerId);
   192→   192→  };
   193→   193→  const onPointerMove = (e: React.PointerEvent<SVGSVGElement>) => {
   194→   194→    // hover crosshair (TradingView-style) — always tracked, pan continues on top
   195→   195→    const rect = e.currentTarget.getBoundingClientRect();
   196→   196→    const py = Math.max(PAD_T, Math.min(PAD_T + plotH, e.clientY - rect.top));
   197→   197→    const i = Math.max(viewStart, Math.min(viewEnd, viewStart + Math.floor((e.clientX - rect.left - PAD_L) / cw)));
   198→   198→    setHover((prev) => (prev && prev.i === i && Math.abs(prev.py - py) < 0.5 ? prev : { i, py }));
   199→   199→    const d = dragRef.current;
   200→   200→    if (!d) return;
   201→   201→    const shift = Math.round((d.x - e.clientX) / cw);
   202→   202→    const end = Math.min(n - 1, Math.max(winSize - 1, d.end + shift));
   203→   203→    setViewEndRaw(end >= n - 1 ? null : end);
   204→   204→  };
   205→   205→  const onPointerUp = () => {
   206→   206→    dragRef.current = null;
   207→   207→  };
   208→   208→  const onPointerLeave = () => {
   209→   209→    dragRef.current = null;
   210→   210→    setHover(null);
   211→   211→  };
   212→   212→
   213→   213→  // ── Equity pane geometry (Đường cong vốn) ──────────────────────────────
   214→   214→  const eqTop = PAD_T + plotH + 10;
   215→   215→  const eqH = eqPaneH - 16;
   216→   216→  const eqLo = Math.min(eq.trough, 0);
   217→   217→  const eqHi = Math.max(eq.peak, eqLo + 0.5);
   218→   218→  const eqY = (v: number) => eqTop + ((eqHi - v) / (eqHi - eqLo)) * eqH;
   219→   219→  const eqX = (bar: number) => Math.max(PAD_L, Math.min(PAD_L + plotW, x(bar)));
   220→   220→  // trades closed up to the viewed bar — the curve stays honest while panning back
   221→   221→  const eqSeen = eqOn ? eq.points.filter((p) => p.bar <= viewEnd) : [];
   222→   222→  const eqLine = (() => {
   223→   223→    if (!eqOn || eqH <= 4) return null;
   224→   224→    if (eqSeen.length === 0) {
   225→   225→      const d = `M${PAD_L},${eqY(0).toFixed(1)} L${(PAD_L + plotW).toFixed(1)},${eqY(0).toFixed(1)}`;
   226→   226→      return { d, area: `${d} Z`, endV: 0, endY: eqY(0) };
   227→   227→    }
   228→   228→    const pts = eqSeen;
   229→   229→    let d = pts[0].bar > viewStart ? `M${PAD_L},${eqY(0).toFixed(1)} L${eqX(pts[0].bar).toFixed(1)},${eqY(pts[0].cum).toFixed(1)}` : `M${eqX(pts[0].bar).toFixed(1)},${eqY(pts[0].cum).toFixed(1)}`;
   230→   230→    for (let k = 1; k < pts.length; k++) d += ` L${eqX(pts[k].bar).toFixed(1)},${eqY(pts[k].cum).toFixed(1)}`;
   231→   231→    const endY = eqY(pts[pts.length - 1].cum);
   232→   232→    d += ` L${(PAD_L + plotW).toFixed(1)},${endY.toFixed(1)}`; // flat extension to "now"
   233→   233→    const area = `${d} L${(PAD_L + plotW).toFixed(1)},${eqY(0).toFixed(1)} L${PAD_L},${eqY(0).toFixed(1)} Z`;
   234→   234→    return { d, area, endV: pts[pts.length - 1].cum, endY };
   235→   235→  })();
   236→   236→
   237→   237→  const sigInView = signals.filter((s) => s.i >= viewStart && s.i <= viewEnd);
   238→   238→  const eqColor = eqLine ? (eqLine.endV >= 0 ? UP : DOWN) : UP;
   239→   239→  const papersInView = papers.filter(
   240→   240→    (p) => p.exitIndex !== undefined ? p.exitIndex >= viewStart || p.entryIndex <= viewEnd : p.entryIndex <= viewEnd,
   241→   241→  );
   242→   242→
   243→   243→  // ── Volume profile over the visible window (buy/sell split via taker data) ─
   244→   244→  const vp = useMemo(() => volumeProfile(klines, viewStart, viewEnd, 26), [klines, viewStart, viewEnd]);
   245→   245→  const vpW = plotW * 0.16;
   246→   246→
   247→   247→  // ── Whale delta histogram over the visible window ────────────────────────
   248→   248→  const whale = useMemo(() => {
   249→   249→    let maxD = 0;
   250→   250→    const rows: { i: number; d: number }[] = [];
   251→   251→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   252→   252→      const k = klines[i];
   253→   253→      if (k.tb == null || k.v <= 0) continue;
   254→   254→      const d = 2 * k.tb - k.v;
   255→   255→      rows.push({ i, d });
   256→   256→      if (Math.abs(d) > maxD) maxD = Math.abs(d);
   257→   257→    }
   258→   258→    return { rows, maxD };
   259→   259→  }, [klines, viewStart, viewEnd, n]);
   260→   260→  const whaleH = plotH * 0.2;
   261→   261→
   262→   262→  // ── Wyckoff layers (only when enabled) ─────────────────────────────────
   263→   263→  const wyOn = !!wyckoffOn && !!wyckoff;
   264→   264→  const wyBoxes = wyOn
   265→   265→    ? wyckoff!.boxes.filter((b) => !b.deleted && b.right >= viewStart && b.left <= viewEnd)
   266→   266→    : [];
   267→   267→  const wyEvents = wyOn
   268→   268→    ? wyckoff!.events.filter((e) => !e.deleted && e.bar >= viewStart && e.bar <= viewEnd)
   269→   269→    : [];
   270→   270→  const wyPivots = wyOn
   271→   271→    ? wyckoff!.pivots.filter((p) => !p.hidden && p.bar >= viewStart && p.bar <= viewEnd)
   272→   272→    : [];
   273→   273→  const tintOf = (i: number): 0 | 1 | -1 => (wyOn ? (wyckoff!.tints[i] as 0 | 1 | -1) : 0);
   274→   274→
   275→   275→  /** polyline path skipping nulls (EMA arrays) */
   276→   276→  const linePath = (arr: (number | null)[]): string => {
   277→   277→    let d = "";
   278→   278→    let pen = false;
   279→   279→    for (let i = viewStart; i <= viewEnd; i++) {
   280→   280→      const v = arr[i];
   281→   281→      if (v == null) {
   282→   282→        pen = false;
   283→   283→        continue;
   284→   284→      }
   285→   285→      d += `${pen ? "L" : "M"}${x(i)},${y(v)} `;
   286→   286→      pen = true;
   287→   287→    }
   288→   288→    return d.trim();
   289→   289→  };
   290→   290→
   291→   291→  /** SMA ± band filled area — one closed subpath per contiguous SMA run */
   292→   292→  const bandPath = (): string => {
   293→   293→    let d = "";
   294→   294→    let top: string[] = [];
   295→   295→    let bot: string[] = [];
   296→   296→    const flush = () => {
   297→   297→      if (top.length >= 2) {
   298→   298→        d += `M${top.join(" L")} L${bot.reverse().join(" L")} Z `;
   299→   299→      }
   300→   300→      top = [];
   301→   301→      bot = [];
   302→   302→    };
   303→   303→    for (let i = viewStart; i <= viewEnd; i++) {
   304→   304→      const v = sma[i];
   305→   305→      if (v == null) {
   306→   306→        flush();
   307→   307→        continue;
   308→   308→      }
   309→   309→      top.push(`${x(i)},${y(v * (1 + BAND_PCT))}`);
   310→   310→      bot.push(`${x(i)},${y(v * (1 - BAND_PCT))}`);
   311→   311→    }
   312→   312→    flush();
   313→   313→    return d.trim();
   314→   314→  };
   315→   315→
   316→   316→  /** UT trailing-stop segments, colored per trend run */
   317→   317→  const stopSegs = (): { d: string; up: boolean }[] => {
   318→   318→    const segs: { d: string; up: boolean }[] = [];
   319→   319→    let cur: { d: string; up: boolean } | null = null;
   320→   320→    for (let i = viewStart; i <= viewEnd; i++) {
   321→   321→      const v = stop?.[i];
   322→   322→      const tr = trend?.[i] ?? 0;
   323→   323→      if (v == null || tr === 0) {
   324→   324→        cur = null;
   325→   325→        continue;
   326→   326→      }
   327→   327→      const up = tr === 1;
   328→   328→      if (cur && cur.up === up) cur.d += ` L${x(i)},${y(v)}`;
   329→   329→      else {
   330→   330→        cur = { d: `M${x(i)},${y(v)}`, up };
   331→   331→        segs.push(cur);
   332→   332→      }
   333→   333→    }
   334→   334→    return segs;
   335→   335→  };
   336→   336→
   337→   337→  const timeLabel = (t: number) => {
   338→   338→    const d = new Date(t);
   339→   339→    const hh = String(d.getHours()).padStart(2, "0");
   340→   340→    const mm = String(d.getMinutes()).padStart(2, "0");
   341→   341→    return `${d.getDate()}/${d.getMonth() + 1} ${hh}:${mm}`;
   342→   342→  };
   343→   343→
   344→   344→  if (n === 0) {
   345→   345→    return (
   346→   346→      <div
   347→   347→        ref={wrapRef}
   348→   348→        className={`flex items-center justify-center rounded-lg border border-zinc-800 bg-zinc-950 text-sm text-zinc-500 ${fill ? "h-full" : "h-64"}`}
   349→   349→      >
   350→   350→        {t(lang, "chart.empty")}
   351→   351→      </div>
   352→   352→    );
   353→   353→  }
   354→   354→
   355→   355→  const gridPrices = Array.from({ length: 5 }, (_, k) => minP + ((maxP - minP) * k) / 4);
   356→   356→  const openEntryLabel = openPaper?.flippedFrom ? t(lang, "chart.line.flipEntry") : t(lang, "chart.line.entry");
   357→   357→
   358→   358→  return (
   359→   359→    <div ref={wrapRef} className={`relative flex select-none flex-col ${fill ? "h-full" : ""}`}>
   360→   360→      <div ref={plotRef} className={`relative min-h-0 overflow-hidden ${fill ? "flex-1" : ""}`}>
   361→   361→        <svg
   362→   362→          width={width}
   363→   363→          height={height}
   364→   364→          className="touch-pan-y rounded-lg border border-zinc-800 bg-zinc-950"
   365→   365→          onPointerDown={onPointerDown}
   366→   366→          onPointerMove={onPointerMove}
   367→   367→          onPointerUp={onPointerUp}
   368→   368→          onPointerLeave={onPointerLeave}
   369→   369→          role="img"
   370→   370→          aria-label={t(lang, "chart.title")}
   371→   371→        >
   372→   372→          <defs>
   373→   373→            <clipPath id="wy-plot-clip">
   374→   374→              <rect x={PAD_L} y={PAD_T} width={plotW} height={plotH} />
   375→   375→            </clipPath>
   376→   376→          </defs>
   377→   377→          {/* grid + price axis */}
   378→   378→          {gridPrices.map((p, k) => (
   379→   379→            <g key={k}>
   380→   380→              <line x1={PAD_L} x2={PAD_L + plotW} y1={y(p)} y2={y(p)} stroke={GRID} strokeWidth={1} />
   381→   381→              <text x={PAD_L + plotW + 6} y={y(p) + 4} fill={TEXT} fontSize={10} fontFamily="ui-monospace, monospace">
   382→   382→                {fmtPrice(p)}
   383→   383→              </text>
   384→   384→            </g>
   385→   385→          ))}
   386→   386→          {/* time axis — edge labels anchor inward, greedy de-dupe avoids collisions */}
   387→   387→          {(() => {
   388→   388→            const last = width < 520 ? 3 : 5;
   389→   389→            const est = (s: string) => s.length * 6.2; // ~px per char @ 10px mono
   390→   390→            let prevRight = -Infinity;
   391→   391→            const nodes: React.ReactNode[] = [];
   392→   392→            for (let k = 0; k <= last; k++) {
   393→   393→              const idx = Math.round(viewStart + ((viewEnd - viewStart) * k) / last);
   394→   394→              const kk = klines[Math.min(idx, n - 1)];
