     1→// ─── engine — the backend auto engine (scan · train · universe · broadcast ·
     2→// cycle · digests · resolve) ──────────────────────────────────────────────────
     3→// Everything heavy that used to live in the BROWSER (auto-panel metronomes,
     4→// the XGB worker pool, the paper-trade cycle runner) now runs HERE, in the
     5→// standalone Node process, 24/7 with no tab open:
     6→//
     7→//   • every 20s  — watcher: SL/TP/liquidation of the open paper position +
     8→//                  one-shot "sát TP/SL" warnings (was the intra-cycle interval)
     9→//   • every 60s  — main tick:
    10→//       · hourly/daily Telegram digests (marker-guarded, cfg-gated)
    11→//       · 5-min bucket → refresh scan if stale + TOP-coin broadcast
    12→//       · 8h bucket   → universe re-rank + broadcast
    13→//       · cycle due   → paper-trade cycle (manage open position / enter best)
    14→//       · every 5 min → resolve pending suggestions against live prices
    15→//
    16→// Heavy combos (scan/train) run through XgbPoolNode — worker_threads, real
    17→// parallelism on the VPS. Job lanes (engine-jobs) give the "nhiều người bấm →
    18→// bắt chờ" semantics: one lane per kind, FIFO, identical queued jobs coalesce.
    19→//

    21→// stays the editing surface) via POST /api/engine {action:"sync"} and persisted
    22→// to $DATA_DIR/engine.json — so a reboot resumes the exact same engine. Until
    23→// the first sync the scheduler stays idle (a fresh deploy must not start
    24→// scanning the whole universe uninvited).

   161→function loadEngine(): void {
   162→  if (g.__elliottEngine?.loaded) return;
   163→  const d = defaults();
   164→  g.__elliottEngine = { ...d, loaded: true };
   165→  // last scan/universe snapshots from disk — a pm2 restart no longer blanks
   166→  // the panel with "No scan data yet" until the next 5-min bucket re-scans
   167→  Object.assign(g.__elliottEngine.v, loadScanState());
   168→  try {
   169→    const file = dataFile("engine.json");
   170→    if (existsSync(file)) {
   171→      const raw = JSON.parse(readFileSync(file, "utf8")) as Partial<EnginePersisted>;
   172→      const p = g.__elliottEngine.p;
   173→      if (raw.cfg) {
   174→        p.cfg = { ...AUTO_DEFAULTS, ...raw.cfg };
   175→        // ngưỡng chung — kẹp + migration legacy 0.55/55 (thời "no UI") trong một nguồn
   176→        Object.assign(p.cfg, sanitizeFloors(p.cfg.minAuc, p.cfg.minAcc, p.cfg.minScore));
   177→        p.cfg.runTf = sanitizeRunTf(p.cfg.runTf);
   178→      }
   179→      if (raw.settings) p.settings = { ...DEFAULT_SETTINGS, ...raw.settings };
   180→      if (Array.isArray(raw.pairs)) p.pairs = raw.pairs.filter((x) => typeof x === "string").slice(0, 200);
   181→      if (typeof raw.leverage === "number" && raw.leverage >= 1) p.leverage = raw.leverage;
   182→      if (typeof raw.syncedAt === "number") p.syncedAt = raw.syncedAt;
   183→      if (typeof raw.nextCycleAt === "number") p.nextCycleAt = raw.nextCycleAt;
   184→      if (raw.warned && typeof raw.warned === "object") p.warned = raw.warned;
   185→      if (raw.dca && typeof raw.dca === "object" && typeof raw.dca.id === "string") {
   186→        p.dca = {
   187→          id: raw.dca.id,
   188→          count: typeof raw.dca.count === "number" ? Math.max(0, raw.dca.count) : 0,
   189→          stake0: typeof raw.dca.stake0 === "number" && raw.dca.stake0 > 0 ? raw.dca.stake0 : 0,
   190→          prices: Array.isArray(raw.dca.prices) ? raw.dca.prices.filter((x) => typeof x === "number" && x > 0) : [],
   191→        };
   192→      }
   193→      if (raw.llmLast) p.llmLast = raw.llmLast;
   194→    }
   195→  } catch {
   196→    // corrupt state — start from defaults, the next sync rewrites it
   197→  }
   198→}

   220→  const { p, v } = eng();

   226→    const perp = aliasFuturesSymbol(name) ?? (/^[A-Z0-9]{5,20}$/.test(name) ? name : null);

   228→    const row = board.get(perp);

   230→      map[name] = { symbol: name, price: row.price, changePct: row.changePct, high: row.high, low: row.low, quoteVolume: row.quoteVolume };

   232→    }
   233→  }
   234→  v.tickers = map;
   235→  v.tickersAt = Date.now();
   236→  return map;
   237→}

   304→/** Close the open paper position — same math as PATCH /api/autotrade close.
   305→ *  Sau khi chốt lời/cắt lỗ, lệnh được GIẢI PHẪU (autopsy — cả thắng lẫn thua)
   306→ *  fire-and-forget: không chặn cycle/watcher, lỗi chỉ log. Bài học nằm lại DB,
   307→ *  nạp lại vào lần duyệt lệnh sau của cùng cặp. exitKind "flip" = đóng vì
   308→ *  AUTO-FLIP (model đảo chiều đủ tin cậy — xem tryFlip). */
   309→async function closeTradeDirect(
   310→  id: string,
   311→  exitPrice: number,
   312→  exitKind: "tp" | "sl" | "manual" | "flip",
   313→): Promise<AutoTradeRow | null> {
   314→  const row = await db.autoTrade.findUnique({ where: { id } });
   315→  if (!row || row.status !== "open") return null;
   316→  const movePct =
   317→    row.side === "SHORT" ? ((row.entryPrice - exitPrice) / row.entryPrice) * 100 : ((exitPrice - row.entryPrice) / row.entryPrice) * 100;
   318→  let pnlPct = Math.round(movePct * row.leverage * 100) / 100;
   319→  const adverse = row.side === "SHORT" ? -movePct : movePct;
   320→  if (row.leverage > 1 && adverse <= -(100 / row.leverage)) pnlPct = -100;
   321→  const capitalBefore = row.capitalBefore ?? 1000;
   322→  const stakePct = row.stakePct ?? 100;
   323→  const delta = Math.round(capitalBefore * (stakePct / 100) * (pnlPct / 100) * 100) / 100;
   324→  const capitalAfter = Math.max(0, Math.round((capitalBefore + delta) * 100) / 100);
   325→  const updated = await db.autoTrade.update({
   326→    where: { id },
   327→    data: { status: "closed", exitPrice, exitTime: new Date(), pnlPct, exitKind, capitalAfter },
   328→  });
   329→  if (pval("AUTOPSY_ON", AUTOPSY_ON)) {
   330→    const { p } = eng();
   331→    void (() => { const ep = resolveLlmEndpoint({ url: p.cfg.llmUrl, model: p.cfg.llmModel }); void autopsyClosedTrade(updated, { url: ep.url, model: ep.model }); })();
   332→  }
   333→  return serialize(updated);
   334→}

   373→async function scanCore(
   374→  report: (patch: { progress?: { done: number; total: number; current: string }; extra?: Record<string, unknown> }) => void,
   375→  opts?: { pairs?: string[]; intervals?: string[] },
   376→): Promise<Record<string, unknown>> {
   377→  const { p, v } = eng();
   378→  const pairs = opts?.pairs ?? p.pairs;
   379→  const intervals = opts?.intervals ?? p.cfg.runTf;
   380→  const pool = new XgbPoolNode();
   381→  const combos = pairs.length * intervals.length;
   382→  const collected: ScanRow[] = [];
   383→  let done = 0;
   384→  report({ progress: { done: 0, total: combos, current: "" }, extra: { workers: pool.parallelism } });
   385→  try {
   386→    await pool.runQueue(
   387→      pairs.flatMap((sym) => intervals.map((itv) => ({ sym, itv, difficulty: 0 }))), // fair race — difficulty pinned 0
   388→      p.settings,
   389→      {
   390→        onComboStart: (sym, itv) => report({ progress: { done, total: combos, current: `${sym} ${itv}` } }),
   391→        onComboDone: (sym, itv, res) => {
   392→          collected.push(rowOf(res, sym, itv));
   393→          done += 1;
   394→          report({ progress: { done, total: combos, current: "" } });
   395→        },
   396→        onComboFail: () => {
   397→          done += 1;
   398→          report({ progress: { done, total: combos, current: "" } });
   399→        },
   400→      },
   401→    );
   402→  } finally {
   403→    pool.destroy();
   404→  }
   405→  // confluence — các model cùng symbol vote chung bằng log-odds (autotrade.ts):
   406→  // khung giờ đồng thuận → prob/score đội lên, lệch nhau → xả về 0.5 → NEUTRAL
   407→  const blended = confluenceBlend(collected);
   408→  const nBlend = blended.filter((r, i) => r.prob !== collected[i].prob).length;
   409→  if (nBlend > 0) console.log(`[engine] confluence — ${nBlend}/${blended.length} row ghép phiếu đa khung cùng symbol`);
   410→  const ranked = rankRows(blended);
   411→  // champion map — best-ever run per combo from the persisted history; a fresh
   412→  // retrain that rolled measurably worse (auc < champ − 0.02) is flagged and
   413→  // the auto-cycle skips it ("mỗi lần train phải tốt hơn" — no gambling on a
   414→  // bad roll). One query, computed in JS — no schema change.
   415→  try {
   416→    const runs = await db.finetuneRun.findMany({
   417→      select: { symbol: true, interval: true, acc: true, auc: true },
   418→      orderBy: { createdAt: "desc" },
   419→      take: 4000,
   420→    });
   421→    const champ = new Map<string, { auc: number; acc: number }>();
   422→    for (const r of runs) {
   423→      const k = `${r.symbol}|${r.interval}`;
   424→      const cur = champ.get(k);
   425→      if (!cur || r.auc > cur.auc || (r.auc === cur.auc && r.acc > cur.acc)) champ.set(k, { auc: r.auc, acc: r.acc });
   426→    }
   427→    for (const row of ranked) {
   428→      const c = champ.get(`${row.symbol}|${row.interval}`);
   429→      if (c) Object.assign(row, withChampion(row, c.auc, c.acc));
   430→    }
   431→    console.log(`[engine] scan enriched — ${ranked.filter((r) => r.regressed).length}/${ranked.length} regressed vs champion`);
   432→  } catch {
   433→    // no history yet → rows stay unflagged
   434→  }
   435→  v.scan = { rows: ranked, at: Date.now() };
   436→  saveScanState({ scan: v.scan });
   437→  // đếm tích luỹ — nền, không chặn/không hỏng quét nếu DB chớp
   438→  void bumpSignalTotals(ranked);
   439→  return { rows: ranked.length, workers: pool.parallelism };
   440→}

   442→async function trainCore(
   443→  report: (patch: { progress?: { done: number; total: number; current: string }; extra?: Record<string, unknown> }) => void,
   444→  opts: { mode: "all" } | { mode: "single"; sym: string; itv: string; persist: boolean },
   445→): Promise<Record<string, unknown>> {
   446→  const { p } = eng();
   447→  const tfs = p.cfg.runTf; // chỉ train các khung đang bật
   448→  const list =
   449→    opts.mode === "all" ? p.pairs.flatMap((sym) => tfs.map((itv) => ({ sym, itv }))) : [{ sym: opts.sym, itv: opts.itv }];
   450→  if (list.length === 0) return { saved: 0, total: 0 };
   451→
   452→  const pool = new XgbPoolNode();
   453→  const combos = list.length;
   454→  let done = 0;
   455→  let saved = 0;
   456→  const ranked: { sym: string; itv: string; res: XgbResult }[] = [];
   457→  let best: { symbol: string; interval: string; acc: number; auc: number; score: number } | null = null;
   458→  let lastResult: Record<string, unknown> | null = null;
   459→  const lessonTotals = { n: 0, wins: 0, agree: 0 };
   460→  // pre-resolve every combo's difficulty from its persisted run count — the
   461→  // worker needs it up-front (browser pool used to fetch it itself)
   462→  const diffs = new Map<string, number>();
   463→  for (const s of list) diffs.set(`${s.sym}|${s.itv}`, await runCount(s.sym, s.itv));
   464→  // win/lose lessons — closed dry-run trades per combo, newest first (capped
   465→  // per combo: recent trades matter, a year-old roll doesn't)
   466→  const tradeMap = new Map<string, TradeLesson[]>();
   467→  try {
   468→    const closed = await db.autoTrade.findMany({
   469→      where: { status: "closed" },
   470→      orderBy: { exitTime: "desc" },
   471→      take: 500,
   472→    });
   473→    for (const t of closed) {
   474→      if (!t.exitTime) continue;
   475→      const key = `${t.symbol}|${t.interval}`;
   476→      const lst = tradeMap.get(key) ?? [];
   477→      if (lst.length < 40) {
   478→        lst.push({
   479→          t: new Date(t.entryTime).getTime(),
   480→          side: t.side === "SHORT" ? "SHORT" : "LONG",
   481→          pnl: t.pnlPct ?? 0,
   482→        });
   483→      }
   484→      tradeMap.set(key, lst);
   485→    }
   486→    if (tradeMap.size > 0) console.log(`[engine] trade lessons — ${[...tradeMap.values()].reduce((a, x) => a + x.length, 0)} closed trades across ${tradeMap.size} combo(s)`);
   487→  } catch {
   488→    // no trade history — trains run on klines only, as before
   489→  }
   490→  report({ progress: { done: 0, total: combos, current: "" }, extra: { workers: pool.parallelism } });
   491→  try {
   492→    await pool.runQueue(
   493→      list.map((s) => ({
   494→        sym: s.sym,
   495→        itv: s.itv,
   496→        difficulty: diffs.get(`${s.sym}|${s.itv}`) ?? 0,
   497→        trades: tradeMap.get(`${s.sym}|${s.itv}`),
   498→      })),
   499→      p.settings,
   500→      {
   501→        onComboStart: (sym, itv) => report({ progress: { done, total: combos, current: `${sym} ${itv}` } }),
   502→        onComboDone: async (sym, itv, res, entryPrice) => {
   503→          if (opts.mode === "single" && !opts.persist) {
   504→            // silent auto-retrain (pair/interval switch) — compute only, the
   505→            // suggestion history stays user-driven (same as the old client flow)
   506→            lastResult = { res, entryPrice };
   507→            done += 1;
   508→            report({ progress: { done, total: combos, current: "" } });
   509→            return;
   510→          }
   511→          try {
   512→            const ids = await persistTrainResult(sym, itv, res, entryPrice, diffs.get(`${sym}|${itv}`) ?? 0);
   513→            saved += 1;
   514→            ranked.push({ sym, itv, res });
   515→            if (res.lessons) {
   516→              lessonTotals.n += res.lessons.n;
   517→              lessonTotals.wins += res.lessons.wins;
   518→              lessonTotals.agree += res.lessons.agree;
   519→            }
   520→            if (opts.mode === "single") lastResult = { res, entryPrice, ...ids };
   521→          } catch {
   522→            // persist failure → combo counts as done but not saved
   523→            if (opts.mode === "single") lastResult = { res, entryPrice };
   524→          }
   525→          done += 1;
   526→          report({ progress: { done, total: combos, current: "" } });
   527→        },
   528→        onComboFail: () => {
   529→          done += 1;
   530→          report({ progress: { done, total: combos, current: "" } });
   531→        },
   532→      },
   533→    );
   534→  } finally {
   535→    pool.destroy();
   536→  }
   537→  for (const r of ranked) {
   538→    const sc = scoreOf(r.res);
   539→    if (!best || sc > best.score) best = { symbol: r.sym, interval: r.itv, acc: r.res.acc, auc: r.res.auc, score: sc };
   540→  }
   541→  if (saved > 0) invalidateTrainSnapshot(); // panel train-history refreshes on the next poll
   542→  // 0.2.36 — Strategy Lab TỰ chạy sau mỗi lượt train có lưu: sinh + backtest
   543→  // lại ứng viên chiến lược (baseline cfg hiện tại + grid + biến thể theo
   544→  // feature gain của champion). maybeAutoLab tự throttle 12h/combo nên pass
   545→  // định kỳ không spam; chạy NỀN tuần tự, không chặn trả kết quả train.
   546→  if (saved > 0) {
   547→    void (async () => {
   548→      for (const r of ranked) {
   549→        await maybeAutoLab(r.sym, r.itv, labCfgOf(p.settings));
   550→      }
   551→      console.log(`[engine] strategy lab nền xong cho ${ranked.length} combo đã lưu`);
   552→    })();
   553→  }
   554→  // channel summary for a full pass — same message the client trainer used to send
   555→  if (opts.mode === "all" && saved > 0 && tgReady()) {
   556→    try {
   557→      const lines = [
   558→        "🧠 <b>FULL TRAIN COMPLETE</b>",
   559→        `${p.pairs.length} pair × ${p.cfg.runTf.length} TF (${p.cfg.runTf.join(",")}) · saved <b>${saved}/${combos}</b> combos`,
   560→        best
   561→          ? `🏆 Best: <b>#${best.symbol}</b> ${best.interval} · ACC ${best.acc.toFixed(1)}% · AUC ${best.auc.toFixed(2)} · Score ${best.score.toFixed(1)}`
   562→          : "",
   563→        lessonTotals.n > 0
   564→          ? `🎓 Learned from <b>${lessonTotals.n}</b> real dry-run trades (${lessonTotals.wins} win · ${lessonTotals.n - lessonTotals.wins} lose) — model agrees on direction ${lessonTotals.agree}/${lessonTotals.n}`
   565→          : "",
   566→        `⚙️ ${pool.parallelism} backend workers${xgbPoolAvailable() ? "" : " (sequential fallback)"}`,
   567→      ].filter(Boolean);
   568→      void tgSendMessage(lines.join("\n"), "HTML");
   569→    } catch {
   570→      // summary is best-effort
   571→    }
   572→  }
   573→  return { saved, total: combos, best, ...(opts.mode === "single" && lastResult ? { result: lastResult } : {}) };
   574→}

   582→export interface TrainRunLite {
   583→  symbol: string;
   584→  interval: string;
   585→  kind: string;
   586→  acc: number;
   587→  auc: number;
   588→  at: number;
   589→  /** this run is its combo's champion (best-ever AUC → ACC) */
   590→  best: boolean;

   593→export interface TrainSnapshot {
   594→  at: number;
   595→  total: number;
   596→  latest: TrainRunLite[];
   597→}

   599→let trainCache: TrainSnapshot | null = null;
   600→
   601→async function trainSnapshot(): Promise<TrainSnapshot> {
   602→  if (trainCache && Date.now() - trainCache.at < 60_000) return trainCache;
   603→  try {
   604→    const [total, runs, hist] = await Promise.all([
   605→      db.finetuneRun.count(),
   606→      db.finetuneRun.findMany({ orderBy: { createdAt: "desc" }, take: 12 }),
   607→      db.finetuneRun.findMany({
   608→        select: { symbol: true, interval: true, acc: true, auc: true },
   609→        orderBy: { createdAt: "desc" },
   610→        take: 4000,
   611→      }),
   612→    ]);
   613→    const champ = new Map<string, { auc: number; acc: number }>();
   614→    for (const r of hist) {
   615→      const k = `${r.symbol}|${r.interval}`;
   616→      const cur = champ.get(k);
   617→      if (!cur || r.auc > cur.auc || (r.auc === cur.auc && r.acc > cur.acc)) champ.set(k, { auc: r.auc, acc: r.acc });
   618→    }
   619→    const latest: TrainRunLite[] = runs.map((r) => {
   620→      const c = champ.get(`${r.symbol}|${r.interval}`);
   621→      return {
   622→        symbol: r.symbol,
   623→        interval: r.interval,
   624→        kind: r.kind,
   625→        acc: r.acc,
   626→        auc: r.auc,
   627→        at: new Date(r.createdAt).getTime(),
   628→        best: c != null && r.auc >= c.auc && r.acc >= c.acc,
   629→      };
   630→    });
   631→    trainCache = { at: Date.now(), total, latest };
   632→  } catch {
   633→    // DB down — keep whatever cache exists (or the empty snapshot)
   634→    if (!trainCache) trainCache = { at: 0, total: 0, latest: [] };
   635→  }
   636→  return trainCache;
   637→}

   639→function invalidateTrainSnapshot(): void {
   640→  trainCache = null;
   641→  void trainSnapshot();
   642→}

  1011→  row: ScanRow,

  1013→  slPct: number,
  1014→  tpPct: number,
  1015→  track?: string,

  1017→  const { p, v } = eng();

  1020→  let lastVerdict: LlmVerdict | null = null;
  1021→  try {

  1023→    // trước, ≤5 bài) đi kèm setup — Risk Officer phán trên hậu quả thật.
  1024→    const lessons = pval("AUTOPSY_ON", AUTOPSY_ON) ? await recentAutopsyLessons(row.symbol).catch(() => []) : [];
  1025→    if (lessons.length > 0) console.log(`[engine] LLM review ${row.symbol} ${row.interval} — nạp ${lessons.length} bài học autopsy`);
  1026→    const failClosed = c.llmFailClosed === true;
  1027→    // Cap at 90s by default — user reports reviews finish in ~1 min; 300s only
  1028→    // masked hangs from json_schema / wrong model id.
  1029→    const timeoutMs = Math.max(15_000, Math.min(180_000, (c.llmTimeoutSec || 90) * 1000));
  1030→    // SAME URL resolution as /ai ping — env wins so "Connected" matches review
  1031→    const ep = resolveLlmEndpoint({ url: c.llmUrl, model: c.llmModel });
  1032→    const verdict = await llmReview(
  1033→      ep.url,
  1034→      ep.model,
  1035→      {

  1037→        interval: row.interval,
  1038→        side,
  1039→        prob: row.prob,

  1041→        auc: row.auc,
  1042→        acc: row.acc,
  1043→        slPct,
  1044→        tpPct,
  1045→        leverage: p.leverage,
  1046→        stakePct: c.riskPct,
  1047→        horizon: row.horizon,
  1048→        samples: row.samples,
  1049→        score: row.score,
  1050→        gatePassed: true, // caller already passed gate + noise band
  1051→        track,
  1052→        ...(row.champAuc ? { champAuc: row.champAuc, champAcc: row.champAcc } : {}),
  1053→        lessons: lessons.length > 0 ? lessons : undefined,
  1054→      },
  1055→      timeoutMs,
  1056→    );

  1060→        symbol: row.symbol,
  1061→        interval: row.interval,

  1063→        verdict: verdict.verdict,
  1064→        reason: verdict.reasons[0] ?? "",
  1065→        ms: verdict.ms ?? 0,
  1066→        at: Date.now(),
  1067→      };

  1069→    } else {
  1070→      p.llmLast = {
  1071→        symbol: row.symbol,
  1072→        interval: row.interval,
  1073→        side,
  1074→        verdict: "unreachable",
  1075→        reason: failClosed
  1076→          ? "llama-server unreachable — fail-CLOSED blocks this entry"
  1077→          : "llama-server unreachable — fail-OPEN lets XGB entry through (default)",
  1078→        ms: timeoutMs,
  1079→        at: Date.now(),
  1080→      };
  1081→      persistEngine();
  1082→      if (failClosed) {
  1083→        vetoed = true;
  1084→        v.lastCycleStatus = `LLM unreachable — fail-closed blocks ${row.symbol} ${row.interval}`;
  1085→      }
  1086→    }
  1087→    if (verdict && verdict.verdict === "skip") {
  1088→      vetoed = true;
  1089→      v.lastCycleStatus = `LLM veto ${row.symbol} ${row.interval}`;
  1090→      if (tgReady()) void tgSendMessage(msgLlmVeto(row, verdict), "HTML");
  1091→    }
  1092→  } catch {
  1093→    if (c.llmFailClosed === true) {
  1094→      vetoed = true;
  1095→      v.lastCycleStatus = `LLM error — fail-closed blocks ${row.symbol} ${row.interval}`;
  1096→    }
  1097→  }
  1098→  return { verdict: lastVerdict, vetoed };
  1099→}

  1113→      const price = tickers[open.symbol]?.price ?? 0;

  1124→            }

  1126→          return; // one position at a time — the next cycle re-evaluates entry
  1127→        }
  1128→        const roe = roeOf(open.side, open.entryPrice, price, open.leverage);
  1129→        const base = `${open.symbol} ${open.interval} ${open.side} · ROE ${roe >= 0 ? "+" : ""}${roe.toFixed(2)}%`;

  1131→        // scan (5 phút/lần) thấy tín hiệu NGƯỢC phía lệnh đủ tin cậy trên

  1133→        // (flipDecision quyết định "đủ tin cậy"; tryFlip thực hiện đóng/mở).
  1134→        if (!c.autoFlip) {

  1139→        const row = scan?.rows.find((r) => r.symbol === open.symbol && r.interval === open.interval);
  1140→        const d = flipDecision(open, row, c);

  1143→          // trả xong là chu kỳ này xong, lệnh mới sống độc lập từ chu kỳ sau

  1151→      }

  1155→    // 2) no open position → rank the scan and enter the best tradeable row
  1156→    let rows = v.scan?.rows ?? [];
  1157→    if (rows.length === 0 || Date.now() - (v.scan?.at ?? 0) > SCAN_TTL) {
  1158→      v.lastCycleStatus = "scanning…";

  1172→      rows,
  1173→      c.minAuc,
  1174→      c.minAcc,
  1175→      st.trades,
  1176→      c.cooldownMin,
  1177→      c.whaleFilter,
  1178→      pval("NOISE_BAND", NOISE_BAND),
  1179→      c.minScore,
  1180→    );
  1181→    // walk the ranked candidates: regressed/cooldown-blocked combos are skipped,
  1182→    // then the chase guard rejects same-side re-entries until price has reversed
  1183→    // back through the level that stopped us ("SL xong vào lại giá cao hơn" → hết)
  1184→    const tickers = await getTickers().catch(() => ({}) as Record<string, Ticker>);
  1185→    let best: ScanRow | null = null;

  1190→      if (px <= 0) continue;

  1202→        guardBlocked > 0

  1312→async function tryFlip(open: AutoTradeRow, row: ScanRow, price: number): Promise<boolean> {
  1313→  const { p, v } = eng();
  1314→  const c = p.cfg;
  1315→  const from = open.side;
  1316→  const to = row.bias === "SHORT" ? "SHORT" : "LONG";
  1317→
  1318→  // 1) đóng lệnh cũ
  1319→  const closed = await closeTradeDirect(open.id, price, "flip");
  1320→  if (!closed) return false; // lệnh đã đóng bởi watcher/luồng khác — bỏ qua
  1321→  console.log(
  1322→    `[engine] FLIP ${open.symbol} ${open.interval} ${from}→${to} @ ${price} — đóng lệnh cũ ROE ${(closed.pnlPct ?? 0) >= 0 ? "+" : ""}${(closed.pnlPct ?? 0).toFixed(2)}%`,
  1323→  );
  1324→  v.lastCycleStatus = `${open.symbol} ${open.interval} LẬT CHIỀU ${from}→${to}`;
  1325→  if (tgReady()) {
  1326→    const s2 = await readAutoState();
  1327→    void tgSendMessage(msgClose(closed, s2.stats.closed, s2.stats.winRate, s2.stats.netPnlPct), "HTML");

  1330→  // 2) SL/TP mới đo tại giá lật (tpMode "atr") — fail-open giữ mức scan row
  1331→  const coef = row.atrCoef > 0 ? row.atrCoef : c.atrCoef;
  1332→  let live: { slPct: number; tpPct: number; atrPct: number } | null = null;
  1333→  if (c.tpMode === "atr") {
  1334→    try {
  1335→      live = atrLevelsAtEntry(await fetchKlinesDirect(row.symbol, row.interval), coef);
  1336→    } catch {
  1337→      // fail-open — keep the scan-row levels
  1338→    }
  1339→  }
  1340→  const slPct = c.tpMode === "manual" ? c.slPct : c.tpMode === "atr" && live ? live.slPct : row.slPct;
  1341→  const tpPct = c.tpMode === "manual" ? c.tpPct : c.tpMode === "atr" && live ? live.tpPct : row.tpPct;
  1342→
  1343→  // 3) Risk Officer duyệt lệnh đảo chiều — Y HỆT luồng vào lệnh (kể cả track record)

  1346→    st.stats.closed > 0
  1347→      ? `${st.stats.wins}W/${st.stats.closed - st.stats.wins}L · win rate ${st.stats.winRate.toFixed(0)}% · net ${st.stats.netPnlPct >= 0 ? "+" : ""}${st.stats.netPnlPct.toFixed(1)}% ROE (${st.stats.closed} paper trades)`
  1348→      : undefined;
  1349→  const { verdict: lastVerdict, vetoed } = await llmGate(row, to, slPct, tpPct, track);

  1367→      tpPct,
  1368→      leverage: p.leverage,

  1371→      acc: row.acc,

  1399→// ─── DCA rescue — average in + re-anchor SL/TP while the model still confirms ─
  1400→//
  1401→// The old flow died exactly at the stop and instantly re-entered the same side
  1402→// at a worse price (the "SL xong vào lại giá cao hơn" bleed). Now, when price
  1403→// pushes against the position — at 70% of the way to the stop (near) or right
  1404→// at it (cross) — and the freshest scan still backs the original side with a
  1405→// decisive probability, the engine adds ONE more tranche of the original stake
  1406→// instead: entryPrice becomes the notional-weighted average of the tranches,
  1407→// and SL/TP re-anchor on that average automatically (they are % distances, the
  1408→// absolute levels always derive from entryPrice). Model stale, flipped, in
  1409→// the coin-flip band or below the DCA quality floor (AUC 0.6 · ACC 70,
  1410→// never looser than the panel's entry floors) → fail-safe: no DCA, the stop
  1411→// does its job. Hard caps:
  1412→// cfg.dcaMax adds per position; liquidation is never rescued.
  1413→
  1414→async function dcaRescue(open: AutoTradeRow, price: number, trigger: "near" | "cross"): Promise<boolean> {
  1415→  const { p, v } = eng();
  1416→  const c = p.cfg;
  1417→  if (c.dcaMax <= 0) return false;
  1418→  if (p.dca.id !== open.id) {
  1419→    // state from an older process / manual open — rebuild from the row itself
  1420→    p.dca = { id: open.id, count: 0, stake0: open.stakePct ?? c.riskPct, prices: [open.entryPrice] };
  1421→    persistEngine();
  1422→  }
  1423→  if (p.dca.count >= c.dcaMax) return false;
  1424→  // fail-safe: a FRESH scan must still confirm the original side decisively
  1425→  const scan = v.scan;
  1426→  if (!scan || Date.now() - scan.at > SCAN_TTL) return false;
  1427→  const row = scan.rows.find((r) => r.symbol === open.symbol && r.interval === open.interval);
  1428→  if (!row || row.bias !== open.side || Math.abs(row.prob - 0.5) < pval("NOISE_BAND", NOISE_BAND)) return false;
  1429→  // model-quality floor — NGƯỠNG CHUNG (cùng bộ ba với broadcast gate): model dưới
  1430→  // ngưỡng thì không bù thêm — để SL làm việc
  1431→  if (row.auc < c.minAuc || row.acc < c.minAcc || row.score < c.minScore) {
  1432→    console.info(
  1433→      `[engine] DCA ${open.symbol} ${open.interval} bỏ: model thoái hóa AUC ${row.auc.toFixed(2)} / ACC ${row.acc.toFixed(1)}% / điểm ${row.score.toFixed(1)} dưới ngưỡng chung ${c.minAuc.toFixed(2)} / ${c.minAcc.toFixed(0)}% / ${c.minScore} — để SL làm việc`,
  1434→    );
  1435→    return false;
  1436→  }
  1437→
  1438→  const stake0 = p.dca.stake0 || open.stakePct || c.riskPct;
  1439→  const tranchePrices = p.dca.prices.length > 0 ? p.dca.prices : [open.entryPrice];
  1440→  const count = p.dca.count;
  1441→  // equal-notional tranches (each = stake0 % of the entry-time capital) → the
  1442→  // average entry is the plain mean of the tranche prices
  1443→  const avgEntry =
  1444→    Math.round(([...tranchePrices, price].reduce((a, x) => a + x, 0) / (tranchePrices.length + 1)) * 1e8) / 1e8;
  1445→  const newStakePct = Math.round(((open.stakePct ?? c.riskPct) + stake0) * 100) / 100;
  1446→  try {
  1447→    const updated = await db.autoTrade.update({
  1448→      where: { id: open.id },
  1449→      data: { entryPrice: avgEntry, stakePct: newStakePct },
  1450→    });
  1451→    const trade = serialize(updated);
  1452→    p.dca = { id: open.id, count: count + 1, stake0, prices: [...tranchePrices, price] };
  1453→    p.warned = { id: open.id, tp: false, sl: false }; // new levels → fresh warns
  1454→    persistEngine();
  1455→    v.lastCycleStatus = `${open.symbol} ${open.interval} DCA #${count + 1}/${c.dcaMax} @ ${price}`;
  1456→    if (tgReady()) {
  1457→      void tgSendMessage(
  1458→        msgDca(trade, price, open.entryPrice, { prob: row.prob, auc: row.auc, acc: row.acc }, count + 1, c.dcaMax, trigger),
  1459→        "HTML",
  1460→      );
  1461→    }
  1462→    console.log(
  1463→      `[engine] DCA ${open.symbol} ${open.interval} #${count + 1}/${c.dcaMax} @ ${price} — avg ${avgEntry}, stake ${newStakePct}%`,
  1464→    );
  1465→    return true;
  1466→  } catch (e) {
  1467→    console.error(`[engine] DCA failed: ${e instanceof Error ? e.message : e}`);
  1468→    return false;
  1469→  }
  1470→}

  1472→// ─── watcher tick — intra-cycle SL/TP/liquidation + near-hit warnings ────────

  1474→async function watcherTick(): Promise<void> {
  1475→  const { p, v } = eng();
  1476→  const c = p.cfg;
  1477→  if (!c.on || !p.syncedAt) return;
  1478→  try {
  1479→    const st = await readAutoState();
  1480→    const open = st.open;
  1481→    if (!open) return;
  1482→    const tickers = await getTickers().catch(() => ({}) as Record<string, Ticker>);
  1483→    const price = tickers[open.symbol]?.price ?? 0;
  1484→    if (price <= 0) return;
  1485→    const kind = slTpHit(open.side, price, open.entryPrice, open.slPct, open.tpPct);
  1486→    const liq = liquidationHit(open.side, price, open.entryPrice, open.leverage);
  1487→    // last-chance DCA at the stop itself — model still confirms → average in,
  1488→    // re-anchor and keep riding instead of locking the loss (never on liquidation)
  1489→    if (kind === "sl" && !liq && (await dcaRescue(open, price, "cross"))) return;
  1490→    if (kind || liq) {
  1491→      const closed = await closeTradeDirect(open.id, price, liq ? "sl" : (kind ?? "sl"));
  1492→      if (closed) {
  1493→        if (tgReady()) {
  1494→          const s2 = await readAutoState();
  1495→          void tgSendMessage(msgClose(closed, s2.stats.closed, s2.stats.winRate, s2.stats.netPnlPct), "HTML");
  1496→        }
  1497→        // book the fresh entry a few seconds later — same as the client flow
  1498→        if (c.on) {
  1499→          setTimeout(() => {
  1500→            void cycleTick();
  1501→          }, 3_000);
  1502→        }
  1503→      }
  1504→      return;
  1505→    }
  1506→    // near-hit warning — once per side per position
  1507→    if (p.warned.id !== open.id) {
  1508→      p.warned = { id: open.id, tp: false, sl: false };
  1509→      persistEngine();
  1510→    }
  1511→    const near = nearHit(open.side, price, open.entryPrice, open.slPct, open.tpPct);
  1512→    if (near && !p.warned[near]) {
  1513→      p.warned[near] = true;
  1514→      persistEngine();
  1515→      if (tgReady()) void tgSendMessage(msgNear(open, price, near), "HTML");
  1516→    }
  1517→    // price inside the near-SL band and the model still confident → DCA now,
  1518→    // before the stop ever fires (the warn above already announced the risk)
  1519→    if (near === "sl") await dcaRescue(open, price, "near");
  1520→  } catch (e) {
  1521→    console.error(`[engine] watcher tick failed: ${e instanceof Error ? e.message : e}`);
  1522→  }
  1523→}
