     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import type { Kline, Paper, Signal } from "@/lib/types";
     5→import type { PendingEntry } from "@/lib/papers";
     6→import type { WyckoffEvent, WyckoffResult } from "@/lib/wyckoff";
     7→import { volumeProfile, fmtCompact } from "@/lib/market";
     8→import { fmtPrice, fmtPct, t, type Lang } from "@/lib/i18n";
     9→import { equityFrom } from "@/lib/equity";
    10→import { Button } from "@/components/ui/button";
    11→import { Minus, Plus } from "lucide-react";
    12→
    13→const UP = "#10b981";
    14→const DOWN = "#f43f5e";
    15→const FLIP = "#f59e0b";
    16→const GRID = "#27272a";
    17→const TEXT = "#a1a1aa";
    18→// Wyckoff palette — mirrors the Pine original
    19→const WY_ACC = "#22c55e";
    20→const WY_DIST = "#ef4444";
    21→const WY_PHC = "#a855f7";
    22→const WY_ACC_BAR = "#84cc16"; // barcolor lime
    23→const WY_DIST_BAR = "#a855f7"; // barcolor purple
    24→const WY_PIVOT = "#f97316";
    25→const WY_EMA_F = "#14b8a6"; // teal
    26→const WY_EMA_S = "#ef4444"; // red
    27→// SMA band tolerance around the gate SMA (visual aid: how far price is from the gate)
    28→const BAND_PCT = 0.005;
    29→
    30→const wyEventColor = (e: WyckoffEvent) => (e.phaseC ? WY_PHC : e.dir === 1 ? WY_ACC : WY_DIST);
    31→
    32→interface Props {
    33→  klines: Kline[];
    34→  sma: (number | null)[];
    35→  signals: Signal[];
    36→  papers: Paper[];
    37→  openPaper?: Paper;
    38→  pending?: PendingEntry;
    39→  pinnedIndex: number | null;
    40→  onPin: (index: number | null) => void;
    41→  lang: Lang;
    42→  /** Wyckoff overlay (computed in page when enabled) */
    43→  wyckoff?: WyckoffResult | null;
    44→  wyckoffOn?: boolean;
    45→  /** UT Bot trailing stop + trend state (from signalsFrom) — "UT stop" overlay */
    46→  stop?: (number | null)[];
    47→  trend?: number[];
    48→  /** stretch to fill the parent height (terminal layout) */
    49→  fill?: boolean;
    50→}
    51→
    52→const PAD_L = 8;
    53→const PAD_R = 64;
    54→const PAD_T = 12;
    55→const PAD_B = 26;
    56→
    57→const autoBars = (w: number) => (w < 520 ? 52 : w < 768 ? 76 : 110);
    58→
    59→export function PriceChart({
    60→  klines,
    61→  sma,
    62→  signals,
    63→  papers,
    64→  openPaper,
    65→  pending,
    66→  pinnedIndex,
    67→  onPin,
    68→  lang,
    69→  wyckoff,
    70→  wyckoffOn,
    71→  stop,
    72→  trend,
    73→  fill = false,
    74→}: Props) {
    75→  const wrapRef = useRef<HTMLDivElement>(null);
    76→  const plotRef = useRef<HTMLDivElement>(null);
    77→  const [width, setWidth] = useState(900);
    78→  const [measuredH, setMeasuredH] = useState(360);
    79→  const [viewEndRaw, setViewEndRaw] = useState<number | null>(null);
    80→  /** user zoom override — visible bar count (null = auto by width) */
    81→  const [bars, setBars] = useState<number | null>(null);
    82→  /** hover crosshair — candle index + cursor y inside the plot */
    83→  const [hover, setHover] = useState<{ i: number; py: number } | null>(null);
    84→  const dragRef = useRef<{ x: number; end: number } | null>(null);
    85→
    86→  useEffect(() => {
    87→    const el = wrapRef.current;
    88→    const plot = plotRef.current;
    89→    if (!el) return;
    90→    const ro = new ResizeObserver((entries) => {
    91→      for (const en of entries) {
    92→        if (en.target === el) {
    93→          setWidth(Math.max(320, Math.floor(en.contentRect.width)));
    94→        } else if (en.target === plot) {
    95→          // floor of 160px keeps the plot usable — going lower would overflow the card
    96→          setMeasuredH(Math.max(160, Math.floor(en.contentRect.height)));
    97→        }
    98→      }
    99→    });
   100→    ro.observe(el);
   101→    if (plot) ro.observe(plot);
   102→    return () => ro.disconnect();
   103→  }, []);
   104→
   105→  // wheel zoom — native listener (passive: false) so preventDefault works
   106→  useEffect(() => {
   107→    const el = wrapRef.current;
   108→    if (!el) return;
   109→    const onWheel = (e: WheelEvent) => {
   110→      if (!e.deltaY) return;
   111→      e.preventDefault();
   112→      setBars((prev) => {
   113→        const base = prev ?? autoBars(width);
   114→        const next = e.deltaY > 0 ? base * 1.18 : base / 1.18;
   115→        return Math.max(24, Math.min(340, Math.round(next)));
   116→      });
   117→    };
   118→    el.addEventListener("wheel", onWheel, { passive: false });
   119→    return () => el.removeEventListener("wheel", onWheel);
   120→  }, [width]);
   121→
   122→  const n = klines.length;
   123→  const winSize = Math.max(20, Math.min(bars ?? autoBars(width), n || 1));
   124→  const viewEnd = Math.min(viewEndRaw ?? n - 1, n - 1);
   125→  const viewStart = Math.max(0, viewEnd - winSize + 1);
   126→
   127→  const zoom = useCallback(
   128→    (dir: 1 | -1) =>
   129→      setBars((prev) => {
   130→        const base = prev ?? autoBars(width);
   131→        const next = dir === 1 ? base / 1.35 : base * 1.35;
   132→        return Math.max(24, Math.min(340, Math.round(next)));
   133→      }),
   134→    [width],
   135→  );
   136→
   137→  const jumpLatest = useCallback(() => setViewEndRaw(null), []);
   138→
   139→  const height = fill ? measuredH : width < 520 ? 280 : 360;
   140→  const plotW = width - PAD_L - PAD_R;
   141→  // ── Equity pane (Đường cong vốn) — a small strip carved out of the plot bottom
   142→  const eq = useMemo(() => equityFrom(papers), [papers]);
   143→  const eqOn = eq.points.length >= 2;
   144→  const EQ_PANE = 58;
   145→  const eqPaneH = eqOn ? EQ_PANE : 0;
   146→  const plotH = height - PAD_T - PAD_B - eqPaneH;
   147→  const cw = plotW / Math.max(1, viewEnd - viewStart + 1);
   148→
   149→  const { minP, maxP } = useMemo(() => {
   150→    let lo = Infinity;
   151→    let hi = -Infinity;
   152→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   153→      const k = klines[i];
   154→      if (k.l < lo) lo = k.l;
   155→      if (k.h > hi) hi = k.h;
   156→      const s = sma[i];
   157→      if (s != null && s < lo) lo = s;
   158→      if (s != null && s > hi) hi = s;
   159→      const st = stop?.[i];
   160→      if (st != null && st < lo) lo = st;
   161→      if (st != null && st > hi) hi = st;
   162→    }
   163→    // NOTE: open SL/TP are deliberately NOT part of the scale — a wide 6% SL
   164→    // would squash the candles. Levels are clipped to the plot instead.
   165→    if (openPaper && openPaper.entry >= lo * 0.98 && openPaper.entry <= hi * 1.02) {
   166→      if (openPaper.entry < lo) lo = openPaper.entry;
   167→      if (openPaper.entry > hi) hi = openPaper.entry;
   168→    }
   169→    if (!isFinite(lo) || !isFinite(hi)) return { minP: 0, maxP: 1 };
   170→    if (hi === lo) {
   171→      hi += hi * 0.001 || 1;
   172→      lo -= lo * 0.001 || 1;
   173→    }
   174→    const pad = (hi - lo) * 0.06;
   175→    return { minP: lo - pad, maxP: hi + pad };
   176→  }, [klines, sma, stop, viewStart, viewEnd, n, openPaper]);
   177→
   178→  const x = (i: number) => PAD_L + (i - viewStart) * cw + cw / 2;
   179→  const y = (p: number) => PAD_T + ((maxP - p) / (maxP - minP)) * plotH;
   180→  /** inverse of y() — price under the crosshair */
   181→  const priceAtY = (py: number) => minP + ((PAD_T + plotH - py) / plotH) * (maxP - minP);
   182→
   183→  const onPointerDown = (e: React.PointerEvent<SVGSVGElement>) => {
   184→    dragRef.current = { x: e.clientX, end: viewEnd };
   185→    (e.target as Element).setPointerCapture?.(e.pointerId);
   186→  };
   187→  const onPointerMove = (e: React.PointerEvent<SVGSVGElement>) => {
   188→    // hover crosshair (TradingView-style) — always tracked, pan continues on top
   189→    const rect = e.currentTarget.getBoundingClientRect();
   190→    const py = Math.max(PAD_T, Math.min(PAD_T + plotH, e.clientY - rect.top));
   191→    const i = Math.max(viewStart, Math.min(viewEnd, viewStart + Math.floor((e.clientX - rect.left - PAD_L) / cw)));
   192→    setHover((prev) => (prev && prev.i === i && Math.abs(prev.py - py) < 0.5 ? prev : { i, py }));
   193→    const d = dragRef.current;
   194→    if (!d) return;
   195→    const shift = Math.round((d.x - e.clientX) / cw);
   196→    const end = Math.min(n - 1, Math.max(winSize - 1, d.end + shift));
   197→    setViewEndRaw(end >= n - 1 ? null : end);
   198→  };
   199→  const onPointerUp = () => {
   200→    dragRef.current = null;
   201→  };
   202→  const onPointerLeave = () => {
   203→    dragRef.current = null;
   204→    setHover(null);
   205→  };
   206→
   207→  // ── Equity pane geometry (Đường cong vốn) ──────────────────────────────
   208→  const eqTop = PAD_T + plotH + 10;
   209→  const eqH = eqPaneH - 16;
   210→  const eqLo = Math.min(eq.trough, 0);
   211→  const eqHi = Math.max(eq.peak, eqLo + 0.5);
   212→  const eqY = (v: number) => eqTop + ((eqHi - v) / (eqHi - eqLo)) * eqH;
   213→  const eqX = (bar: number) => Math.max(PAD_L, Math.min(PAD_L + plotW, x(bar)));
   214→  // trades closed up to the viewed bar — the curve stays honest while panning back
   215→  const eqSeen = eqOn ? eq.points.filter((p) => p.bar <= viewEnd) : [];
   216→  const eqLine = (() => {
   217→    if (!eqOn || eqH <= 4) return null;
   218→    if (eqSeen.length === 0) {
   219→      const d = `M${PAD_L},${eqY(0).toFixed(1)} L${(PAD_L + plotW).toFixed(1)},${eqY(0).toFixed(1)}`;
   220→      return { d, area: `${d} Z`, endV: 0, endY: eqY(0) };
   221→    }
   222→    const pts = eqSeen;
   223→    let d = pts[0].bar > viewStart ? `M${PAD_L},${eqY(0).toFixed(1)} L${eqX(pts[0].bar).toFixed(1)},${eqY(pts[0].cum).toFixed(1)}` : `M${eqX(pts[0].bar).toFixed(1)},${eqY(pts[0].cum).toFixed(1)}`;
   224→    for (let k = 1; k < pts.length; k++) d += ` L${eqX(pts[k].bar).toFixed(1)},${eqY(pts[k].cum).toFixed(1)}`;
   225→    const endY = eqY(pts[pts.length - 1].cum);
   226→    d += ` L${(PAD_L + plotW).toFixed(1)},${endY.toFixed(1)}`; // flat extension to "now"
   227→    const area = `${d} L${(PAD_L + plotW).toFixed(1)},${eqY(0).toFixed(1)} L${PAD_L},${eqY(0).toFixed(1)} Z`;
   228→    return { d, area, endV: pts[pts.length - 1].cum, endY };
   229→  })();
   230→
   231→  const sigInView = signals.filter((s) => s.i >= viewStart && s.i <= viewEnd);
   232→  const eqColor = eqLine ? (eqLine.endV >= 0 ? UP : DOWN) : UP;
   233→  const papersInView = papers.filter(
   234→    (p) => p.exitIndex !== undefined ? p.exitIndex >= viewStart || p.entryIndex <= viewEnd : p.entryIndex <= viewEnd,
   235→  );
   236→
   237→  // ── Volume profile over the visible window (buy/sell split via taker data) ─
   238→  const vp = useMemo(() => volumeProfile(klines, viewStart, viewEnd, 26), [klines, viewStart, viewEnd]);
   239→  const vpW = plotW * 0.16;
   240→
   241→  // ── Whale delta histogram over the visible window ────────────────────────
   242→  const whale = useMemo(() => {
   243→    let maxD = 0;
   244→    const rows: { i: number; d: number }[] = [];
   245→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   246→      const k = klines[i];
   247→      if (k.tb == null || k.v <= 0) continue;
   248→      const d = 2 * k.tb - k.v;
   249→      rows.push({ i, d });
   250→      if (Math.abs(d) > maxD) maxD = Math.abs(d);
   251→    }
   252→    return { rows, maxD };
   253→  }, [klines, viewStart, viewEnd, n]);
   254→  const whaleH = plotH * 0.2;
   255→
   256→  // ── Wyckoff layers (only when enabled) ─────────────────────────────────
   257→  const wyOn = !!wyckoffOn && !!wyckoff;
   258→  const wyBoxes = wyOn
   259→    ? wyckoff!.boxes.filter((b) => !b.deleted && b.right >= viewStart && b.left <= viewEnd)
   260→    : [];
   261→  const wyEvents = wyOn
   262→    ? wyckoff!.events.filter((e) => !e.deleted && e.bar >= viewStart && e.bar <= viewEnd)
   263→    : [];
   264→  const wyPivots = wyOn
   265→    ? wyckoff!.pivots.filter((p) => !p.hidden && p.bar >= viewStart && p.bar <= viewEnd)
   266→    : [];
   267→  const tintOf = (i: number): 0 | 1 | -1 => (wyOn ? (wyckoff!.tints[i] as 0 | 1 | -1) : 0);
   268→
   269→  /** polyline path skipping nulls (EMA arrays) */
   270→  const linePath = (arr: (number | null)[]): string => {
   271→    let d = "";
   272→    let pen = false;
   273→    for (let i = viewStart; i <= viewEnd; i++) {
   274→      const v = arr[i];
   275→      if (v == null) {
   276→        pen = false;
   277→        continue;
   278→      }
   279→      d += `${pen ? "L" : "M"}${x(i)},${y(v)} `;
   280→      pen = true;
   281→    }
   282→    return d.trim();
   283→  };
   284→
   285→  /** SMA ± band filled area — one closed subpath per contiguous SMA run */
   286→  const bandPath = (): string => {
   287→    let d = "";
   288→    let top: string[] = [];
   289→    let bot: string[] = [];
   290→    const flush = () => {
   291→      if (top.length >= 2) {
   292→        d += `M${top.join(" L")} L${bot.reverse().join(" L")} Z `;
   293→      }
   294→      top = [];
   295→      bot = [];
   296→    };
   297→    for (let i = viewStart; i <= viewEnd; i++) {
   298→      const v = sma[i];
   299→      if (v == null) {
   300→        flush();
   301→        continue;
   302→      }
   303→      top.push(`${x(i)},${y(v * (1 + BAND_PCT))}`);
   304→      bot.push(`${x(i)},${y(v * (1 - BAND_PCT))}`);
   305→    }
   306→    flush();
   307→    return d.trim();
   308→  };
   309→
   310→  /** UT trailing-stop segments, colored per trend run */
   311→  const stopSegs = (): { d: string; up: boolean }[] => {
   312→    const segs: { d: string; up: boolean }[] = [];
   313→    let cur: { d: string; up: boolean } | null = null;
   314→    for (let i = viewStart; i <= viewEnd; i++) {
   315→      const v = stop?.[i];
   316→      const tr = trend?.[i] ?? 0;
   317→      if (v == null || tr === 0) {
   318→        cur = null;
   319→        continue;
   320→      }
   321→      const up = tr === 1;
   322→      if (cur && cur.up === up) cur.d += ` L${x(i)},${y(v)}`;
   323→      else {
   324→        cur = { d: `M${x(i)},${y(v)}`, up };
   325→        segs.push(cur);
   326→      }
   327→    }
   328→    return segs;
   329→  };
   330→
   331→  const timeLabel = (t: number) => {
   332→    const d = new Date(t);
   333→    const hh = String(d.getHours()).padStart(2, "0");
   334→    const mm = String(d.getMinutes()).padStart(2, "0");
   335→    return `${d.getDate()}/${d.getMonth() + 1} ${hh}:${mm}`;
   336→  };
   337→
   338→  if (n === 0) {
   339→    return (
   340→      <div
   341→        ref={wrapRef}
   342→        className={`flex items-center justify-center rounded-lg border border-zinc-800 bg-zinc-950 text-sm text-zinc-500 ${fill ? "h-full" : "h-64"}`}
   343→      >
   344→        {t(lang, "chart.empty")}
   345→      </div>
   346→    );
   347→  }
   348→
   349→  const gridPrices = Array.from({ length: 5 }, (_, k) => minP + ((maxP - minP) * k) / 4);
   350→  const openEntryLabel = openPaper?.flippedFrom ? t(lang, "chart.line.flipEntry") : t(lang, "chart.line.entry");
   351→
   352→  return (
   353→    <div ref={wrapRef} className={`relative flex select-none flex-col ${fill ? "h-full" : ""}`}>
   354→      <div ref={plotRef} className={`relative min-h-0 overflow-hidden ${fill ? "flex-1" : ""}`}>
   355→        <svg
   356→          width={width}
   357→          height={height}
   358→          className="touch-pan-y rounded-lg border border-zinc-800 bg-zinc-950"
   359→          onPointerDown={onPointerDown}
   360→          onPointerMove={onPointerMove}
   361→          onPointerUp={onPointerUp}
   362→          onPointerLeave={onPointerLeave}
   363→          role="img"
   364→          aria-label={t(lang, "chart.title")}
   365→        >
   366→          <defs>
   367→            <clipPath id="wy-plot-clip">
   368→              <rect x={PAD_L} y={PAD_T} width={plotW} height={plotH} />
   369→            </clipPath>
   370→          </defs>
   371→          {/* grid + price axis */}
   372→          {gridPrices.map((p, k) => (
   373→            <g key={k}>
   374→              <line x1={PAD_L} x2={PAD_L + plotW} y1={y(p)} y2={y(p)} stroke={GRID} strokeWidth={1} />
   375→              <text x={PAD_L + plotW + 6} y={y(p) + 4} fill={TEXT} fontSize={10} fontFamily="ui-monospace, monospace">
   376→                {fmtPrice(p)}
   377→              </text>
   378→            </g>
   379→          ))}
   380→          {/* time axis — edge labels anchor inward, greedy de-dupe avoids collisions */}
   381→          {(() => {
   382→            const last = width < 520 ? 3 : 5;
   383→            const est = (s: string) => s.length * 6.2; // ~px per char @ 10px mono
   384→            let prevRight = -Infinity;
   385→            const nodes: React.ReactNode[] = [];
   386→            for (let k = 0; k <= last; k++) {
   387→              const idx = Math.round(viewStart + ((viewEnd - viewStart) * k) / last);
   388→              const kk = klines[Math.min(idx, n - 1)];
   389→              const label = timeLabel(kk.t);
   390→              const w = est(label);
   391→              const anchor = k === 0 ? "start" : k === last ? "end" : "middle";
   392→              const cx = k === 0 ? PAD_L : k === last ? PAD_L + plotW : Math.min(x(idx), PAD_L + plotW - 10);
   393→              const left = anchor === "start" ? cx : anchor === "end" ? cx - w : cx - w / 2;
   394→              if (left < prevRight + 10) continue; // would collide with the previous label
   395→              prevRight = left + w;
   396→              nodes.push(
   397→                <text
   398→                  key={k}
   399→                  x={cx}
   400→                  y={height - 8}
   401→                  fill={TEXT}
   402→                  fontSize={10}
   403→                  textAnchor={anchor}
   404→                  fontFamily="ui-monospace, monospace"
   405→                >
   406→                  {label}
   407→                </text>,
   408→              );
   409→            }
   410→            return nodes;
   411→          })()}
   412→
   413→          {/* ── Whale taker-delta histogram (bottom of the plot) ───────── */}
   414→          {whale.maxD > 0 && (
   415→            <g clipPath="url(#wy-plot-clip)">
   416→              <line
   417→                x1={PAD_L}
   418→                x2={PAD_L + plotW}
   419→                y1={PAD_T + plotH}
   420→                y2={PAD_T + plotH}
   421→                stroke={GRID}
   422→                strokeWidth={1}
   423→              />
   424→              {whale.rows.map(({ i, d }) => {
   425→                const h = Math.max(1, (Math.abs(d) / whale.maxD) * whaleH);
   426→                const bw = Math.max(1.5, cw * 0.55);
   427→                return (
   428→                  <rect
   429→                    key={`wh-${i}`}
   430→                    x={x(i) - bw / 2}
   431→                    y={PAD_T + plotH - h}
   432→                    width={bw}
   433→                    height={h}
   434→                    fill={d >= 0 ? UP : DOWN}
   435→                    opacity={0.34}
   436→                  />
   437→                );
   438→              })}
   439→            </g>
   440→          )}
   441→
   442→          {/* ── Equity pane (Đường cong vốn) — cumulative ROE of closed trades ── */}
   443→          {eqOn && (
   444→            <g pointerEvents="none">
   445→              <defs>
   446→                <linearGradient id="eq-pane-fill" x1="0" y1="0" x2="0" y2="1">
   447→                  <stop offset="0%" stopColor={eqColor} stopOpacity={0.25} />
   448→                  <stop offset="100%" stopColor={eqColor} stopOpacity={0.02} />
   449→                </linearGradient>
   450→              </defs>
   451→              <line x1={PAD_L} x2={PAD_L + plotW} y1={PAD_T + plotH + 4} y2={PAD_T + plotH + 4} stroke={GRID} strokeWidth={1} />
   452→              <text x={PAD_L + 2} y={eqTop + 1} fill="#71717a" fontSize={9} fontFamily="ui-monospace, monospace">
   453→                {t(lang, "tabs.equity")} · {eq.count}
   454→              </text>
   455→              <line x1={PAD_L} x2={PAD_L + plotW} y1={eqY(0)} y2={eqY(0)} stroke="#3f3f46" strokeDasharray="4 4" strokeWidth={1} />
   456→              {eqLine && <path d={eqLine.area} fill="url(#eq-pane-fill)" />}
   457→              {eqLine && <path d={eqLine.d} fill="none" stroke={eqColor} strokeWidth={1.6} strokeLinejoin="round" />}
   458→              {eqLine && (
   459→                <text
   460→                  x={PAD_L + plotW}
   461→                  y={Math.max(eqTop + 8, eqLine.endY - 4)}
   462→                  textAnchor="end"
   463→                  fill={eqColor}
   464→                  fontSize={10}
   465→                  fontWeight={700}
   466→                  fontFamily="ui-monospace, monospace"
   467→                >
   468→                  {fmtPct(eqLine.endV)}
