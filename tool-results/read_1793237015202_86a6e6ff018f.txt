    47→const QUOTE_RE = /(USDT|USDC|FDUSD|TUSD|BUSD|BTC|ETH)$/;

    49→/** "pepe" → "PEPEUSDT", "PEPEUSDT" → "PEPEUSDT" */
    50→const normalizePair = (raw: string): string => {
    51→  const s = raw.trim().toUpperCase().replace(/[^A-Z0-9]/g, "");
    52→  if (!s) return "";
    53→  return QUOTE_RE.test(s) ? s : `${s}USDT`;

    56→/** Live VN + UTC clocks — 1 Hz external "store", SSR-safe (server snapshot 0). */
    57→function Clocks({ lang }: { lang: Lang }) {
    58→  // seconds-precision snapshot keeps getSnapshot stable within a second
    59→  const sec = useSyncExternalStore(
    60→    (onStoreChange) => {
    61→      const id = window.setInterval(onStoreChange, 1000);
    62→      return () => window.clearInterval(id);
    63→    },
    64→    () => Math.floor(Date.now() / 1000),
    65→    () => 0,
    66→  );
    67→  const fmt = (tz: string) =>
    68→    sec === 0
    69→      ? "--:--:--"
    70→      : new Date(sec * 1000).toLocaleTimeString("en-GB", {
    71→          hour: "2-digit",
    72→          minute: "2-digit",
    73→          second: "2-digit",
    74→          hour12: false,
    75→          timeZone: tz,
    76→        });
    77→  return (
    78→    <div className="hidden flex-col items-end leading-tight lg:flex" aria-hidden>
    79→      <span className="font-mono text-[11px] text-zinc-300">
    80→        <span className="text-zinc-600">{t(lang, "header.clockVN")}</span> {fmt("Asia/Ho_Chi_Minh")}
    81→      </span>
    82→      <span className="font-mono text-[11px] text-zinc-500">
    83→        <span className="text-zinc-600">{t(lang, "header.clockUTC")}</span> {fmt("UTC")}
    84→      </span>
    85→    </div>
    86→  );
    87→}
