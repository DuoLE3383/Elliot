    45→const QUOTE_RE = /(USDT|USDC|FDUSD|TUSD|BUSD|BTC|ETH)$/;

    47→/** "pepe" → "PEPEUSDT", "PEPEUSDT" → "PEPEUSDT" */
    48→const normalizePair = (raw: string): string => {
    49→  const s = raw.trim().toUpperCase().replace(/[^A-Z0-9]/g, "");
    50→  if (!s) return "";
    51→  return QUOTE_RE.test(s) ? s : `${s}USDT`;
