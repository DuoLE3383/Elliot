     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import type { Kline, Paper, Signal } from "@/lib/types";
     5→import type { PendingEntry } from "@/lib/papers";
     6→import type { WyckoffEvent, WyckoffResult } from "@/lib/wyckoff";
     7→import { volumeProfile } from "@/lib/market";
     8→import { fmtPrice, t, type Lang } from "@/lib/i18n";
     9→import { Button } from "@/components/ui/button";
    10→import { Minus, Plus } from "lucide-react";
    11→
    12→const UP = "#10b981";
    13→const DOWN = "#f43f5e";
    14→const FLIP = "#f59e0b";
    15→const GRID = "#27272a";
    16→const TEXT = "#a1a1aa";
    17→// Wyckoff palette — mirrors the Pine original
    18→const WY_ACC = "#22c55e";
    19→const WY_DIST = "#ef4444";
    20→const WY_PHC = "#a855f7";
    21→const WY_ACC_BAR = "#84cc16"; // barcolor lime
    22→const WY_DIST_BAR = "#a855f7"; // barcolor purple
    23→const WY_PIVOT = "#f97316";
    24→const WY_EMA_F = "#14b8a6"; // teal
    25→const WY_EMA_S = "#ef4444"; // red
    26→// SMA band tolerance around the gate SMA (visual aid: how far price is from the gate)
    27→const BAND_PCT = 0.005;
    28→
    29→const wyEventColor = (e: WyckoffEvent) => (e.phaseC ? WY_PHC : e.dir === 1 ? WY_ACC : WY_DIST);
    30→
    31→interface Props {
    32→  klines: Kline[];
    33→  sma: (number | null)[];
    34→  signals: Signal[];
    35→  papers: Paper[];
    36→  openPaper?: Paper;
    37→  pending?: PendingEntry;
    38→  pinnedIndex: number | null;
    39→  onPin: (index: number | null) => void;
    40→  lang: Lang;
    41→  /** Wyckoff overlay (computed in page when enabled) */
    42→  wyckoff?: WyckoffResult | null;
    43→  wyckoffOn?: boolean;
    44→  /** UT Bot trailing stop + trend state (from signalsFrom) — "UT stop" overlay */
    45→  stop?: (number | null)[];
    46→  trend?: number[];
    47→  /** stretch to fill the parent height (terminal layout) */
    48→  fill?: boolean;
    49→}
    50→
    51→const PAD_L = 8;
    52→const PAD_R = 64;
    53→const PAD_T = 12;
    54→const PAD_B = 26;
    55→
    56→const autoBars = (w: number) => (w < 520 ? 52 : w < 768 ? 76 : 110);
    57→
    58→export function PriceChart({
    59→  klines,
    60→  sma,
    61→  signals,
    62→  papers,
    63→  openPaper,
    64→  pending,
    65→  pinnedIndex,
    66→  onPin,
    67→  lang,
    68→  wyckoff,
    69→  wyckoffOn,
    70→  stop,
    71→  trend,
    72→  fill = false,
    73→}: Props) {
    74→  const wrapRef = useRef<HTMLDivElement>(null);
    75→  const plotRef = useRef<HTMLDivElement>(null);
    76→  const [width, setWidth] = useState(900);
    77→  const [measuredH, setMeasuredH] = useState(360);
    78→  const [viewEndRaw, setViewEndRaw] = useState<number | null>(null);
    79→  /** user zoom override — visible bar count (null = auto by width) */
    80→  const [bars, setBars] = useState<number | null>(null);
    81→  const dragRef = useRef<{ x: number; end: number } | null>(null);
    82→
    83→  useEffect(() => {
    84→    const el = wrapRef.current;
    85→    const plot = plotRef.current;
    86→    if (!el) return;
    87→    const ro = new ResizeObserver((entries) => {
    88→      for (const en of entries) {
    89→        if (en.target === el) {
    90→          setWidth(Math.max(320, Math.floor(en.contentRect.width)));
    91→        } else if (en.target === plot) {
    92→          // floor of 160px keeps the plot usable — going lower would overflow the card
    93→          setMeasuredH(Math.max(160, Math.floor(en.contentRect.height)));
    94→        }
    95→      }
    96→    });
    97→    ro.observe(el);
    98→    if (plot) ro.observe(plot);
    99→    return () => ro.disconnect();
   100→  }, []);
   101→
   102→  // wheel zoom — native listener (passive: false) so preventDefault works
   103→  useEffect(() => {
   104→    const el = wrapRef.current;
   105→    if (!el) return;
   106→    const onWheel = (e: WheelEvent) => {
   107→      if (!e.deltaY) return;
   108→      e.preventDefault();
   109→      setBars((prev) => {
   110→        const base = prev ?? autoBars(width);
   111→        const next = e.deltaY > 0 ? base * 1.18 : base / 1.18;
   112→        return Math.max(24, Math.min(340, Math.round(next)));
   113→      });
   114→    };
   115→    el.addEventListener("wheel", onWheel, { passive: false });
   116→    return () => el.removeEventListener("wheel", onWheel);
   117→  }, [width]);
   118→
   119→  const n = klines.length;
   120→  const winSize = Math.max(20, Math.min(bars ?? autoBars(width), n || 1));
   121→  const viewEnd = Math.min(viewEndRaw ?? n - 1, n - 1);
   122→  const viewStart = Math.max(0, viewEnd - winSize + 1);
   123→
   124→  const zoom = useCallback(
   125→    (dir: 1 | -1) =>
   126→      setBars((prev) => {
   127→        const base = prev ?? autoBars(width);
   128→        const next = dir === 1 ? base / 1.35 : base * 1.35;
   129→        return Math.max(24, Math.min(340, Math.round(next)));
   130→      }),
   131→    [width],
   132→  );
   133→
   134→  const jumpLatest = useCallback(() => setViewEndRaw(null), []);
   135→
   136→  const height = fill ? measuredH : width < 520 ? 280 : 360;
   137→  const plotW = width - PAD_L - PAD_R;
   138→  const plotH = height - PAD_T - PAD_B;
   139→  const cw = plotW / Math.max(1, viewEnd - viewStart + 1);
   140→
   141→  const { minP, maxP } = useMemo(() => {
   142→    let lo = Infinity;
   143→    let hi = -Infinity;
   144→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   145→      const k = klines[i];
   146→      if (k.l < lo) lo = k.l;
   147→      if (k.h > hi) hi = k.h;
   148→      const s = sma[i];
   149→      if (s != null && s < lo) lo = s;
   150→      if (s != null && s > hi) hi = s;
   151→      const st = stop?.[i];
   152→      if (st != null && st < lo) lo = st;
   153→      if (st != null && st > hi) hi = st;
   154→    }
   155→    // NOTE: open SL/TP are deliberately NOT part of the scale — a wide 6% SL
   156→    // would squash the candles. Levels are clipped to the plot instead.
   157→    if (openPaper && openPaper.entry >= lo * 0.98 && openPaper.entry <= hi * 1.02) {
   158→      if (openPaper.entry < lo) lo = openPaper.entry;
   159→      if (openPaper.entry > hi) hi = openPaper.entry;
   160→    }
   161→    if (!isFinite(lo) || !isFinite(hi)) return { minP: 0, maxP: 1 };
   162→    if (hi === lo) {
   163→      hi += hi * 0.001 || 1;
   164→      lo -= lo * 0.001 || 1;
   165→    }
   166→    const pad = (hi - lo) * 0.06;
   167→    return { minP: lo - pad, maxP: hi + pad };
   168→  }, [klines, sma, stop, viewStart, viewEnd, n, openPaper]);
   169→
   170→  const x = (i: number) => PAD_L + (i - viewStart) * cw + cw / 2;
   171→  const y = (p: number) => PAD_T + ((maxP - p) / (maxP - minP)) * plotH;
   172→
   173→  const onPointerDown = (e: React.PointerEvent<SVGSVGElement>) => {
   174→    dragRef.current = { x: e.clientX, end: viewEnd };
   175→    (e.target as Element).setPointerCapture?.(e.pointerId);
   176→  };
   177→  const onPointerMove = (e: React.PointerEvent<SVGGElement>) => {
   178→    const d = dragRef.current;
   179→    if (!d) return;
   180→    const shift = Math.round((d.x - e.clientX) / cw);
   181→    const end = Math.min(n - 1, Math.max(winSize - 1, d.end + shift));
   182→    setViewEndRaw(end >= n - 1 ? null : end);
   183→  };
   184→  const onPointerUp = () => {
   185→    dragRef.current = null;
   186→  };
   187→
   188→  const sigInView = signals.filter((s) => s.i >= viewStart && s.i <= viewEnd);
   189→  const papersInView = papers.filter(
   190→    (p) => p.exitIndex !== undefined ? p.exitIndex >= viewStart || p.entryIndex <= viewEnd : p.entryIndex <= viewEnd,
   191→  );
   192→
   193→  // ── Volume profile over the visible window (buy/sell split via taker data) ─
   194→  const vp = useMemo(() => volumeProfile(klines, viewStart, viewEnd, 26), [klines, viewStart, viewEnd]);
   195→  const vpW = plotW * 0.16;
   196→
   197→  // ── Whale delta histogram over the visible window ────────────────────────
   198→  const whale = useMemo(() => {
   199→    let maxD = 0;
   200→    const rows: { i: number; d: number }[] = [];
   201→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   202→      const k = klines[i];
   203→      if (k.tb == null || k.v <= 0) continue;
   204→      const d = 2 * k.tb - k.v;
   205→      rows.push({ i, d });
   206→      if (Math.abs(d) > maxD) maxD = Math.abs(d);
   207→    }
   208→    return { rows, maxD };
   209→  }, [klines, viewStart, viewEnd, n]);
   210→  const whaleH = plotH * 0.2;
   211→
   212→  // ── Wyckoff layers (only when enabled) ─────────────────────────────────
   213→  const wyOn = !!wyckoffOn && !!wyckoff;
   214→  const wyBoxes = wyOn
   215→    ? wyckoff!.boxes.filter((b) => !b.deleted && b.right >= viewStart && b.left <= viewEnd)
   216→    : [];
   217→  const wyEvents = wyOn
   218→    ? wyckoff!.events.filter((e) => !e.deleted && e.bar >= viewStart && e.bar <= viewEnd)
   219→    : [];
   220→  const wyPivots = wyOn
   221→    ? wyckoff!.pivots.filter((p) => !p.hidden && p.bar >= viewStart && p.bar <= viewEnd)
   222→    : [];
   223→  const tintOf = (i: number): 0 | 1 | -1 => (wyOn ? (wyckoff!.tints[i] as 0 | 1 | -1) : 0);
   224→
   225→  /** polyline path skipping nulls (EMA arrays) */
   226→  const linePath = (arr: (number | null)[]): string => {
   227→    let d = "";
   228→    let pen = false;
   229→    for (let i = viewStart; i <= viewEnd; i++) {
   230→      const v = arr[i];
   231→      if (v == null) {
   232→        pen = false;
   233→        continue;
   234→      }
   235→      d += `${pen ? "L" : "M"}${x(i)},${y(v)} `;
   236→      pen = true;
   237→    }
   238→    return d.trim();
   239→  };
   240→
   241→  /** SMA ± band filled area — one closed subpath per contiguous SMA run */
   242→  const bandPath = (): string => {
   243→    let d = "";
   244→    let top: string[] = [];
   245→    let bot: string[] = [];
   246→    const flush = () => {
   247→      if (top.length >= 2) {
   248→        d += `M${top.join(" L")} L${bot.reverse().join(" L")} Z `;
   249→      }
   250→      top = [];
   251→      bot = [];
   252→    };
   253→    for (let i = viewStart; i <= viewEnd; i++) {
   254→      const v = sma[i];
   255→      if (v == null) {
   256→        flush();
   257→        continue;
   258→      }
   259→      top.push(`${x(i)},${y(v * (1 + BAND_PCT))}`);
   260→      bot.push(`${x(i)},${y(v * (1 - BAND_PCT))}`);
   261→    }
   262→    flush();
   263→    return d.trim();
   264→  };
   265→
   266→  /** UT trailing-stop segments, colored per trend run */
   267→  const stopSegs = (): { d: string; up: boolean }[] => {
   268→    const segs: { d: string; up: boolean }[] = [];
   269→    let cur: { d: string; up: boolean } | null = null;
   270→    for (let i = viewStart; i <= viewEnd; i++) {
   271→      const v = stop?.[i];
   272→      const tr = trend?.[i] ?? 0;
   273→      if (v == null || tr === 0) {
   274→        cur = null;
   275→        continue;
   276→      }
   277→      const up = tr === 1;
   278→      if (cur && cur.up === up) cur.d += ` L${x(i)},${y(v)}`;
   279→      else {
   280→        cur = { d: `M${x(i)},${y(v)}`, up };
   281→        segs.push(cur);
   282→      }
   283→    }
   284→    return segs;
   285→  };
   286→
   287→  const timeLabel = (t: number) => {
   288→    const d = new Date(t);
   289→    const hh = String(d.getHours()).padStart(2, "0");
   290→    const mm = String(d.getMinutes()).padStart(2, "0");
   291→    return `${d.getDate()}/${d.getMonth() + 1} ${hh}:${mm}`;
   292→  };
   293→
   294→  if (n === 0) {
   295→    return (
   296→      <div
   297→        ref={wrapRef}
   298→        className={`flex items-center justify-center rounded-lg border border-zinc-800 bg-zinc-950 text-sm text-zinc-500 ${fill ? "h-full" : "h-64"}`}
   299→      >
   300→        {t(lang, "chart.empty")}
   301→      </div>
   302→    );
   303→  }
   304→
   305→  const gridPrices = Array.from({ length: 5 }, (_, k) => minP + ((maxP - minP) * k) / 4);
   306→  const openEntryLabel = openPaper?.flippedFrom ? t(lang, "chart.line.flipEntry") : t(lang, "chart.line.entry");
   307→
   308→  return (
   309→    <div ref={wrapRef} className={`relative flex select-none flex-col ${fill ? "h-full" : ""}`}>
   310→      <div ref={plotRef} className={`relative min-h-0 overflow-hidden ${fill ? "flex-1" : ""}`}>
   311→        <svg
   312→          width={width}
   313→          height={height}
   314→          className="touch-pan-y rounded-lg border border-zinc-800 bg-zinc-950"
   315→          onPointerDown={onPointerDown}
   316→          onPointerMove={onPointerMove}
   317→          onPointerUp={onPointerUp}
   318→          onPointerLeave={onPointerUp}
   319→          role="img"
   320→          aria-label={t(lang, "chart.title")}
   321→        >
   322→          <defs>
   323→            <clipPath id="wy-plot-clip">
   324→              <rect x={PAD_L} y={PAD_T} width={plotW} height={plotH} />
   325→            </clipPath>
   326→          </defs>
   327→          {/* grid + price axis */}
   328→          {gridPrices.map((p, k) => (
   329→            <g key={k}>
   330→              <line x1={PAD_L} x2={PAD_L + plotW} y1={y(p)} y2={y(p)} stroke={GRID} strokeWidth={1} />
   331→              <text x={PAD_L + plotW + 6} y={y(p) + 4} fill={TEXT} fontSize={10} fontFamily="ui-monospace, monospace">
   332→                {fmtPrice(p)}
   333→              </text>
   334→            </g>
   335→          ))}
   336→          {/* time axis — edge labels anchor inward, greedy de-dupe avoids collisions */}
   337→          {(() => {
   338→            const last = width < 520 ? 3 : 5;
   339→            const est = (s: string) => s.length * 6.2; // ~px per char @ 10px mono
   340→            let prevRight = -Infinity;
   341→            const nodes: React.ReactNode[] = [];
   342→            for (let k = 0; k <= last; k++) {
   343→              const idx = Math.round(viewStart + ((viewEnd - viewStart) * k) / last);
   344→              const kk = klines[Math.min(idx, n - 1)];
   345→              const label = timeLabel(kk.t);
   346→              const w = est(label);
   347→              const anchor = k === 0 ? "start" : k === last ? "end" : "middle";
   348→              const cx = k === 0 ? PAD_L : k === last ? PAD_L + plotW : Math.min(x(idx), PAD_L + plotW - 10);
   349→              const left = anchor === "start" ? cx : anchor === "end" ? cx - w : cx - w / 2;
   350→              if (left < prevRight + 10) continue; // would collide with the previous label
   351→              prevRight = left + w;
   352→              nodes.push(
   353→                <text
   354→                  key={k}
   355→                  x={cx}
   356→                  y={height - 8}
   357→                  fill={TEXT}
   358→                  fontSize={10}
   359→                  textAnchor={anchor}
   360→                  fontFamily="ui-monospace, monospace"
   361→                >
   362→                  {label}
   363→                </text>,
   364→              );
   365→            }
   366→            return nodes;
   367→          })()}
   368→
   369→          {/* ── Whale taker-delta histogram (bottom of the plot) ───────── */}
   370→          {whale.maxD > 0 && (
   371→            <g clipPath="url(#wy-plot-clip)">
   372→              <line
   373→                x1={PAD_L}
   374→                x2={PAD_L + plotW}
   375→                y1={PAD_T + plotH}
   376→                y2={PAD_T + plotH}
   377→                stroke={GRID}
   378→                strokeWidth={1}
   379→              />
   380→              {whale.rows.map(({ i, d }) => {
   381→                const h = Math.max(1, (Math.abs(d) / whale.maxD) * whaleH);
   382→                const bw = Math.max(1.5, cw * 0.55);
   383→                return (
   384→                  <rect
   385→                    key={`wh-${i}`}
   386→                    x={x(i) - bw / 2}
   387→                    y={PAD_T + plotH - h}
   388→                    width={bw}
   389→                    height={h}
   390→                    fill={d >= 0 ? UP : DOWN}
   391→                    opacity={0.34}
   392→                  />
   393→                );
   394→              })}
   395→            </g>
   396→          )}
   397→
   398→          {/* ── Volume profile (volume by price, right-anchored) ────────── */}
   399→          {vp && (
   400→            <g clipPath="url(#wy-plot-clip)">
   401→              {vp.rows.map((r, k) => {
   402→                const yTop = y(r.high);
   403→                const yBot = y(r.low);
   404→                const h = Math.max(1, yBot - yTop - 1);
   405→                const tot = r.buy + r.sell;
   406→                if (tot <= 0) return null;
   407→                const w = (tot / vp.max) * vpW;
   408→                const bw = (r.buy / tot) * w;
   409→                const right = PAD_L + plotW;
   410→                return (
   411→                  <g key={`vp-${k}`}>
   412→                    <rect x={right - bw} y={yTop} width={Math.max(0, bw)} height={h} fill={UP} opacity={0.2} />
   413→                    <rect x={right - w} y={yTop} width={Math.max(0, w - bw)} height={h} fill={DOWN} opacity={0.18} />
   414→                  </g>
   415→                );
   416→              })}
   417→              <line
   418→                x1={PAD_L}
   419→                x2={PAD_L + plotW}
   420→                y1={y(vp.poc)}
   421→                y2={y(vp.poc)}
   422→                stroke="#e4e4e7"
   423→                strokeOpacity={0.22}
   424→                strokeDasharray="3 5"
   425→                strokeWidth={1}
   426→              />
   427→            </g>
   428→          )}
   429→
   430→          {/* ── Wyckoff trading-range boxes (under everything) ─────────── */}
   431→          {wyOn && (
   432→            <g clipPath="url(#wy-plot-clip)">
   433→              {wyBoxes.map((b, k) => {
   434→                const x1 = Math.max(PAD_L, x(b.left) - cw / 2);
   435→                const x2 = Math.min(PAD_L + plotW, x(b.right) + cw / 2);
   436→                const yTop = y(b.top);
   437→                const yBot = y(b.bottom);
   438→                const col = b.dir === 1 ? WY_ACC : WY_DIST;
   439→                return (
   440→                  <rect
   441→                    key={`wyb-${b.left}-${b.dir}-${k}`}
   442→                    x={x1}
   443→                    y={Math.min(yTop, yBot)}
   444→                    width={Math.max(0, x2 - x1)}
   445→                    height={Math.max(1, Math.abs(yBot - yTop))}
   446→                    fill={col}
   447→                    fillOpacity={0.07}
   448→                    stroke={col}
   449→                    strokeOpacity={0.35}
   450→                    strokeWidth={1}
   451→                  />
   452→                );
   453→              })}
   454→            </g>
   455→          )}
   456→
   457→          {/* closed papers connectors */}
   458→          {papersInView.map((p) => {
   459→            const isFlip = p.exitKind === "flip";
   460→            const color = p.side === "LONG" ? UP : DOWN;
   461→            if (p.exitIndex === undefined || p.exit === undefined) {
   462→              // open paper handled below
   463→              return null;
   464→            }
   465→            const a = Math.max(p.entryIndex, viewStart);
   466→            const b = Math.min(p.exitIndex, viewEnd);
   467→            if (b < viewStart || a > viewEnd) return null;
   468→            return (
   469→              <g key={p.id} opacity={0.85}>
   470→                <line
   471→                  x1={x(a)}
   472→                  x2={x(b)}
   473→                  y1={y(p.entry)}
   474→                  y2={y(p.exit)}
   475→                  stroke={color}
   476→                  strokeWidth={1.2}
   477→                  strokeDasharray={isFlip ? "5 3" : "2 3"}
   478→                  opacity={0.7}
   479→                />
   480→                <circle cx={x(a)} cy={y(p.entry)} r={3} fill={color} />
   481→                <circle
   482→                  cx={x(Math.min(p.exitIndex, viewEnd))}
   483→                  cy={y(p.exit)}
   484→                  r={3.5}
   485→                  fill={p.exitKind === "flip" ? FLIP : color}
   486→                  stroke="#18181b"
   487→                  strokeWidth={1}
   488→                >
   489→                  <title>{`#${p.side} ${fmtPrice(p.entry)} → ${fmtPrice(p.exit)}`}</title>
   490→                </circle>
   491→              </g>
   492→            );
   493→          })}
   494→
   495→          {/* open paper levels — clipped to the plot, hidden when out of range */}
   496→          {openPaper && (
   497→            <g clipPath="url(#wy-plot-clip)">
   498→              {[
   499→                { p: openPaper.entry, c: "#e4e4e7", label: openEntryLabel, dash: "6 4" },
   500→                { p: openPaper.sl, c: DOWN, label: "SL", dash: "4 4" },
   501→                { p: openPaper.tp, c: UP, label: "TP", dash: "4 4" },
   502→              ].map((row, k) => {
   503→                const ly = y(row.p);
   504→                if (ly < PAD_T - 1 || ly > PAD_T + plotH + 1) return null;
   505→                return (
   506→                  <g key={k}>
   507→                    <line
   508→                      x1={PAD_L}
   509→                      x2={PAD_L + plotW}
   510→                      y1={ly}
   511→                      y2={ly}
   512→                      stroke={row.c}
   513→                      strokeWidth={1.2}
   514→                      strokeDasharray={row.dash}
   515→                      opacity={0.9}
