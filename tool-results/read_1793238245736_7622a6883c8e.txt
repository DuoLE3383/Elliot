    55→// ── engine replica (kept in sync with src/lib) ──────────────────────────────
    56→function sma(klines, period) {
    57→  const out = new Array(klines.length).fill(null);
    58→  let sum = 0;
    59→  for (let i = 0; i < klines.length; i++) {
    60→    sum += klines[i].c;
    61→    if (i >= period) sum -= klines[i - period].c;
    62→    if (i >= period - 1) out[i] = sum / period;
    63→  }
    64→  return out;
    65→}
    66→
    67→function atr(klines, period) {
    68→  const out = new Array(klines.length).fill(null);
    69→  const tr = klines.map((k, i) =>
    70→    i === 0 ? k.h - k.l : Math.max(k.h - k.l, Math.abs(k.h - klines[i - 1].c), Math.abs(k.l - klines[i - 1].c)),
    71→  );
    72→  let acc = 0;
    73→  for (let i = 0; i < klines.length; i++) {
    74→    if (i < period) {
    75→      acc += tr[i];
    76→      if (i === period - 1) out[i] = acc / period;
    77→    } else out[i] = (out[i - 1] * (period - 1) + tr[i]) / period;
    78→  }
    79→  return out;
    80→}

    82→function signalsFrom(klines, atrKey, atrPeriod, smaPeriod) {
    83→  const a = atr(klines, atrPeriod);
    84→  const smaArr = sma(klines, smaPeriod);
    85→  const n = klines.length;
    86→  const stop = new Array(n).fill(null);
    87→  const trend = new Array(n).fill(0);
    88→  let prevStop = 0, prevTrend = 0;
    89→  for (let i = 0; i < n; i++) {
    90→    if (a[i] == null) continue;
    91→    const close = klines[i].c, loss = atrKey * a[i];
    92→    if (prevTrend === 0) { prevStop = close - loss; prevTrend = 1; stop[i] = prevStop; trend[i] = 1; continue; }
    93→    let s = prevStop, t = prevTrend;
    94→    if (t === 1) { s = Math.max(prevStop, close - loss); if (close < s) { t = -1; s = close + loss; } }
    95→    else { s = Math.min(prevStop, close + loss); if (close > s) { t = 1; s = close - loss; } }
    96→    stop[i] = s; trend[i] = t; prevStop = s; prevTrend = t;
    97→  }
    98→  const signals = [];
    99→  for (let i = 1; i < n; i++) {
   100→    if (trend[i - 1] === 0) continue;
   101→    const k = klines[i], sv = smaArr[i];
   102→    // null SMA must NOT block (matches Pine computing SMA over full history)
   103→    if (trend[i] === 1 && trend[i - 1] === -1)
   104→      signals.push({ i, t: k.t, dir: "BUY", smaOk: sv == null || k.c > sv, price: k.c });
   105→    else if (trend[i] === -1 && trend[i - 1] === 1)
   106→      signals.push({ i, t: k.t, dir: "SELL", smaOk: sv == null || k.c < sv, price: k.c });
   107→  }
   108→  return { signals, sma: smaArr };
   109→}

   111→function papersFrom(klines, signals, smaArr, symbol, interval, slPct, tpPct, leverage = 1, smaFilter = true) {
   112→  const papers = [];
   113→  let open, pending;
   114→  const sigByIndex = new Map(signals.map((s) => [s.i, s]));
   115→  const last = klines.length - 1;
   116→  const gateOkAt = (side, idx) => {
   117→    if (!smaFilter) return true;
   118→    const sv = smaArr[idx];
   119→    if (sv == null) return true;
   120→    return side === "LONG" ? klines[idx].c > sv : klines[idx].c < sv;
   121→  };
   122→  const closeAt = (p, price, index, kind, liquidated = false) => {
   123→    p.status = "closed"; p.exit = price; p.exitIndex = index; p.exitTime = klines[index].t; p.exitKind = kind;
   124→    const pricePct = p.side === "LONG"
   125→      ? ((price - p.entry) / p.entry) * 100
   126→      : ((p.entry - price) / p.entry) * 100;
   127→    p.pricePnlPct = pricePct;
   128→    if (liquidated) { p.liquidated = true; p.pnlPct = -100; }
   129→    else p.pnlPct = pricePct * p.leverage;
   130→    open = undefined;
   131→  };
   132→  const enterAt = (side, price, index, flippedFrom) => {
   133→    const p = {
   134→      id: `${symbol}-${interval}-${side}-${klines[index].t}`, symbol, interval, side,
   135→      entry: price, entryTime: klines[index].t, entryIndex: index, leverage,
   136→      sl: side === "LONG" ? price * (1 - slPct / 100) : price * (1 + slPct / 100),
   137→      tp: side === "LONG" ? price * (1 + tpPct / 100) : price * (1 - tpPct / 100),
   138→      status: "open", flippedFrom,
   139→    };
   140→    papers.push(p); open = p; pending = undefined;
   141→  };
   142→  for (let i = 0; i < klines.length; i++) {
   143→    const k = klines[i];
   144→    if (pending && !open && i > pending.signalIndex && gateOkAt(pending.side, i - 1))
   145→      enterAt(pending.side, k.o, i);
   146→    if (open && open.entryIndex <= i) {
   147→      const liqPrice = open.side === "LONG"
   148→        ? open.entry * (1 - 1 / open.leverage)
   149→        : open.entry * (1 + 1 / open.leverage);
   150→      if (open.side === "LONG") {
   151→        const stopLevel = Math.max(open.sl, liqPrice);
   152→        const isLiq = liqPrice > open.sl + 1e-12;
   153→        if (k.l <= stopLevel) closeAt(open, stopLevel, i, "sl", isLiq);
   154→        else if (k.h >= open.tp) closeAt(open, open.tp, i, "tp");
   155→      } else {
   156→        const stopLevel = Math.min(open.sl, liqPrice);
   157→        const isLiq = liqPrice < open.sl - 1e-12;
   158→        if (k.h >= stopLevel) closeAt(open, stopLevel, i, "sl", isLiq);
   159→        else if (k.l <= open.tp) closeAt(open, open.tp, i, "tp");
   160→      }
   161→    }
   162→    const sig = sigByIndex.get(i);
   163→    if (!sig || i >= last) continue;
   164→    const newSide = sig.dir === "BUY" ? "LONG" : "SHORT";
   165→    if (open && open.side !== newSide) {
   166→      // reversal: ALWAYS close + open opposite (gate never blocks a flip)
   167→      const old = open, fi = i + 1, fp = klines[fi].o;
   168→      closeAt(old, fp, fi, "flip");
   169→      enterAt(newSide, fp, fi, old.id);
   170→    } else if (open) {
   171→      // pyramiding=0
   172→    } else if (pending && pending.side !== newSide) pending = { side: newSide, signalIndex: i };
   173→    else if (pending) pending = { ...pending, signalIndex: i };
   174→    else if (gateOkAt(newSide, i)) enterAt(newSide, klines[i + 1].o, i + 1);
   175→    else pending = { side: newSide, signalIndex: i };
   176→  }
   177→  return { papers, open, pending };
   178→}

   181→function ema(klines, period) {
   182→  const out = new Array(klines.length).fill(null);
   183→  if (klines.length < period) return out;
   184→  let sum = 0;
   185→  for (let i = 0; i < period; i++) sum += klines[i].c;
   186→  let prev = sum / period;
   187→  out[period - 1] = prev;
   188→  const k = 2 / (period + 1);
   189→  for (let i = period; i < klines.length; i++) {
   190→    prev = klines[i].c * k + prev * (1 - k);
   191→    out[i] = prev;
   192→  }
   193→  return out;
   194→}

   196→function rsi(klines, period) {
   197→  const out = new Array(klines.length).fill(null);
   198→  if (klines.length <= period) return out;
   199→  let gain = 0, loss = 0;
   200→  for (let i = 1; i <= period; i++) {
   201→    const d = klines[i].c - klines[i - 1].c;
   202→    if (d >= 0) gain += d; else loss -= d;
   203→  }
   204→  let avgG = gain / period, avgL = loss / period;
   205→  out[period] = avgL === 0 ? 100 : 100 - 100 / (1 + avgG / avgL);
   206→  for (let i = period + 1; i < klines.length; i++) {
   207→    const d = klines[i].c - klines[i - 1].c;
   208→    const g = d > 0 ? d : 0, l = d < 0 ? -d : 0;
   209→    avgG = (avgG * (period - 1) + g) / period;
   210→    avgL = (avgL * (period - 1) + l) / period;
   211→    out[i] = avgL === 0 ? 100 : 100 - 100 / (1 + avgG / avgL);
   212→  }
   213→  return out;
   214→}

   216→function pivotRange(klines, endIndex, pivot = 5, lookback = 150) {
   217→  const end = endIndex + 1;
   218→  const start = Math.max(pivot, end - 1 - lookback);
   219→  let high = null, low = null;
   220→  for (let j = end - 1 - pivot; j >= start; j--) {
   221→    let isHigh = true, isLow = true;
   222→    for (let k = j - pivot; k <= j + pivot; k++) {
   223→      if (k < 0 || k >= end || k === j) continue;
   224→      if (klines[k].h >= klines[j].h) isHigh = false;
   225→      if (klines[k].l <= klines[j].l) isLow = false;
   226→    }
   227→    if (high == null && isHigh) high = klines[j].h;
   228→    if (low == null && isLow) low = klines[j].l;
   229→    if (high != null && low != null) break;
   230→  }
   231→  return high != null && low != null && high > low ? { high, low } : null;
   232→}

   234→function runAnalysisSuite(klines, tag) {
   235→  const closed = klines.length - 2;
   236→  // EMA seeds with the SMA of the first `period` closes and stays finite
   237→  const e50 = ema(klines, 50);
   238→  let smaSeed = 0;
   239→  for (let i = 0; i < 50; i++) smaSeed += klines[i].c;
   240→  check(Math.abs(e50[49] - smaSeed / 50) < 1e-9, `${tag}: ema50 seed == sma50`);
   241→  check(e50[closed] != null && isFinite(e50[closed]), `${tag}: ema50 finite on closed candle`);
   242→
   243→  // RSI bounds + extremes on monotone synthetic runs
   244→  const r = rsi(klines, 14);
   245→  let inBounds = true, hasVal = false;
   246→  for (const v of r) {
   247→    if (v == null) continue;
   248→    hasVal = true;
   249→    if (!(v >= 0 && v <= 100)) inBounds = false;
   250→  }
   251→  check(hasVal && inBounds, `${tag}: rsi14 within [0,100]`);
   252→  const upRun = Array.from({ length: 40 }, (_, i) => ({ t: i, o: 1 + i, h: 1.5 + i, l: 0.9 + i, c: 2 + i, v: 1 }));
   253→  const downRun = Array.from({ length: 40 }, (_, i) => ({ t: i, o: 41 - i, h: 41.5 - i, l: 40.9 - i, c: 40 - i, v: 1 }));
   254→  check(rsi(upRun, 14)[39] === 100, `${tag}: rsi == 100 on pure uptrend`);
   255→  check(rsi(downRun, 14)[39] === 0, `${tag}: rsi == 0 on pure downtrend`);
   256→
   257→  // pivot structure on live data
   258→  const range = pivotRange(klines, closed);
   259→  check(range != null, `${tag}: pivot range found on 500 candles`);
   260→  if (range) {
   261→    check(range.high > range.low, `${tag}: range high > low`);
   262→    const c = klines[closed].c;
   263→    const pos = ((c - range.low) / (range.high - range.low)) * 100;
   264→    check(isFinite(pos), `${tag}: range position finite`);
   265→    check(
   266→      (pos >= 0 && pos <= 100) || pos > 100 || pos < 0,
   267→      `${tag}: structure classifiable`,
   268→      `pos=${pos}`,
   269→    );
   270→  }
   271→
   272→  // directional gate on live data: BUYs below SMA must not have entered directly
   273→  const { signals, sma } = signalsFrom(klines, 2, 10, 200);
   274→  for (const s of signals) {
   275→    const sv = sma[s.i];
   276→    if (sv == null) continue;
   277→    if (s.dir === "BUY") check(s.smaOk === (klines[s.i].c > sv), `${tag}: BUY smaOk directional`, `i=${s.i}`);
   278→    else check(s.smaOk === (klines[s.i].c < sv), `${tag}: SELL smaOk directional`, `i=${s.i}`);
   279→  }
   280→  // a downtrend stretch must produce SELL smaOk=true — the gate is NOT a wall
   281→  const sells = signals.filter((s) => s.dir === "SELL");
   282→  check(sells.length > 0, `${tag}: has SELL signals`);
   283→}

   294→async function runSuite(klines, symbol, interval, cfg) {
   295→  const { slPct, tpPct, leverage, smaFilter } = cfg;
   296→  const { signals, sma } = signalsFrom(klines, 2, 10, 200);
   297→  const { papers, open, pending } = papersFrom(klines, signals, sma, symbol, interval, slPct, tpPct, leverage, smaFilter);
   298→
   299→  const tag = `${symbol} ${interval} ${cfg.name}`;
   300→  const flips = papers.filter((p) => p.exitKind === "flip");
   301→  const closed = papers.filter((p) => p.status === "closed");
   302→  const tpC = closed.filter((p) => p.exitKind === "tp").length;
   303→  const slC = closed.filter((p) => p.exitKind === "sl").length;
   304→  const liqC = closed.filter((p) => p.liquidated).length;
   305→
   306→  check(papers.length > 0, `${tag}: has papers`, `n=${papers.length}`);
   307→
   308→  // determinism
   309→  const again = papersFrom(klines, signals, sma, symbol, interval, slPct, tpPct, leverage, smaFilter);
   310→  check(JSON.stringify(again.papers) === JSON.stringify(papers), `${tag}: deterministic`);
   311→
   312→  // max 1 open at any time
   313→  let concurrent = 0;
   314→  const events = [];
   315→  for (const p of papers) {
   316→    events.push({ t: p.entryTime, d: 1 });
   317→    if (p.exitTime) events.push({ t: p.exitTime, d: -1 });
   318→  }
   319→  events.sort((a, b) => a.t - b.t || a.d - b.d);
   320→  for (const e of events) {
   321→    concurrent += e.d;
   322→    check(concurrent <= 1, `${tag}: <=1 open position`, `concurrent=${concurrent}`);
   323→  }
   324→
   325→  // reversal contract — EVERY flip has an immediate opposite child
   326→  for (const f of flips) {
   327→    check(f.exit != null, `${tag}: flip has exit price`);
   328→    const sig = signals.find((s) => s.i === f.exitIndex - 1);
   329→    check(sig != null, `${tag}: flip has originating signal`);
   330→    if (!sig) continue;
   331→    const kOpen = klines[f.exitIndex].o;
   332→    check(Math.abs(f.exit - kOpen) < 1e-9, `${tag}: flip exit == next candle open`,
   333→      `exit=${f.exit} open=${kOpen}`);
   334→    check(f.exitTime === klines[f.exitIndex].t, `${tag}: flip exitTime == candle open time`);
   335→    // gate NEVER blocks a reversal → child always exists, same price/time
   336→    const child = papers.find((p) => p.flippedFrom === f.id);
   337→    check(child != null, `${tag}: flip always opens child (gate-proof)`, `sigIdx=${sig.i} smaOk=${sig.smaOk}`);
   338→    if (child) {
   339→      check(child.side !== f.side, `${tag}: child side is opposite`);
   340→      check(child.entryIndex === f.exitIndex, `${tag}: child enters on the flip candle`,
   341→        `child=${child.entryIndex} flip=${f.exitIndex}`);
   342→      check(Math.abs(child.entry - f.exit) < 1e-9, `${tag}: child entry == flip exit price`);
   343→      check(child.entryTime === f.exitTime, `${tag}: child entryTime == flip exitTime`);
   344→      check(child.leverage === f.leverage, `${tag}: child inherits leverage`);
   345→    }
   346→  }
   347→
   348→  // flippedFrom links
   349→  for (const p of papers) {
   350→    if (!p.flippedFrom) continue;
   351→    const old = papers.find((q) => q.id === p.flippedFrom);
   352→    check(old != null, `${tag}: flippedFrom link valid`, p.id);
   353→    if (old) check(old.side !== p.side, `${tag}: flippedFrom side opposite`);
   354→    check(old?.exitKind === "flip", `${tag}: flippedFrom source closed as flip`);
   355→  }
   356→
   357→  // entries: direct (signal at entryIndex-1) or queued-pending (gate-on only)
   358→  const gateOkAt = (side, idx) => {
   359→    if (!smaFilter) return true;
   360→    const sv = sma[idx];
   361→    if (sv == null) return true;
   362→    return side === "LONG" ? klines[idx].c > sv : klines[idx].c < sv;
   363→  };
   364→  for (const p of papers) {
   365→    if (p.flippedFrom) continue; // flip entries checked above
   366→    const sig = signals.find((s) => s.i === p.entryIndex - 1);
   367→    let pendingOk = false;
   368→    if (!sig && smaFilter) {
   369→      const prior = signals.filter(
   370→        (s) => s.i < p.entryIndex && s.dir === (p.side === "LONG" ? "BUY" : "SELL"),
   371→      );
   372→      if (prior.length) {
   373→        const s0 = prior[prior.length - 1];
   374→        if (gateOkAt(p.side, p.entryIndex - 1)) {
   375→          let blocked = true;
   376→          for (let k = s0.i; k <= p.entryIndex - 2; k++) {
   377→            if (gateOkAt(p.side, k)) { blocked = false; break; }
   378→          }
   379→          pendingOk = blocked;
   380→        }
   381→      }
   382→    }
   383→    check(sig != null || pendingOk, `${tag}: entry is direct or queued-pending`, p.id);
   384→    if (sig && smaFilter) {
   385→      check(sig.dir === (p.side === "LONG" ? "BUY" : "SELL"), `${tag}: entry direction matches signal`);
   386→      check(sig.smaOk, `${tag}: direct entry signal passed gate`, p.id);
   387→    }
   388→    if (!smaFilter) check(pending == null || pending.side != null, `${tag}: gate-off pending shape`);
   389→    check(Math.abs(p.entry - klines[p.entryIndex].o) < 1e-9, `${tag}: entry == candle open`);
   390→  }
   391→
   392→  // gate-off → no pending entries at all
   393→  if (!smaFilter) check(pending == null, `${tag}: gate off → no pending`, JSON.stringify(pending));
   394→
   395→  // SL/TP geometry (price distances)
   396→  for (const p of papers) {
   397→    if (p.side === "LONG") check(p.sl < p.entry && p.tp > p.entry, `${tag}: LONG sl<entry<tp`);
   398→    else check(p.sl > p.entry && p.tp < p.entry, `${tag}: SHORT sl>entry>tp`);
   399→    check(Math.abs(Math.abs(p.sl / p.entry - 1) * 100 - slPct) < 1e-6, `${tag}: SL distance == slPct`);
   400→    check(Math.abs(Math.abs(p.tp / p.entry - 1) * 100 - tpPct) < 1e-6, `${tag}: TP distance == tpPct`);
   401→    check(p.leverage === leverage, `${tag}: leverage snapshot == settings`);
   402→  }
   403→
   404→  // exit kinds + PnL math
   405→  for (const p of closed) {
   406→    check(["tp", "sl", "flip"].includes(p.exitKind), `${tag}: valid exitKind`);
   407→    if (p.liquidated) {
   408→      check(p.exitKind === "sl", `${tag}: liquidation exits as sl`, p.exitKind);
   409→      check(p.pnlPct === -100, `${tag}: liquidation ROE == -100`, String(p.pnlPct));
   410→      const dist = Math.abs(p.pricePnlPct);
   411→      check(Math.abs(dist - 100 / leverage) < 1e-6, `${tag}: liq distance == 100/lev`,
   412→        `dist=${dist} expect=${100 / leverage}`);
   413→    } else {
   414→      check(
   415→        Math.abs(p.pnlPct - p.pricePnlPct * leverage) < 1e-6,
   416→        `${tag}: ROE == price move × leverage`,
   417→        `pnl=${p.pnlPct} price=${p.pricePnlPct} lev=${p.leverage}`,
   418→      );
   419→    }
   420→    if (p.exitKind === "tp") check(p.pnlPct > 0 || Math.abs(p.pnlPct) < 1e-9, `${tag}: TP exit non-negative PnL`);
   421→    if (p.exitKind === "sl") check(p.pnlPct < 0 || Math.abs(p.pnlPct) < 1e-9, `${tag}: SL exit non-positive PnL`);
   422→  }
   423→
   424→  // KPI math
   425→  const wins = closed.filter((p) => p.pnlPct > 0).length;
   426→  check(tpC + slC + flips.length === closed.length, `${tag}: closed == tp+sl+flip`);
   427→  check(wins + closed.filter((p) => p.pnlPct <= 0).length === closed.length, `${tag}: wins+losses == closed`);
   428→  check(!open || papers.filter((p) => p.status === "open").length === 1, `${tag}: at most 1 open paper`);
   429→
   430→  return { tag, papers: papers.length, closed: closed.length, flips: flips.length, tpC, slC, liqC, wins, open: !!open, pending: !!pending };
   431→}

   433→// ── Wyckoff suite (imports the real src/lib/wyckoff.ts) ───────────────────────
   434→function runWyckoffSuite(klines, tag) {
   435→  const cfgs = [
   436→    { name: "auto", s: { ...DEFAULT_WYCKOFF } },
   437→    { name: "noPhaseC", s: { ...DEFAULT_WYCKOFF, reqPhaseC: false } },
   438→    { name: "fast", s: { ...DEFAULT_WYCKOFF, speed: "fast" } },
   439→    { name: "manual", s: { ...DEFAULT_WYCKOFF, mode: "manual" } },
   440→  ];
   441→  const KINDS = new Set(["SC", "AR", "ST", "Spring", "SOS", "LPS", "Markup", "BC", "UT", "SOW", "LPSY", "Markdown"]);
   442→  const LOW_EVENTS = ["SC", "ST", "Spring", "LPS"]; // acc events sitting on a pivot low
   443→  const HIGH_DIST_EVENTS = ["BC", "ST", "UT", "LPSY"]; // dist events sitting on a real-space HIGH
   444→
   445→  for (const c of cfgs) {
   446→    const wy1 = wyckoffFrom(klines, "5m", c.s);
   447→    const wy2 = wyckoffFrom(klines, "5m", c.s);
   448→    const ser = (w) =>
   449→      JSON.stringify({ e: w.events, b: w.boxes, p: w.pivots, d: w.dash, t: Array.from(w.tints) });
   450→    check(ser(wy1) === ser(wy2), `wy ${tag} ${c.name}: deterministic`);
   451→
   452→    let lastBar = -1;
   453→    for (const e of wy1.events) {
   454→      check(KINDS.has(e.kind), `wy ${tag} ${c.name}: kind enum`, e.kind);
   455→      check(e.bar >= 0 && e.bar <= wy1.lastClosed, `wy ${tag} ${c.name}: bar in range`, `${e.kind} bar=${e.bar}`);
   456→      check(isFinite(e.price) && e.price > 0, `wy ${tag} ${c.name}: finite price`, `${e.kind} ${e.price}`);
   457→      check(e.bar >= lastBar, `wy ${tag} ${c.name}: events sorted by bar`);
   458→      lastBar = e.bar;
   459→
   460→      // anchor: the event price must be EXACTLY the extreme of its own bar —
   461→      // acc low events sit on the pivot low, dist low events mirror to the high, etc.
   462→      const k = klines[e.bar];
   463→      const anchor =
   464→        e.dir === 1
   465→          ? LOW_EVENTS.includes(e.kind) ? k.l : k.h
   466→          : HIGH_DIST_EVENTS.includes(e.kind) ? k.h : k.l;
   467→      check(Math.abs(e.price - anchor) < 1e-9, `wy ${tag} ${c.name}: ${e.kind} anchored to bar extreme`,
   468→        `price=${e.price} anchor=${anchor}`);
   469→
   470→      // label orientation (Pine f_lbl: below = (d==1) == atLowT)
   471→      const belowExp =
   472→        e.dir === 1
   473→          ? LOW_EVENTS.includes(e.kind)
   474→          : ["AR", "SOW", "Markdown"].includes(e.kind);
   475→      check(e.below === belowExp, `wy ${tag} ${c.name}: below flag for ${e.kind}`);
   476→    }
   477→    for (const b of wy1.boxes) {
   478→      check(b.top > b.bottom, `wy ${tag} ${c.name}: box top>bottom`, `${b.top}/${b.bottom}`);
   479→      check(b.left <= b.right, `wy ${tag} ${c.name}: box left<=right`);
   480→      check(isFinite(b.top) && isFinite(b.bottom), `wy ${tag} ${c.name}: box finite`);
   481→    }
   482→    check(wy1.tints.length === klines.length, `wy ${tag} ${c.name}: tints aligned`);
   483→    check(Array.from(wy1.tints).every((t) => t === 1 || t === -1 || t === 0), `wy ${tag} ${c.name}: tint domain`);
   484→    const d = wy1.dash;
   485→    check(d.st >= 0 && d.st <= 7, `wy ${tag} ${c.name}: dash st range`, String(d.st));
   486→    check(d.rsi == null || (d.rsi >= 0 && d.rsi <= 100), `wy ${tag} ${c.name}: dash rsi bounds`);
   487→    check(d.pivLen >= 2, `wy ${tag} ${c.name}: pivLen sane`, String(d.pivLen));
   488→    check(d.volRatio == null || isFinite(d.volRatio), `wy ${tag} ${c.name}: volRatio finite`);
   489→  }
   490→
   491→  // reqPhaseC=true: every live SOS/SOW must carry its Spring/UT (same structure,
   492→  // same direction, pivot bar not after the SOS bar). Trajectory-level event
   493→  // counts are NOT monotone in reqPhaseC — the earlier st5 entry changes which
   494→  // failure rule applies — so only the rule-level invariant is asserted.
   495→  const wyPc = wyckoffFrom(klines, "5m", { ...DEFAULT_WYCKOFF });
   496→  const wyNoPc = wyckoffFrom(klines, "5m", { ...DEFAULT_WYCKOFF, reqPhaseC: false });
   497→  const liveEv = (w) => w.events.filter((e) => !e.deleted);
   498→  for (const e of liveEv(wyPc)) {
   499→    if (e.kind !== "SOS" && e.kind !== "SOW") continue;
   500→    const hasPhC = liveEv(wyPc).some(
   501→      (p) => p.dir === e.dir && (p.kind === "Spring" || p.kind === "UT") && p.bar <= e.bar,
   502→    );
   503→    check(hasPhC, `${tag} wy: reqPhaseC SOS has Spring/UT`, `${e.kind}@${e.bar}`);
   504→  }
   505→  // reqPhaseC=false only OPENS a door — the run itself must stay deterministic
   506→  const wyNoPc2 = wyckoffFrom(klines, "5m", { ...DEFAULT_WYCKOFF, reqPhaseC: false });
   507→  check(
   508→    JSON.stringify(liveEv(wyNoPc)) === JSON.stringify(liveEv(wyNoPc2)),
   509→    `${tag} wy: reqPhaseC=false deterministic`,
   510→  );
   511→
   512→  // speed changes the pivot length (auto mapping: 8 → fast 5, safe 12 on 5m)
   513→  const wyFast = wyckoffFrom(klines, "5m", { ...DEFAULT_WYCKOFF, speed: "fast" });
   514→  const wySafe = wyckoffFrom(klines, "5m", { ...DEFAULT_WYCKOFF, speed: "safe" });
   515→  check(wyFast.dash.pivLen < wySafe.dash.pivLen, `${tag} wy: fast pivLen < safe pivLen`,
   516→    `${wyFast.dash.pivLen}/${wySafe.dash.pivLen}`);
   517→}

   519→/** Synthetic accumulation: flat → climax dump → base → rally → flat peak.
   520→ *  Must yield SC (bar 79, price = bar low) and AR (flat peak), and the exact
   521→ *  inverted series must yield BC/AR(dist) at mirrored prices (T-space proof). */
   522→function runWyckoffSynthetic() {
   523→  const tag = "wy synth";
   524→  const mk = (invert) => {
   525→    const raw = [];
   526→    const t0 = 1700000000000;
   527→    const step = 300000;
   528→    let i = 0;
   529→    const push = (o, h, l, c, v) => {
   530→      raw.push(
   531→        invert
   532→          ? { t: t0 + i * step, o: 200 - o, h: 200 - l, l: 200 - h, c: 200 - c, v }
   533→          : { t: t0 + i * step, o, h, l, c, v },
   534→      );
   535→      i++;
   536→    };
   537→    for (let k = 0; k < 70; k++) push(99.9, 100.2, 99.8, 100, 1); // quiet
   538→    for (let k = 1; k <= 10; k++)
   539→      push(100 - (k - 1) * 1.5, 100 - (k - 1) * 1.5 + 0.3, 100 - k * 1.5 - 0.4, 100 - k * 1.5, 8); // climax dump
   540→    for (let k = 0; k < 30; k++) {
   541→      const c = 85.5 + (k % 3) * 0.5;
   542→      push(c - 0.2, c + 0.3, c - 0.5, c, 0.8); // base
   543→    }
   544→    for (let k = 1; k <= 20; k++) {
   545→      const c = 86 + k * 0.45;
   546→      push(86 + (k - 1) * 0.45, c + 0.2, c - 0.4, c, 2); // rally
   547→    }
   548→    for (let k = 0; k < 14; k++) {
   549→      const h = 95.4 - Math.abs(k - 3) * 0.06;
   550→      push(94.8, h, 94.6, 94.85, 1); // flat with a single confirmed peak at bar 133
   551→    }
   552→    return raw;
   553→  };
   554→  const s = { ...DEFAULT_WYCKOFF };
   555→
   556→  const acc = wyckoffFrom(mk(false), "5m", s);
   557→  const liveAcc = acc.events.filter((e) => !e.deleted);
   558→  const sc = liveAcc.find((e) => e.kind === "SC" && e.dir === 1);
   559→  const ar = liveAcc.find((e) => e.kind === "AR" && e.dir === 1);
   560→  check(sc != null, `${tag}: SC detected`, JSON.stringify(liveAcc.map((e) => e.kind)));
   561→  check(ar != null, `${tag}: AR detected`, JSON.stringify(liveAcc.map((e) => e.kind)));
   562→  if (sc) {
   563→    check(sc.bar === 79, `${tag}: SC at dump-bottom bar`, String(sc.bar));
   564→    check(Math.abs(sc.price - 84.6) < 1e-9, `${tag}: SC price == bar low`, String(sc.price));
   565→    check(sc.below, `${tag}: SC label below`);
   566→  }
   567→  if (ar) {
   568→    check(ar.bar === 133, `${tag}: AR at flat-peak bar`, String(ar.bar));
   569→    check(Math.abs(ar.price - 95.4) < 1e-9, `${tag}: AR price == bar high`, String(ar.price));
   570→    check(!ar.below, `${tag}: AR label above`);
   571→  }
   572→  if (sc && ar) check(ar.bar > sc.bar, `${tag}: AR after SC`);
   573→  const accBox = acc.boxes.find((b) => !b.deleted && b.dir === 1);
   574→  check(accBox != null, `${tag}: acc TR box exists`);
   575→  if (accBox) {
   576→    check(accBox.top > accBox.bottom, `${tag}: acc box geometry`);
   577→    check(Math.abs(accBox.top - 95.4) < 1e-9, `${tag}: acc box top == AR high`, String(accBox.top));
   578→    check(Math.abs(accBox.bottom - 84.6) < 1e-9, `${tag}: acc box bottom == SC low`, String(accBox.bottom));
   579→  }
   580→
   581→  // T-space proof: the exact inverted series must give BC at the mirrored top
   582→  const dist = wyckoffFrom(mk(true), "5m", s);
   583→  const liveDist = dist.events.filter((e) => !e.deleted);
   584→  const bc = liveDist.find((e) => e.kind === "BC" && e.dir === -1);
   585→  check(bc != null, `${tag}: BC detected on inverted series`, JSON.stringify(liveDist.map((e) => `${e.dir}:${e.kind}`)));
   586→  if (bc) {
   587→    check(bc.bar === 79, `${tag}: BC at pump-top bar`, String(bc.bar));
   588→    check(Math.abs(bc.price - 115.4) < 1e-9, `${tag}: BC price == mirrored bar high`, String(bc.price));
   589→    check(!bc.below, `${tag}: BC label above`);
   590→  }
   591→}

   593→// ── market context: sessions / whale delta / volume profile ─────────────────
   594→function runMarketSuite(klines, tag) {
   595→  const closed = klines.length - 2;
   596→  if (closed < 50) return;
   597→
   598→  // session windows: UTC hour boundaries
   599→  const at = (h) => Date.UTC(2025, 0, 15, h, 0, 0);
   600→  const lny = sessionOf(at(13));
   601→  check(lny.includes("london") && lny.includes("newyork") && lny.length === 2, "sessions: 13h = london+ny", JSON.stringify(lny));
   602→  const syd = sessionOf(at(23));
   603→  check(syd.length === 1 && syd[0] === "sydney", "sessions: 23h = sydney", JSON.stringify(syd));
   604→  const tok = sessionOf(at(2));
   605→  check(tok.includes("tokyo") && tok.includes("sydney"), "sessions: 02h = sydney+tokyo", JSON.stringify(tok));
   606→
   607→  // stats: bull% within [0,100], every closed candle counted at least once
   608→  const stats = sessionStats(klines);
   609→  check(stats.length === 4, "sessions: 4 sessions returned");
   610→  let nSum = 0;
   611→  for (const s of stats) {
   612→    if (s.bullPct != null) {
   613→      check(s.bullPct >= 0 && s.bullPct <= 100, `${tag}: bull% in [0,100] (${s.key})`, String(s.bullPct));
   614→      check(isFinite(s.driftPct ?? 0), `${tag}: drift finite (${s.key})`);
   615→    }
   616→    nSum += s.candles;
   617→  }
   618→  check(nSum >= closed, `${tag}: session candle coverage ≥ closed count`, `${nSum} vs ${closed}`);
   619→
   620→  // whale delta bounded by total volume; null without taker data
   621→  const withTb = klines.map((k) => ({ ...k, tb: k.v * 0.6 }));
   622→  const wd = whaleDelta(withTb, 20);
   623→  check(wd != null, `${tag}: whale delta present with taker data`);
   624→  if (wd) {
   625→    let volSum = 0;
   626→    for (let i = Math.max(0, closed - 19); i <= closed; i++) volSum += klines[i].v;
   627→    check(Math.abs(wd.net) <= volSum + 1e-9, `${tag}: |net delta| ≤ window volume`, `${wd.net} vs ${volSum}`);
   628→    check(wd.strongBuy >= 0 && wd.strongSell >= 0, `${tag}: strong deltas non-negative`);
   629→    check(Math.abs(wd.last - (2 * 0.6 - 1) * klines[closed].v) < 1e-6, `${tag}: last delta = 2tb − v`);
   630→  }
   631→  // fetched klines carry no tb field → whale panel must degrade to null
   632→  check(whaleDelta(klines, 20) === null, `${tag}: whale delta null without taker data`);
   633→
   634→  // volume profile distributes exactly the window volume
   635→  const vp = volumeProfile(withTb, 0, closed, 26);
   636→  check(vp != null, `${tag}: volume profile built`);
   637→  if (vp) {
   638→    check(vp.rows.length === 26, `${tag}: vp rows = 26`);
   639→    let tot = 0;
   640→    for (const r of vp.rows) tot += r.buy + r.sell;
   641→    let vSum = 0;
   642→    for (let i = 0; i <= closed; i++) vSum += klines[i].v;
   643→    check(Math.abs(tot - vSum) < vSum * 1e-9, `${tag}: vp conserves volume`, `${tot} vs ${vSum}`);
   644→    check(vp.max > 0 && vp.poc >= vp.minP && vp.poc <= vp.maxP, `${tag}: vp poc in range`);
   645→    const buyRatioTot = vp.rows.reduce((a, r) => a + r.buy, 0) / Math.max(tot, 1e-9);
   646→    check(Math.abs(buyRatioTot - 0.6) < 0.02, `${tag}: vp buy split ≈ taker share`, String(buyRatioTot));
   647→  }
   648→
   649→  // compact formatter
   650→  check(fmtCompact(1234567) === "1.23M", "fmtCompact: M", fmtCompact(1234567));
   651→  check(fmtCompact(-9876) === "-9.9K", "fmtCompact: -K", fmtCompact(-9876));
   652→  check(fmtCompact(512) === "512", "fmtCompact: plain");
   653→}

   656→function runEquitySuite(klines, symbol, interval, cfg) {
   657→  const { slPct, tpPct, leverage, smaFilter } = cfg;
   658→  const tag = `${symbol} ${interval} ${cfg.name} equity`;
   659→  const { signals, sma } = signalsFrom(klines, 2, 10, 200);

   662→  const eq = equityFrom(papers);
   663→  const closed = papers.filter((p) => p.status === "closed");
   664→  check(eq.count === closed.length, `${tag}: point count == closed`, `${eq.count} vs ${closed.length}`);
   665→  check(eq.peak >= 0 && eq.trough <= 0, `${tag}: peak ≥ 0 / trough ≤ 0`, `${eq.peak}/${eq.trough}`);
   666→
   667→  const kpi = kpiFrom(papers);
   668→  check(Math.abs(eq.last - kpi.netPnl) < 1e-9, `${tag}: last cum == kpi netPnl`, `${eq.last} vs ${kpi.netPnl}`);
   669→  const flipNet = closed.filter((p) => p.exitKind === "flip").reduce((a, p) => a + (p.pnlPct ?? 0), 0);
   670→  check(Math.abs(kpi.flipNetPnl - flipNet) < 1e-9, `${tag}: flipNetPnl consistent`);
   671→
   672→  // exit-order times non-decreasing
   673→  let sorted = true;
   674→  for (let i = 1; i < eq.points.length; i++)
   675→    if (eq.points[i].time < eq.points[i - 1].time) sorted = false;
   676→  check(sorted, `${tag}: equity points time-sorted`);
   677→
   678→  // exact synthetic math
   679→  const fake = [
   680→    { status: "closed", pnlPct: 10, entryIndex: 0, entryTime: 500, exitIndex: 1, exitTime: 1500, exitKind: "tp", leverage: 1 },
   681→    { status: "closed", pnlPct: -4, entryIndex: 2, entryTime: 2500, exitIndex: 3, exitTime: 3500, exitKind: "sl", leverage: 1 },
   682→    { status: "open", pnlPct: undefined, entryIndex: 4, entryTime: 4500, exitIndex: undefined, exitTime: undefined, leverage: 1 },
   683→  ];
   684→  const e2 = equityFrom(fake);
   685→  check(e2.count === 2 && Math.abs(e2.last - 6) < 1e-9 && Math.abs(e2.peak - 10) < 1e-9 && Math.abs(e2.trough - 0) < 1e-9, "equity: synthetic cum 10→6, peak 10, trough 0", JSON.stringify(e2));
   686→  const k2 = kpiFrom(fake);
   687→  check(Math.abs(k2.netPnl - 6) < 1e-9 && k2.flipNetPnl === 0, "kpi: synthetic net 6, flipNet 0");
   688→  const fakeFlip = [...fake, { status: "closed", pnlPct: -2, entryIndex: 5, entryTime: 5500, exitIndex: 6, exitTime: 6500, exitKind: "flip", leverage: 1 }];
   689→  check(Math.abs(kpiFrom(fakeFlip).flipNetPnl + 2) < 1e-9, "kpi: flipNetPnl sums flip trades only");
   690→}
