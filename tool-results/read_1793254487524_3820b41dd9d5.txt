   106→   106→  const roe = roeOf(r.side, r.entryPrice, price, r.leverage);

   108→   108→    kind === "tp" ? "⚠️ AUTO SẮP CHẠM TP" : "⚠️ AUTO SẮP CHẠM SL",

   115→   115→const statLines = (st: AutoState, sinceMs: number, title: string): string[] => {
   116→   116→  const recent = st.trades.filter((x) => x.status === "closed" && x.exitTime && new Date(x.exitTime).getTime() >= sinceMs);
   117→   117→  const wins = recent.filter((x) => (x.pnlPct ?? 0) > 0).length;

   119→   119→  const lines = [
   120→   120→    title,
   121→   121→    `Khoảng qua: ${recent.length} lệnh · win ${recent.length ? Math.round((wins / recent.length) * 100) : 0}% · ROE ${net >= 0 ? "+" : ""}${net.toFixed(2)}%`,
   122→   122→  ];
   123→   123→  if (st.open) lines.push(`Đang mở: #${st.open.symbol} ${st.open.interval} ${st.open.side} @ ${fmtPrice(st.open.entryPrice)}`);
   124→   124→  lines.push(`Tổng: ${st.stats.closed} lệnh · win ${st.stats.winRate.toFixed(0)}% · vốn ${st.capital != null ? `${st.capital.toFixed(2)} USDT` : "?"}`);
   125→   125→  const best = (st.byCombo ?? []).filter((c) => c.closed >= 3).sort((a, b) => b.winRate - a.winRate)[0];

   315→   315→          }

   318→   318→      }

   323→   323→        setStatus(t(langRef.current, "auto.scan.busy", { done: 0, total: pairsRef.current.length * INTERVALS.length, current: "…" }));
   324→   324→        current = await doScan();
   325→   325→      }

   330→   330→      const { best, allInCooldown } = bestTradeableAvoid(current, c.minAuc, c.minAcc, st.trades, c.cooldownMin);
   331→   331→      if (!best) {

   333→   333→        return;
   334→   334→      }
   335→   335→      const price = tickersRef.current[best.symbol]?.price ?? 0;
   336→   336→      if (price <= 0) return; // no live price for that pair — wait for the next cycle
   337→   337→      const capitalNow = (await refreshAuto())?.capital ?? c.capital;
   338→   338→      const trade = await openAutoTrade({
   339→   339→        symbol: best.symbol,
   340→   340→        interval: best.interval,
   341→   341→        side: best.bias,
   342→   342→        entryPrice: price,
   343→   343→        slPct: best.slPct,
   344→   344→        tpPct: best.tpPct,
   345→   345→        leverage: leverageRef.current,
   346→   346→        prob: best.prob,
   347→   347→        acc: best.acc,
   348→   348→        auc: best.auc,
   349→   349→        score: best.score,
   350→   350→        capitalBefore: capitalNow,
   351→   351→      });
   352→   352→      if (trade) {
   353→   353→        pushToast({
   354→   354→          kind: "entry",
   355→   355→          text: t(langRef.current, "auto.opened", { side: trade.side, pair: trade.symbol, tf: trade.interval, price: fmtPrice(trade.entryPrice) }),
   356→   356→        });

   358→   358→        await refreshAuto();
   359→   359→        setStatus(`${trade.symbol} ${trade.interval} ${trade.side} @ ${fmtPrice(trade.entryPrice)}`);
   360→   360→      }
   361→   361→    } finally {
   362→   362→      busyRef.current = false;
   363→   363→      nextAtRef.current = Date.now() + cfgRef.current.cycleMin * 60_000;
   364→   364→    }
   365→   365→  }, [closeOpen, doScan, pushToast, refreshAuto]);
   366→   366→
   367→   367→  // auto cycle — immediate tick on enable, then every cycleMin minutes
   368→   368→  useEffect(() => {
   369→   369→    if (!cfg.on) {
   370→   370→      setCountdown(null);
   371→   371→      return;
   372→   372→    }
   373→   373→    nextAtRef.current = Date.now() + 2_000;
   374→   374→    void runCycle();
   375→   375→    const id = window.setInterval(() => void runCycle(), cfg.cycleMin * 60_000);
   376→   376→    return () => window.clearInterval(id);
   377→   377→  }, [cfg.on, cfg.cycleMin, runCycle]);
   378→   378→
   379→   379→  // intra-cycle watcher — SL/TP/liquidation can hit BETWEEN cycles; this books
   380→   380→  // them within ~15s from the live tickers instead of waiting for the next
