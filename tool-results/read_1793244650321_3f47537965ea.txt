    58→  const s = raw.trim().toUpperCase().replace(/[^A-Z0-9]/g, "");

    63→/** Live VN + UTC clocks — 1 Hz external "store", SSR-safe (server snapshot 0). */
    64→function Clocks({ lang }: { lang: Lang }) {
    65→  // seconds-precision snapshot keeps getSnapshot stable within a second
    66→  const sec = useSyncExternalStore(
    67→    (onStoreChange) => {
    68→      const id = window.setInterval(onStoreChange, 1000);
    69→      return () => window.clearInterval(id);
    70→    },
    71→    () => Math.floor(Date.now() / 1000),
    72→    () => 0,
    73→  );
    74→  const fmt = (tz: string) =>
    75→    sec === 0
    76→      ? "--:--:--"
    77→      : new Date(sec * 1000).toLocaleTimeString("en-GB", {
    78→          hour: "2-digit",
    79→          minute: "2-digit",
    80→          second: "2-digit",
    81→          hour12: false,
    82→          timeZone: tz,
    83→        });
    84→  return (
    85→    <div className="hidden flex-col items-end leading-tight lg:flex" aria-hidden>
    86→      <span className="font-mono text-[11px] text-zinc-300">
    87→        <span className="text-zinc-600">{t(lang, "header.clockVN")}</span> {fmt("Asia/Ho_Chi_Minh")}
    88→      </span>
    89→      <span className="font-mono text-[11px] text-zinc-500">
    90→        <span className="text-zinc-600">{t(lang, "header.clockUTC")}</span> {fmt("UTC")}
    91→      </span>
    92→    </div>
    93→  );
    94→}
