   101→const msgNear = (r: AutoTradeRow, price: number, kind: "tp" | "sl"): string => {
   102→  const dir = r.side === "SHORT" ? -1 : 1;
   103→  const targetPrice =
   104→    kind === "tp" ? r.entryPrice * (1 + (dir * Math.abs(r.tpPct)) / 100) : r.entryPrice * (1 - (dir * Math.abs(r.slPct)) / 100);
   105→  const awayPct = price > 0 ? Math.abs((targetPrice - price) / price) * 100 : 0;
   106→  const roe = roeOf(r.side, r.entryPrice, price, r.leverage);
   107→  return [
   108→    kind === "tp" ? "⚠️ AUTO SẮP CHẠM TP" : "⚠️ AUTO SẮP CHẠM SL",
   109→    `#${r.symbol} ${r.interval} ${r.side}`,
   110→    `Giá: ${fmtPrice(price)} · cách ${kind === "tp" ? "TP" : "SL"} ${awayPct.toFixed(2)}%`,
   111→    `ROE: ${roe >= 0 ? "+" : ""}${roe.toFixed(1)}% · đòn bẩy ×${r.leverage % 1 === 0 ? r.leverage : r.leverage.toFixed(1)}`,

   115→const statLines = (st: AutoState, sinceMs: number, title: string): string[] => {
   116→  const recent = st.trades.filter((x) => x.status === "closed" && x.exitTime && new Date(x.exitTime).getTime() >= sinceMs);
   117→  const wins = recent.filter((x) => (x.pnlPct ?? 0) > 0).length;
   118→  const net = recent.reduce((a, x) => a + (x.pnlPct ?? 0), 0);
   119→  const lines = [
   120→    title,
   121→    `Khoảng qua: ${recent.length} lệnh · win ${recent.length ? Math.round((wins / recent.length) * 100) : 0}% · ROE ${net >= 0 ? "+" : ""}${net.toFixed(2)}%`,
   122→  ];
   123→  if (st.open) lines.push(`Đang mở: #${st.open.symbol} ${st.open.interval} ${st.open.side} @ ${fmtPrice(st.open.entryPrice)}`);
   124→  lines.push(`Tổng: ${st.stats.closed} lệnh · win ${st.stats.winRate.toFixed(0)}% · vốn ${st.capital != null ? `${st.capital.toFixed(2)} USDT` : "?"}`);
   125→  const best = (st.byCombo ?? []).filter((c) => c.closed >= 3).sort((a, b) => b.winRate - a.winRate)[0];
   126→  if (best) lines.push(`Tốt nhất: #${best.symbol} ${best.interval} · win ${best.winRate.toFixed(0)}% (${best.closed} lệnh)`);
   127→  return lines;
   128→};

   195→    settingsRef.current = settings;
   196→  }, [settings]);

   201→    langRef.current = lang;
   202→  }, [lang]);

   206→      mountedRef.current = false;

   212→      const st = await fetchAutoState();

   228→    async (open: AutoTradeRow, price: number, kind: "sl" | "tp" | "manual"): Promise<AutoTradeRow | null> => {
   229→      if (closingRef.current) return null;
   230→      closingRef.current = true;
   231→      try {
   232→        const closed = await closeAutoTrade(open.id, price, kind);
   233→        if (closed && mountedRef.current) {
   234→          const st = await refreshAuto(); // stats now include this close
   235→          pushToast({
   236→            kind: "flip",
   237→            text: t(langRef.current, "auto.closed", { pair: closed.symbol, tf: closed.interval, pnl: (closed.pnlPct ?? 0).toFixed(2) }),
   238→          });

   240→        }
   241→        return closed;

   246→    [refreshAuto, pushToast],

   250→  useEffect(() => {
   251→    void refreshAuto();
   252→    void telegramConfigured().then((ok) => mountedRef.current && setTgConfigured(ok));
   253→    void telegramStatus().then((s) => mountedRef.current && setTgChatId(s.chatId));
   254→    // start clocks at the CURRENT hour/day so the first digest goes out at the
   255→    // next boundary, not the second the app opens
   256→    const now = new Date();
   257→    lastHourRef.current = `${now.getFullYear()}-${now.getMonth()}-${now.getDate()}-${now.getHours()}`;
   258→    lastDayRef.current = `${now.getFullYear()}-${now.getMonth()}-${now.getDate()}`;
   259→    try {
   260→      const raw = localStorage.getItem("utp.auto.warn");
   261→      if (raw) warnedRef.current = JSON.parse(raw) as { id: string; tp: boolean; sl: boolean };

   263→      // stale/corrupt flag — start fresh
   264→    }
   265→  }, [refreshAuto]);
   266→
   267→  const doScan = useCallback(async (): Promise<ScanRow[]> => {
   268→    if (scanBusyRef.current) return rowsRef.current;
   269→    scanBusyRef.current = true;
   270→    setScanning({ done: 0, total: pairsRef.current.length * INTERVALS.length, current: "" });

   272→      const out = await scanAll({
   273→        pairs: pairsRef.current,

   279→        setScanAt(Date.now());

   281→      if (out.length > 0) {
   282→        void sendTelegram(msgScan(out, pairsRef.current.length, INTERVALS.length));
   283→      }
   284→      return out;
   285→    } finally {
   286→      scanBusyRef.current = false;
   287→      if (mountedRef.current) setScanning(null);
   288→    }
   289→  }, []);
   290→
   291→  // ── one auto cycle: manage the open position, then maybe enter the best pick
   292→  const runCycle = useCallback(async () => {
   293→    if (busyRef.current) return;
   294→    busyRef.current = true;
   295→    const c = cfgRef.current;
   296→    try {
   297→      const st = await refreshAuto();
   298→      if (!st) {
   299→        setStatus("DB error");
   300→        return;
   301→      }
   302→
   303→      // 1) manage the open position against the live price
   304→      if (st.open) {
   305→        const open = st.open;
   306→        const price = tickersRef.current[open.symbol]?.price ?? 0;
   307→        if (price > 0) {
   308→          const kind = slTpHit(open.side, price, open.entryPrice, open.slPct, open.tpPct);
   309→          const liq = liquidationHit(open.side, price, open.entryPrice, open.leverage);
   310→          if (kind || liq) {
   311→            await closeOpen(open, price, liq ? "sl" : (kind ?? "sl"));
   312→          } else {
   313→            const roe = roeOf(open.side, open.entryPrice, price, open.leverage);
   314→            setStatus(`${open.symbol} ${open.interval} ${open.side} · ROE ${roe >= 0 ? "+" : ""}${roe.toFixed(2)}%`);
   315→          }
   316→        }
   317→        return; // one position at a time — no new entry while one is open
   318→      }
   319→
   320→      // 2) no open position → rank the scan and enter the best tradeable row
   321→      let current = rowsRef.current;
   322→      if (current.length === 0 || Date.now() - (scanAgeRef.current ?? 0) > SCAN_TTL) {
   323→        setStatus(t(langRef.current, "auto.scan.busy", { done: 0, total: pairsRef.current.length * INTERVALS.length, current: "…" }));
   324→        current = await doScan();
   325→      }
   326→      if (current.length === 0) {
   327→        setStatus(t(langRef.current, "auto.scan.empty"));
   328→        return;
   329→      }

   331→      if (!best) {
   332→        setStatus(allInCooldown ? t(langRef.current, "auto.cooldown.all") : t(langRef.current, "auto.rank.hint"));
   333→        return;
   334→      }
   335→      const price = tickersRef.current[best.symbol]?.price ?? 0;
   336→      if (price <= 0) return; // no live price for that pair — wait for the next cycle
   337→      const capitalNow = (await refreshAuto())?.capital ?? c.capital;
   338→      const trade = await openAutoTrade({
   339→        symbol: best.symbol,
   340→        interval: best.interval,
   341→        side: best.bias,
   342→        entryPrice: price,
   343→        slPct: best.slPct,
   344→        tpPct: best.tpPct,
   345→        leverage: leverageRef.current,
   346→        prob: best.prob,
   347→        acc: best.acc,
   348→        auc: best.auc,
   349→        score: best.score,
   350→        capitalBefore: capitalNow,
   351→      });
   352→      if (trade) {
   353→        pushToast({
   354→          kind: "entry",
   355→          text: t(langRef.current, "auto.opened", { side: trade.side, pair: trade.symbol, tf: trade.interval, price: fmtPrice(trade.entryPrice) }),
   356→        });
   357→        void sendTelegram(msgOpen(trade, capitalNow));
   358→        await refreshAuto();
   359→        setStatus(`${trade.symbol} ${trade.interval} ${trade.side} @ ${fmtPrice(trade.entryPrice)}`);
   360→      }
   361→    } finally {
   362→      busyRef.current = false;
   363→      nextAtRef.current = Date.now() + cfgRef.current.cycleMin * 60_000;
   364→    }
   365→  }, [closeOpen, doScan, pushToast, refreshAuto]);

   369→    if (!cfg.on) {
   370→      setCountdown(null);
   371→      return;
   372→    }
   373→    nextAtRef.current = Date.now() + 2_000;

   375→    const id = window.setInterval(() => void runCycle(), cfg.cycleMin * 60_000);
   376→    return () => window.clearInterval(id);
   377→  }, [cfg.on, cfg.cycleMin, runCycle]);
   378→
   379→  // intra-cycle watcher — SL/TP/liquidation can hit BETWEEN cycles; this books
   380→  // them within ~15s from the live tickers instead of waiting for the next
   381→  // full cycle, then kicks a fresh entry a few seconds later. It also fires
   382→  // the one-shot "approaching TP/SL" Telegram warning (70% of the way there).
   383→  useEffect(() => {
   384→    if (!cfg.on) return;
   385→    const id = window.setInterval(() => {
   386→      const open = autoRef.current?.open;
   387→      if (!open || busyRef.current || closingRef.current) return;
   388→      const price = tickersRef.current[open.symbol]?.price ?? 0;
   389→      if (price <= 0) return;
   390→      const kind = slTpHit(open.side, price, open.entryPrice, open.slPct, open.tpPct);
   391→      const liq = liquidationHit(open.side, price, open.entryPrice, open.leverage);
   392→      if (kind || liq) {
   393→        void closeOpen(open, price, liq ? "sl" : (kind ?? "sl")).then((closed) => {
   394→          if (closed && cfgRef.current.on) {
   395→            window.setTimeout(() => void runCycle(), 3_000);
   396→          }
   397→        });
   398→        return;
   399→      }
   400→      // near-hit warning — once per side per position
   401→      if (warnedRef.current.id !== open.id) {
   402→        warnedRef.current = { id: open.id, tp: false, sl: false };
   403→        try {
   404→          localStorage.setItem("utp.auto.warn", JSON.stringify(warnedRef.current));
   405→        } catch {
   406→          // storage blocked — flags live for this session only
   407→        }
   408→      }
   409→      const near = nearHit(open.side, price, open.entryPrice, open.slPct, open.tpPct);
   410→      if (near && !warnedRef.current[near]) {
   411→        warnedRef.current[near] = true;
   412→        try {
   413→          localStorage.setItem("utp.auto.warn", JSON.stringify(warnedRef.current));
   414→        } catch {
   415→          // storage blocked — flags live for this session only
   416→        }
   417→        void sendTelegram(msgNear(open, price, near));
   418→        pushToast({
   419→          kind: "wy",
   420→          text: t(
   421→            langRef.current,
   422→            near === "tp" ? "auto.near.tp" : "auto.near.sl",
   423→            { pair: open.symbol, tf: open.interval, roe: roeOf(open.side, open.entryPrice, price, open.leverage).toFixed(1) },
   424→          ),
   425→        });
   426→      }
   427→    }, 15_000);
   428→    return () => window.clearInterval(id);
   429→  }, [cfg.on, closeOpen, runCycle, pushToast]);

   431→  // hourly / daily Telegram digests — checked every minute; each boundary
   432→  // (top of the hour, calendar day) sends exactly one stats message.
   433→  // force=true → skip the boundary check (manual "send now" button).
   434→  const sendPeriodicStats = useCallback(async (force = false) => {
   435→    if (statBusyRef.current) return;
   436→    const c = cfgRef.current;
   437→    const now = new Date();
   438→    const hourKey = `${now.getFullYear()}-${now.getMonth()}-${now.getDate()}-${now.getHours()}`;
   439→    const dayKey = `${now.getFullYear()}-${now.getMonth()}-${now.getDate()}`;
   440→    if (force) {
   441→      statBusyRef.current = true;
   442→      try {
   443→        lastHourRef.current = hourKey;
   444→        lastDayRef.current = dayKey;
   445→        const st = await refreshAuto();

   865→        ) : (
   866→          rows.length > 0 && <p className="text-[11px] text-zinc-600">{t(lang, "auto.pos.none")}</p>
   867→        )}
   868→
   869→        {/* capital + stats */}
   870→        <div className="grid grid-cols-3 gap-2">
   871→          <div className="rounded-lg border border-zinc-800 bg-zinc-950/40 p-2">
   872→            <p className="truncate text-[10px] uppercase tracking-wide text-zinc-500">{t(lang, "auto.capital.now")}</p>
   873→            <p className="mt-0.5 truncate font-mono text-sm font-bold text-zinc-100">{capitalNow.toFixed(2)}</p>
   874→            <p className="truncate text-[9px] text-zinc-600">{t(lang, "auto.capital.start", { n: cfg.capital })}</p>
   875→          </div>
   876→          <div className="rounded-lg border border-zinc-800 bg-zinc-950/40 p-2">
   877→            <p className="truncate text-[10px] uppercase tracking-wide text-zinc-500">Net ROE</p>
   878→            <p className={`mt-0.5 truncate font-mono text-sm font-bold ${(auto?.stats.netPnlPct ?? 0) >= 0 ? "text-emerald-400" : "text-rose-400"}`}>
   879→              {(auto?.stats.netPnlPct ?? 0) >= 0 ? "+" : ""}
   880→              {(auto?.stats.netPnlPct ?? 0).toFixed(1)}%
   881→            </p>
   882→            <p className="truncate text-[9px] text-zinc-600">{t(lang, "auto.stats.trades", { n: auto?.stats.closed ?? 0 })}</p>
   883→          </div>
   884→          <div className="rounded-lg border border-zinc-800 bg-zinc-950/40 p-2">
   885→            <p className="truncate text-[10px] uppercase tracking-wide text-zinc-500">Win</p>
   886→            <p className="mt-0.5 truncate font-mono text-sm font-bold text-zinc-100">{(auto?.stats.winRate ?? 0).toFixed(0)}%</p>
   887→            <p className="truncate text-[9px] text-zinc-600">
   888→              {auto?.stats.wins ?? 0}/{auto?.stats.closed ?? 0}
   889→            </p>
   890→          </div>
   891→        </div>
   892→
   893→        {/* win-rate by pair × timeframe */}
   894→        {(auto?.byCombo?.length ?? 0) > 0 && (
   895→          <div className="border-t border-zinc-800 pt-2">
   896→            <p className="text-[11px] font-medium uppercase tracking-wide text-zinc-500">{t(lang, "auto.combo.title")}</p>
   897→            <div className="mt-1.5 overflow-hidden rounded-lg border border-zinc-800">
   898→              <div className="grid grid-cols-[1fr_30px_44px_44px_54px] gap-x-1 border-b border-zinc-800 bg-zinc-950/60 px-2 py-1 text-[9px] font-semibold uppercase tracking-wide text-zinc-500">
   899→                <span>{t(lang, "auto.col.pair")}</span>
   900→                <span>{t(lang, "auto.col.tf")}</span>
   901→                <span className="text-right">{t(lang, "auto.combo.col.trades")}</span>
   902→                <span className="text-right">{t(lang, "auto.combo.col.win")}</span>
   903→                <span className="text-right">{t(lang, "auto.combo.col.net")}</span>
   904→              </div>
   905→              <div className="rail-scroll max-h-36 overflow-y-auto">
   906→                {(auto?.byCombo ?? []).map((c: ComboStat) => (
   907→                  <button
   908→                    key={`${c.symbol}-${c.interval}`}
   909→                    type="button"
   910→                    onClick={() => onPickPair(c.symbol, c.interval)}
   911→                    className="grid w-full grid-cols-[1fr_30px_44px_44px_54px] items-center gap-x-1 px-2 py-1 text-left font-mono text-[10px] hover:bg-zinc-800/60"
   912→                  >
   913→                    <span className="truncate text-zinc-200">{c.symbol}</span>
   914→                    <span className="text-zinc-500">{c.interval}</span>
   915→                    <span className="text-right text-zinc-400">{c.closed}</span>
   916→                    <span className={`text-right font-bold ${c.winRate >= 50 ? "text-emerald-400" : "text-rose-400"}`}>{c.winRate.toFixed(0)}%</span>
   917→                    <span className={`text-right ${(c.netPnlPct ?? 0) >= 0 ? "text-emerald-400" : "text-rose-400"}`}>
   918→                      {c.netPnlPct >= 0 ? "+" : ""}
   919→                      {c.netPnlPct.toFixed(1)}%
   920→                    </span>
   921→                  </button>
   922→                ))}
   923→              </div>
   924→            </div>
   925→            <p className="mt-1 text-[10px] leading-snug text-zinc-600">{t(lang, "auto.combo.hint")}</p>
   926→          </div>
   927→        )}
   928→
   929→        {/* history */}
   930→        <div className="border-t border-zinc-800 pt-2">
   931→          <p className="text-[11px] font-medium uppercase tracking-wide text-zinc-500">{t(lang, "auto.hist")}</p>
   932→          {closedHist.length === 0 ? (
   933→            <p className="mt-1 text-[11px] text-zinc-600">{t(lang, "auto.hist.empty")}</p>
   934→          ) : (
   935→            <div className="rail-scroll mt-1.5 max-h-28 space-y-1 overflow-y-auto pr-0.5">
   936→              {closedHist.map((r) => (
   937→                <div key={r.id} className="flex items-center gap-2 rounded-md border border-zinc-800/80 bg-zinc-950/40 px-2 py-1 text-[10px]">
   938→                  <span className="shrink-0 text-zinc-500">
   939→                    {new Date(r.exitTime ?? r.createdAt).toLocaleTimeString(lang === "vi" ? "vi-VN" : "en-GB", { hour: "2-digit", minute: "2-digit", day: "2-digit", month: "2-digit" })}
   940→                  </span>
   941→                  <span className="truncate font-mono text-zinc-300">
   942→                    {r.symbol} {r.interval}
   943→                  </span>
   944→                  <span className={`shrink-0 font-mono ${sideClass(r.side)}`}>{r.side}</span>
   945→                  <span className={`ml-auto shrink-0 font-mono font-bold ${(r.pnlPct ?? 0) >= 0 ? "text-emerald-400" : "text-rose-400"}`}>
   946→                    {(r.pnlPct ?? 0) >= 0 ? "+" : ""}
   947→                    {(r.pnlPct ?? 0).toFixed(2)}%
   948→                  </span>
   949→                  <span className="shrink-0 font-mono text-zinc-600">{r.exitKind === "tp" ? "TP" : r.exitKind === "sl" ? "SL" : "…"}</span>
   950→                </div>
   951→              ))}
   952→            </div>
   953→          )}
   954→        </div>
   955→      </CardContent>
   956→    </Card>
   957→  );
   958→}
