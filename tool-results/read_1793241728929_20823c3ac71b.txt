    54→/** "pepe" → "PEPEUSDT", "PEPEUSDT" → "PEPEUSDT" */
    55→const normalizePair = (raw: string): string => {
    56→  const s = raw.trim().toUpperCase().replace(/[^A-Z0-9]/g, "");

    61→/** Live VN + UTC clocks — 1 Hz external "store", SSR-safe (server snapshot 0). */
    62→function Clocks({ lang }: { lang: Lang }) {
    63→  // seconds-precision snapshot keeps getSnapshot stable within a second
    64→  const sec = useSyncExternalStore(
    65→    (onStoreChange) => {
    66→      const id = window.setInterval(onStoreChange, 1000);
    67→      return () => window.clearInterval(id);
    68→    },
    69→    () => Math.floor(Date.now() / 1000),
    70→    () => 0,
    71→  );
    72→  const fmt = (tz: string) =>
    73→    sec === 0
    74→      ? "--:--:--"
    75→      : new Date(sec * 1000).toLocaleTimeString("en-GB", {
    76→          hour: "2-digit",
    77→          minute: "2-digit",
    78→          second: "2-digit",
    79→          hour12: false,
    80→          timeZone: tz,
    81→        });
    82→  return (
    83→    <div className="hidden flex-col items-end leading-tight lg:flex" aria-hidden>
    84→      <span className="font-mono text-[11px] text-zinc-300">
    85→        <span className="text-zinc-600">{t(lang, "header.clockVN")}</span> {fmt("Asia/Ho_Chi_Minh")}
    86→      </span>
    87→      <span className="font-mono text-[11px] text-zinc-500">
    88→        <span className="text-zinc-600">{t(lang, "header.clockUTC")}</span> {fmt("UTC")}
    89→      </span>
    90→    </div>
    91→  );
    92→}
