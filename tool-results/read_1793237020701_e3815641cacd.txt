    56→    56→/** Live VN + UTC clocks — 1 Hz external "store", SSR-safe (server snapshot 0). */
    57→    57→function Clocks({ lang }: { lang: Lang }) {
    58→    58→  // seconds-precision snapshot keeps getSnapshot stable within a second
    59→    59→  const sec = useSyncExternalStore(
    60→    60→    (onStoreChange) => {
    61→    61→      const id = window.setInterval(onStoreChange, 1000);
    62→    62→      return () => window.clearInterval(id);
    63→    63→    },
    64→    64→    () => Math.floor(Date.now() / 1000),
    65→    65→    () => 0,
    66→    66→  );
    67→    67→  const fmt = (tz: string) =>
    68→    68→    sec === 0
    69→    69→      ? "--:--:--"
    70→    70→      : new Date(sec * 1000).toLocaleTimeString("en-GB", {
    71→    71→          hour: "2-digit",
    72→    72→          minute: "2-digit",
    73→    73→          second: "2-digit",
    74→    74→          hour12: false,
    75→    75→          timeZone: tz,
    76→    76→        });
    77→    77→  return (
    78→    78→    <div className="hidden flex-col items-end leading-tight lg:flex" aria-hidden>
    79→    79→      <span className="font-mono text-[11px] text-zinc-300">
    80→    80→        <span className="text-zinc-600">{t(lang, "header.clockVN")}</span> {fmt("Asia/Ho_Chi_Minh")}
    81→    81→      </span>
    82→    82→      <span className="font-mono text-[11px] text-zinc-500">
    83→    83→        <span className="text-zinc-600">{t(lang, "header.clockUTC")}</span> {fmt("UTC")}
    84→    84→      </span>
    85→    85→    </div>
    86→    86→  );
    87→    87→}
