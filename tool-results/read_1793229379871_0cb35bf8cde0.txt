     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import type { Kline, Paper, Signal } from "@/lib/types";
     5→import type { PendingEntry } from "@/lib/papers";
     6→import type { WyckoffEvent, WyckoffResult } from "@/lib/wyckoff";
     7→import { volumeProfile } from "@/lib/market";
     8→import { fmtPrice, t, type Lang } from "@/lib/i18n";
     9→import { Button } from "@/components/ui/button";
    10→import { Minus, Plus } from "lucide-react";
    11→
    12→const UP = "#10b981";
    13→const DOWN = "#f43f5e";
    14→const FLIP = "#f59e0b";
    15→const GRID = "#27272a";
    16→const TEXT = "#a1a1aa";
    17→// Wyckoff palette — mirrors the Pine original
    18→const WY_ACC = "#22c55e";
    19→const WY_DIST = "#ef4444";
    20→const WY_PHC = "#a855f7";
    21→const WY_ACC_BAR = "#84cc16"; // barcolor lime
    22→const WY_DIST_BAR = "#a855f7"; // barcolor purple
    23→const WY_PIVOT = "#f97316";
    24→const WY_EMA_F = "#14b8a6"; // teal
    25→const WY_EMA_S = "#ef4444"; // red
    26→// SMA band tolerance around the gate SMA (visual aid: how far price is from the gate)
    27→const BAND_PCT = 0.005;
    28→
    29→const wyEventColor = (e: WyckoffEvent) => (e.phaseC ? WY_PHC : e.dir === 1 ? WY_ACC : WY_DIST);
    30→
    31→interface Props {
    32→  klines: Kline[];
    33→  sma: (number | null)[];
    34→  signals: Signal[];
    35→  papers: Paper[];
    36→  openPaper?: Paper;
    37→  pending?: PendingEntry;
    38→  pinnedIndex: number | null;
    39→  onPin: (index: number | null) => void;
    40→  lang: Lang;
    41→  /** Wyckoff overlay (computed in page when enabled) */
    42→  wyckoff?: WyckoffResult | null;
    43→  wyckoffOn?: boolean;
    44→  /** UT Bot trailing stop + trend state (from signalsFrom) — "UT stop" overlay */
    45→  stop?: (number | null)[];
    46→  trend?: number[];
    47→  /** stretch to fill the parent height (terminal layout) */
    48→  fill?: boolean;
    49→}
    50→
    51→const PAD_L = 8;
    52→const PAD_R = 64;
    53→const PAD_T = 12;
    54→const PAD_B = 26;
    55→
    56→const autoBars = (w: number) => (w < 520 ? 52 : w < 768 ? 76 : 110);
    57→
    58→export function PriceChart({
    59→  klines,
    60→  sma,
    61→  signals,
    62→  papers,
    63→  openPaper,
    64→  pending,
    65→  pinnedIndex,
    66→  onPin,
    67→  lang,
    68→  wyckoff,
    69→  wyckoffOn,
    70→  stop,
    71→  trend,
    72→  fill = false,
    73→}: Props) {
    74→  const wrapRef = useRef<HTMLDivElement>(null);
    75→  const plotRef = useRef<HTMLDivElement>(null);
    76→  const [width, setWidth] = useState(900);
    77→  const [measuredH, setMeasuredH] = useState(360);
    78→  const [viewEndRaw, setViewEndRaw] = useState<number | null>(null);
    79→  /** user zoom override — visible bar count (null = auto by width) */
    80→  const [bars, setBars] = useState<number | null>(null);
    81→  const dragRef = useRef<{ x: number; end: number } | null>(null);
    82→
    83→  useEffect(() => {
    84→    const el = wrapRef.current;
    85→    const plot = plotRef.current;
    86→    if (!el) return;
    87→    const ro = new ResizeObserver((entries) => {
    88→      for (const en of entries) {
    89→        if (en.target === el) {
    90→          setWidth(Math.max(320, Math.floor(en.contentRect.width)));
    91→        } else if (en.target === plot) {
    92→          setMeasuredH(Math.max(240, Math.floor(en.contentRect.height)));
    93→        }
    94→      }
    95→    });
    96→    ro.observe(el);
    97→    if (plot) ro.observe(plot);
    98→    return () => ro.disconnect();
    99→  }, []);
   100→
   101→  // wheel zoom — native listener (passive: false) so preventDefault works
   102→  useEffect(() => {
   103→    const el = wrapRef.current;
   104→    if (!el) return;
   105→    const onWheel = (e: WheelEvent) => {
   106→      if (!e.deltaY) return;
   107→      e.preventDefault();
   108→      setBars((prev) => {
   109→        const base = prev ?? autoBars(width);
   110→        const next = e.deltaY > 0 ? base * 1.18 : base / 1.18;
   111→        return Math.max(24, Math.min(340, Math.round(next)));
   112→      });
   113→    };
   114→    el.addEventListener("wheel", onWheel, { passive: false });
   115→    return () => el.removeEventListener("wheel", onWheel);
   116→  }, [width]);
   117→
   118→  const n = klines.length;
   119→  const winSize = Math.max(20, Math.min(bars ?? autoBars(width), n || 1));
   120→  const viewEnd = Math.min(viewEndRaw ?? n - 1, n - 1);
   121→  const viewStart = Math.max(0, viewEnd - winSize + 1);
   122→
   123→  const zoom = useCallback(
   124→    (dir: 1 | -1) =>
   125→      setBars((prev) => {
   126→        const base = prev ?? autoBars(width);
   127→        const next = dir === 1 ? base / 1.35 : base * 1.35;
   128→        return Math.max(24, Math.min(340, Math.round(next)));
   129→      }),
   130→    [width],
   131→  );
   132→
   133→  const jumpLatest = useCallback(() => setViewEndRaw(null), []);
   134→
   135→  const height = fill ? measuredH : width < 520 ? 280 : 360;
   136→  const plotW = width - PAD_L - PAD_R;
   137→  const plotH = height - PAD_T - PAD_B;
   138→  const cw = plotW / Math.max(1, viewEnd - viewStart + 1);
   139→
   140→  const { minP, maxP } = useMemo(() => {
   141→    let lo = Infinity;
   142→    let hi = -Infinity;
   143→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   144→      const k = klines[i];
   145→      if (k.l < lo) lo = k.l;
   146→      if (k.h > hi) hi = k.h;
   147→      const s = sma[i];
   148→      if (s != null && s < lo) lo = s;
   149→      if (s != null && s > hi) hi = s;
   150→      const st = stop?.[i];
   151→      if (st != null && st < lo) lo = st;
   152→      if (st != null && st > hi) hi = st;
   153→    }
   154→    // NOTE: open SL/TP are deliberately NOT part of the scale — a wide 6% SL
   155→    // would squash the candles. Levels are clipped to the plot instead.
   156→    if (openPaper && openPaper.entry >= lo * 0.98 && openPaper.entry <= hi * 1.02) {
   157→      if (openPaper.entry < lo) lo = openPaper.entry;
   158→      if (openPaper.entry > hi) hi = openPaper.entry;
   159→    }
   160→    if (!isFinite(lo) || !isFinite(hi)) return { minP: 0, maxP: 1 };
   161→    if (hi === lo) {
   162→      hi += hi * 0.001 || 1;
   163→      lo -= lo * 0.001 || 1;
   164→    }
   165→    const pad = (hi - lo) * 0.06;
   166→    return { minP: lo - pad, maxP: hi + pad };
   167→  }, [klines, sma, stop, viewStart, viewEnd, n, openPaper]);
   168→
   169→  const x = (i: number) => PAD_L + (i - viewStart) * cw + cw / 2;
   170→  const y = (p: number) => PAD_T + ((maxP - p) / (maxP - minP)) * plotH;
   171→
   172→  const onPointerDown = (e: React.PointerEvent<SVGSVGElement>) => {
   173→    dragRef.current = { x: e.clientX, end: viewEnd };
   174→    (e.target as Element).setPointerCapture?.(e.pointerId);
   175→  };
   176→  const onPointerMove = (e: React.PointerEvent<SVGGElement>) => {
   177→    const d = dragRef.current;
   178→    if (!d) return;
   179→    const shift = Math.round((d.x - e.clientX) / cw);
   180→    const end = Math.min(n - 1, Math.max(winSize - 1, d.end + shift));
   181→    setViewEndRaw(end >= n - 1 ? null : end);
   182→  };
   183→  const onPointerUp = () => {
   184→    dragRef.current = null;
   185→  };
   186→
   187→  const sigInView = signals.filter((s) => s.i >= viewStart && s.i <= viewEnd);
   188→  const papersInView = papers.filter(
   189→    (p) => p.exitIndex !== undefined ? p.exitIndex >= viewStart || p.entryIndex <= viewEnd : p.entryIndex <= viewEnd,
   190→  );
   191→
   192→  // ── Volume profile over the visible window (buy/sell split via taker data) ─
   193→  const vp = useMemo(() => volumeProfile(klines, viewStart, viewEnd, 26), [klines, viewStart, viewEnd]);
   194→  const vpW = plotW * 0.16;
   195→
   196→  // ── Whale delta histogram over the visible window ────────────────────────
   197→  const whale = useMemo(() => {
   198→    let maxD = 0;
   199→    const rows: { i: number; d: number }[] = [];
   200→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   201→      const k = klines[i];
   202→      if (k.tb == null || k.v <= 0) continue;
   203→      const d = 2 * k.tb - k.v;
   204→      rows.push({ i, d });
   205→      if (Math.abs(d) > maxD) maxD = Math.abs(d);
   206→    }
   207→    return { rows, maxD };
   208→  }, [klines, viewStart, viewEnd, n]);
   209→  const whaleH = plotH * 0.2;
   210→
   211→  // ── Wyckoff layers (only when enabled) ─────────────────────────────────
   212→  const wyOn = !!wyckoffOn && !!wyckoff;
   213→  const wyBoxes = wyOn
   214→    ? wyckoff!.boxes.filter((b) => !b.deleted && b.right >= viewStart && b.left <= viewEnd)
   215→    : [];
   216→  const wyEvents = wyOn
   217→    ? wyckoff!.events.filter((e) => !e.deleted && e.bar >= viewStart && e.bar <= viewEnd)
   218→    : [];
   219→  const wyPivots = wyOn
   220→    ? wyckoff!.pivots.filter((p) => !p.hidden && p.bar >= viewStart && p.bar <= viewEnd)
   221→    : [];
   222→  const tintOf = (i: number): 0 | 1 | -1 => (wyOn ? (wyckoff!.tints[i] as 0 | 1 | -1) : 0);
   223→
   224→  /** polyline path skipping nulls (EMA arrays) */
   225→  const linePath = (arr: (number | null)[]): string => {
   226→    let d = "";
   227→    let pen = false;
   228→    for (let i = viewStart; i <= viewEnd; i++) {
   229→      const v = arr[i];
   230→      if (v == null) {
   231→        pen = false;
   232→        continue;
   233→      }
   234→      d += `${pen ? "L" : "M"}${x(i)},${y(v)} `;
   235→      pen = true;
   236→    }
   237→    return d.trim();
   238→  };
   239→
   240→  /** SMA ± band filled area — one closed subpath per contiguous SMA run */
   241→  const bandPath = (): string => {
   242→    let d = "";
   243→    let top: string[] = [];
   244→    let bot: string[] = [];
   245→    const flush = () => {
   246→      if (top.length >= 2) {
   247→        d += `M${top.join(" L")} L${bot.reverse().join(" L")} Z `;
   248→      }
   249→      top = [];
   250→      bot = [];
   251→    };
   252→    for (let i = viewStart; i <= viewEnd; i++) {
   253→      const v = sma[i];
   254→      if (v == null) {
   255→        flush();
   256→        continue;
   257→      }
   258→      top.push(`${x(i)},${y(v * (1 + BAND_PCT))}`);
   259→      bot.push(`${x(i)},${y(v * (1 - BAND_PCT))}`);
   260→    }
   261→    flush();
   262→    return d.trim();
   263→  };
   264→
   265→  /** UT trailing-stop segments, colored per trend run */
   266→  const stopSegs = (): { d: string; up: boolean }[] => {
   267→    const segs: { d: string; up: boolean }[] = [];
   268→    let cur: { d: string; up: boolean } | null = null;
   269→    for (let i = viewStart; i <= viewEnd; i++) {
   270→      const v = stop?.[i];
   271→      const tr = trend?.[i] ?? 0;
   272→      if (v == null || tr === 0) {
   273→        cur = null;
   274→        continue;
   275→      }
   276→      const up = tr === 1;
   277→      if (cur && cur.up === up) cur.d += ` L${x(i)},${y(v)}`;
   278→      else {
   279→        cur = { d: `M${x(i)},${y(v)}`, up };
   280→        segs.push(cur);
   281→      }
   282→    }
   283→    return segs;
   284→  };
   285→
   286→  const timeLabel = (t: number) => {
   287→    const d = new Date(t);
   288→    const hh = String(d.getHours()).padStart(2, "0");
   289→    const mm = String(d.getMinutes()).padStart(2, "0");
   290→    return `${d.getDate()}/${d.getMonth() + 1} ${hh}:${mm}`;
   291→  };
   292→
   293→  if (n === 0) {
   294→    return (
   295→      <div
   296→        ref={wrapRef}
   297→        className={`flex items-center justify-center rounded-lg border border-zinc-800 bg-zinc-950 text-sm text-zinc-500 ${fill ? "h-full" : "h-64"}`}
   298→      >
   299→        {t(lang, "chart.empty")}
   300→      </div>
   301→    );
   302→  }
   303→
   304→  const gridPrices = Array.from({ length: 5 }, (_, k) => minP + ((maxP - minP) * k) / 4);
   305→  const openEntryLabel = openPaper?.flippedFrom ? t(lang, "chart.line.flipEntry") : t(lang, "chart.line.entry");
   306→
   307→  return (
   308→    <div ref={wrapRef} className={`relative flex select-none flex-col ${fill ? "h-full" : ""}`}>
   309→      <div ref={plotRef} className={`relative min-h-0 ${fill ? "flex-1" : ""}`}>
   310→        <svg
   311→          width={width}
   312→          height={height}
   313→          className="touch-pan-y rounded-lg border border-zinc-800 bg-zinc-950"
   314→          onPointerDown={onPointerDown}
   315→          onPointerMove={onPointerMove}
   316→          onPointerUp={onPointerUp}
   317→          onPointerLeave={onPointerUp}
   318→          role="img"
   319→          aria-label={t(lang, "chart.title")}
   320→        >
   321→          <defs>
   322→            <clipPath id="wy-plot-clip">
   323→              <rect x={PAD_L} y={PAD_T} width={plotW} height={plotH} />
   324→            </clipPath>
   325→          </defs>
   326→          {/* grid + price axis */}
   327→          {gridPrices.map((p, k) => (
   328→            <g key={k}>
   329→              <line x1={PAD_L} x2={PAD_L + plotW} y1={y(p)} y2={y(p)} stroke={GRID} strokeWidth={1} />
   330→              <text x={PAD_L + plotW + 6} y={y(p) + 4} fill={TEXT} fontSize={10} fontFamily="ui-monospace, monospace">
   331→                {fmtPrice(p)}
   332→              </text>
   333→            </g>
   334→          ))}
   335→          {/* time axis */}
   336→          {Array.from({ length: width < 520 ? 4 : 6 }, (_, k) => {
   337→            const idx = Math.round(viewStart + ((viewEnd - viewStart) * k) / (width < 520 ? 3 : 5));
   338→            const kk = klines[Math.min(idx, n - 1)];
   339→            return (
   340→              <text
   341→                key={k}
   342→                x={Math.min(x(idx), PAD_L + plotW - 10)}
   343→                y={height - 8}
   344→                fill={TEXT}
   345→                fontSize={10}
   346→                textAnchor="middle"
   347→                fontFamily="ui-monospace, monospace"
   348→              >
   349→                {timeLabel(kk.t)}
   350→              </text>
   351→            );
   352→          })}
   353→
   354→          {/* ── Whale taker-delta histogram (bottom of the plot) ───────── */}
   355→          {whale.maxD > 0 && (
   356→            <g clipPath="url(#wy-plot-clip)">
   357→              <line
   358→                x1={PAD_L}
   359→                x2={PAD_L + plotW}
   360→                y1={PAD_T + plotH}
   361→                y2={PAD_T + plotH}
   362→                stroke={GRID}
   363→                strokeWidth={1}
   364→              />
   365→              {whale.rows.map(({ i, d }) => {
   366→                const h = Math.max(1, (Math.abs(d) / whale.maxD) * whaleH);
   367→                const bw = Math.max(1.5, cw * 0.55);
   368→                return (
   369→                  <rect
   370→                    key={`wh-${i}`}
   371→                    x={x(i) - bw / 2}
   372→                    y={PAD_T + plotH - h}
   373→                    width={bw}
   374→                    height={h}
   375→                    fill={d >= 0 ? UP : DOWN}
   376→                    opacity={0.34}
   377→                  />
   378→                );
   379→              })}
   380→            </g>
   381→          )}
   382→
   383→          {/* ── Volume profile (volume by price, right-anchored) ────────── */}
   384→          {vp && (
   385→            <g clipPath="url(#wy-plot-clip)">
   386→              {vp.rows.map((r, k) => {
   387→                const yTop = y(r.high);
   388→                const yBot = y(r.low);
   389→                const h = Math.max(1, yBot - yTop - 1);
   390→                const tot = r.buy + r.sell;
   391→                if (tot <= 0) return null;
   392→                const w = (tot / vp.max) * vpW;
   393→                const bw = (r.buy / tot) * w;
   394→                const right = PAD_L + plotW;
   395→                return (
   396→                  <g key={`vp-${k}`}>
   397→                    <rect x={right - bw} y={yTop} width={Math.max(0, bw)} height={h} fill={UP} opacity={0.2} />
   398→                    <rect x={right - w} y={yTop} width={Math.max(0, w - bw)} height={h} fill={DOWN} opacity={0.18} />
   399→                  </g>
   400→                );
   401→              })}
   402→              <line
   403→                x1={PAD_L}
   404→                x2={PAD_L + plotW}
   405→                y1={y(vp.poc)}
   406→                y2={y(vp.poc)}
   407→                stroke="#e4e4e7"
   408→                strokeOpacity={0.22}
   409→                strokeDasharray="3 5"
   410→                strokeWidth={1}
   411→              />
   412→            </g>
   413→          )}
   414→
   415→          {/* ── Wyckoff trading-range boxes (under everything) ─────────── */}
   416→          {wyOn && (
   417→            <g clipPath="url(#wy-plot-clip)">
   418→              {wyBoxes.map((b, k) => {
   419→                const x1 = Math.max(PAD_L, x(b.left) - cw / 2);
   420→                const x2 = Math.min(PAD_L + plotW, x(b.right) + cw / 2);
   421→                const yTop = y(b.top);
   422→                const yBot = y(b.bottom);
   423→                const col = b.dir === 1 ? WY_ACC : WY_DIST;
   424→                return (
   425→                  <rect
   426→                    key={`wyb-${b.left}-${b.dir}-${k}`}
   427→                    x={x1}
   428→                    y={Math.min(yTop, yBot)}
   429→                    width={Math.max(0, x2 - x1)}
   430→                    height={Math.max(1, Math.abs(yBot - yTop))}
   431→                    fill={col}
   432→                    fillOpacity={0.07}
   433→                    stroke={col}
   434→                    strokeOpacity={0.35}
   435→                    strokeWidth={1}
   436→                  />
   437→                );
   438→              })}
   439→            </g>
   440→          )}
   441→
   442→          {/* closed papers connectors */}
   443→          {papersInView.map((p) => {
   444→            const isFlip = p.exitKind === "flip";
   445→            const color = p.side === "LONG" ? UP : DOWN;
   446→            if (p.exitIndex === undefined || p.exit === undefined) {
   447→              // open paper handled below
   448→              return null;
   449→            }
   450→            const a = Math.max(p.entryIndex, viewStart);
   451→            const b = Math.min(p.exitIndex, viewEnd);
   452→            if (b < viewStart || a > viewEnd) return null;
   453→            return (
   454→              <g key={p.id} opacity={0.85}>
   455→                <line
   456→                  x1={x(a)}
   457→                  x2={x(b)}
   458→                  y1={y(p.entry)}
   459→                  y2={y(p.exit)}
   460→                  stroke={color}
   461→                  strokeWidth={1.2}
   462→                  strokeDasharray={isFlip ? "5 3" : "2 3"}
   463→                  opacity={0.7}
   464→                />
   465→                <circle cx={x(a)} cy={y(p.entry)} r={3} fill={color} />
   466→                <circle
   467→                  cx={x(Math.min(p.exitIndex, viewEnd))}
   468→                  cy={y(p.exit)}
   469→                  r={3.5}
   470→                  fill={p.exitKind === "flip" ? FLIP : color}
   471→                  stroke="#18181b"
   472→                  strokeWidth={1}
   473→                >
   474→                  <title>{`#${p.side} ${fmtPrice(p.entry)} → ${fmtPrice(p.exit)}`}</title>
   475→                </circle>
   476→              </g>
   477→            );
   478→          })}
   479→
   480→          {/* open paper levels — clipped to the plot, hidden when out of range */}
   481→          {openPaper && (
   482→            <g clipPath="url(#wy-plot-clip)">
   483→              {[
   484→                { p: openPaper.entry, c: "#e4e4e7", label: openEntryLabel, dash: "6 4" },
   485→                { p: openPaper.sl, c: DOWN, label: "SL", dash: "4 4" },
   486→                { p: openPaper.tp, c: UP, label: "TP", dash: "4 4" },
   487→              ].map((row, k) => {
   488→                const ly = y(row.p);
   489→                if (ly < PAD_T - 1 || ly > PAD_T + plotH + 1) return null;
   490→                return (
   491→                  <g key={k}>
   492→                    <line
   493→                      x1={PAD_L}
   494→                      x2={PAD_L + plotW}
   495→                      y1={ly}
   496→                      y2={ly}
   497→                      stroke={row.c}
   498→                      strokeWidth={1.2}
   499→                      strokeDasharray={row.dash}
   500→                      opacity={0.9}
   501→                    />
   502→                    <rect
   503→                      x={PAD_L + plotW + 2}
   504→                      y={ly - 8}
   505→                      width={60}
   506→                      height={16}
   507→                      rx={3}
   508→                      fill={row.c}
   509→                      opacity={0.16}
   510→                    />
   511→                    <text
   512→                      x={PAD_L + plotW + 6}
   513→                      y={ly + 4}
   514→                      fill={row.c}
   515→                      fontSize={9.5}
   516→                      fontWeight={700}
   517→                      fontFamily="ui-monospace, monospace"
   518→                    >
   519→                      {row.label} {fmtPrice(row.p)}
   520→                    </text>
   521→                  </g>
   522→                );
   523→              })}
   524→            </g>
   525→          )}
   526→
   527→          {/* ── SMA ± band (gate tolerance zone) ───────────────────────── */}
   528→          <path d={bandPath()} fill="#facc15" fillOpacity={0.07} stroke="none" />
   529→
   530→          {/* SMA line */}
   531→          <path
   532→            d={sma
   533→              .slice(viewStart, viewEnd + 1)
   534→              .map((v, k) => (v == null ? null : `${k === 0 || sma.slice(viewStart, viewStart + k).every((s) => s == null) ? "M" : "L"}${x(viewStart + k)},${y(v)}`))
   535→              .filter(Boolean)
   536→              .join(" ")}
   537→            fill="none"
   538→            stroke="#facc15"
   539→            strokeWidth={1.4}
   540→            opacity={0.75}
   541→          />
   542→
   543→          {/* ── Wyckoff EMA 50/200 ─────────────────────────────────────── */}
   544→          {wyOn && (
   545→            <g clipPath="url(#wy-plot-clip)">
   546→              <path d={linePath(wyckoff!.emaS)} fill="none" stroke={WY_EMA_S} strokeWidth={1} opacity={0.5} />

   548→            </g>
   549→          )}
   550→
   551→          {/* ── UT Bot trailing stop (trend-colored) ───────────────────── */}
   552→          {stop && trend && (
   553→            <g clipPath="url(#wy-plot-clip)">
   554→              {stopSegs().map((seg, k) => (

   556→                  key={`stop-${k}`}
   557→                  d={seg.d}
   558→                  fill="none"
   559→                  stroke={seg.up ? UP : DOWN}
   560→                  strokeWidth={1.3}
   561→                  strokeDasharray="1 3"
   562→                  opacity={0.85}
   563→                  strokeLinecap="round"
   564→                />
   565→              ))}
   566→            </g>
   567→          )}
   568→
   569→          {/* candles */}
   570→          {klines.slice(viewStart, viewEnd + 1).map((k, idx) => {

   574→            const tint = tintOf(i);
   575→            const color = tint === 1 ? WY_ACC_BAR : tint === -1 ? WY_DIST_BAR : up ? UP : DOWN;
   576→            const bw = Math.max(2, cw * 0.62);
   577→            const yo = y(k.o);
   578→            const yc = y(k.c);
   579→            const top = Math.min(yo, yc);
   580→            const h = Math.max(1.5, Math.abs(yc - yo));
   581→            return (
   582→              <g key={k.t}>
   583→                <line x1={x(i)} x2={x(i)} y1={y(k.h)} y2={y(k.l)} stroke={color} strokeWidth={1} opacity={tint !== 0 ? 0.9 : 1} />
   584→                <rect
   585→                  x={x(i) - bw / 2}
   586→                  y={top}
   587→                  width={bw}
   588→                  height={h}
   589→                  fill={up || tint !== 0 ? (tint !== 0 ? color : "transparent") : color}
   590→                  stroke={color}
   591→                  strokeWidth={1}
   592→                  opacity={tint !== 0 ? 0.85 : 1}
   593→                />
   594→              </g>
   595→            );
   596→          })}
   597→
   598→          {/* ── Wyckoff orange pivots + event labels ───────────────────── */}
   599→          {wyOn && (
   600→            <g clipPath="url(#wy-plot-clip)">
   601→              {wyPivots.map((p, k) => {
   602→                const px = x(p.bar);
   603→                const py = y(p.price);
   604→                // tiny triangle: up under a pivot low, down over a pivot high
   605→                const d = p.isHigh
   606→                  ? `M${px},${py - 2} ${px - 3.5},${py - 9} ${px + 3.5},${py - 9}`
   607→                  : `M${px},${py + 2} ${px - 3.5},${py + 9} ${px + 3.5},${py + 9}`;
   608→                return <path key={`wyp-${p.bar}-${p.isHigh ? "h" : "l"}-${k}`} d={d} fill={WY_PIVOT} opacity={0.75} />;
   609→              })}
   610→              {wyEvents.map((e, k) => {
   611→                const px = x(e.bar);
   612→                const py = y(e.price);
   613→                const col = wyEventColor(e);
   614→                const tri = e.below
   615→                  ? `M${px},${py + 2} ${px - 4},${py + 9} ${px + 4},${py + 9}`
   616→                  : `M${px},${py - 2} ${px - 4},${py - 9} ${px + 4},${py - 9}`;
   617→                return (
   618→                  <g key={`wye-${e.kind}-${e.time}-${e.dir}-${k}`} opacity={0.95}>
   619→                    <path d={tri} fill={col} />
   620→                    <text
   621→                      x={px}
   622→                      y={e.below ? py + 20 : py - 13}
   623→                      textAnchor="middle"
   624→                      fontSize={9}
   625→                      fontWeight={700}
   626→                      fill={col}
   627→                      fontFamily="ui-monospace, monospace"
   628→                    >
   629→                      {e.kind}
   630→                    </text>
   631→                    <title>{`Wyckoff ${e.kind} · ${fmtPrice(e.price)} · ${e.dir === 1 ? "ACC" : "DIST"}`}</title>
   632→                  </g>
   633→                );
   634→              })}
   635→            </g>
   636→          )}
   637→
   638→          {/* signal markers */}
   639→          {sigInView.map((s) => {
   640→            const k = klines[s.i];
   641→            const buy = s.dir === "BUY";
   642→            const isPin = pinnedIndex === s.i;
   643→            const color = buy ? UP : DOWN;
   644→            const mx = x(s.i);
   645→            const my = buy ? y(k.l) + 14 : y(k.h) - 14;
   646→            const tri = buy
   647→              ? `${mx},${my - 6} ${mx - 5},${my + 3} ${mx + 5},${my + 3}`
   648→              : `${mx},${my + 6} ${mx - 5},${my - 3} ${mx + 5},${my - 3}`;
   649→            return (
   650→              <g
   651→                key={`${s.t}-${s.dir}`}
   652→                className="cursor-pointer"
   653→                onPointerDown={(e) => e.stopPropagation()}
   654→                onClick={(e) => {
   655→                  e.stopPropagation();
   656→                  onPin(isPin ? null : s.i);
   657→                }}
   658→              >
   659→                {isPin && <circle cx={mx} cy={my} r={10} fill="none" stroke={FLIP} strokeWidth={1.6} />}
   660→                <path
   661→                  d={`M${tri}`}
   662→                  fill={s.smaOk ? color : "none"}
   663→                  stroke={s.smaOk ? color : FLIP}
   664→                  strokeWidth={s.smaOk ? 0 : 1.2}
   665→                  strokeDasharray={s.smaOk ? undefined : "2 2"}
   666→                >
   667→                  <title>
   668→                    {`${s.dir} · ${fmtPrice(s.price)} · ${s.smaOk ? t(lang, "dock.gate.open") : t(lang, "chart.mark.blocked")}`}
   669→                  </title>
   670→                </path>
   671→              </g>
   672→            );
   673→          })}
   674→        </svg>
   675→      </div>
   676→
   677→      {/* legend + zoom controls + hint */}
   678→      <div className="mt-1.5 flex shrink-0 flex-wrap items-center gap-x-3 gap-y-1 px-1 text-[11px] text-zinc-500">
   679→        <span className="flex shrink-0 items-center gap-0.5">
   680→          <Button
   681→            size="sm"
   682→            variant="outline"
   683→            className="h-6 w-6 rounded-md border-zinc-700 p-0 text-zinc-300 hover:bg-zinc-800"
   684→            title={t(lang, "chart.zoom.in")}
   685→            aria-label={t(lang, "chart.zoom.in")}
   686→            onClick={() => zoom(1)}
   687→          >
   688→            <Plus className="h-3 w-3" />
   689→          </Button>
   690→          <Button
   691→            size="sm"
   692→            variant="outline"
   693→            className="h-6 w-6 rounded-md border-zinc-700 p-0 text-zinc-300 hover:bg-zinc-800"
   694→            title={t(lang, "chart.zoom.out")}
   695→            aria-label={t(lang, "chart.zoom.out")}
   696→            onClick={() => zoom(-1)}
   697→          >
   698→            <Minus className="h-3 w-3" />
   699→          </Button>
   700→          {viewEnd < n - 1 && (
   701→            <Button
   702→              size="sm"
   703→              variant="outline"
   704→              className="ml-1 h-6 rounded-md border-zinc-700 px-2 text-[11px] text-zinc-300 hover:bg-zinc-800"
   705→              onClick={jumpLatest}
   706→            >
   707→              ⟶ {t(lang, "chart.zoom.latest")}
   708→            </Button>
   709→          )}
   710→        </span>
   711→        <span className="flex items-center gap-1.5">
   712→          <span className="inline-block h-2.5 w-2.5 rounded-sm border border-emerald-500" />
   713→          {t(lang, "chart.legend.candles")}
   714→        </span>
   715→        <span className="flex items-center gap-1.5">
   716→          <span className="inline-block h-0.5 w-4 bg-yellow-400" />
   717→          {t(lang, "chart.legend.sma")}
   718→        </span>
   719→        <span className="flex items-center gap-1.5">
   720→          <span className="inline-block h-2 w-2.5 rounded-sm bg-yellow-400/25 ring-1 ring-yellow-500/50" />
   721→          {t(lang, "chart.legend.band")}
   722→        </span>
   723→        {stop && trend && (
   724→          <span className="flex items-center gap-1.5">
   725→            <span className="inline-block h-0.5 w-4 border-t border-dashed border-emerald-400" />
   726→            {t(lang, "chart.legend.stop")}
   727→          </span>

   729→        <span className="flex items-center gap-1.5">
   730→          <span className="inline-block h-2.5 w-2.5 rounded-sm bg-emerald-500" />
   731→          {t(lang, "chart.legend.trades")}
   732→        </span>
   733→        {whale.maxD > 0 && (
   734→          <span className="flex items-center gap-1.5" title={t(lang, "whale.note")}>
   735→            <span className="inline-block h-2.5 w-1.5 rounded-sm bg-emerald-500/40" />

   740→        <span className="flex items-center gap-1.5" title={`${t(lang, "vp.buy")} / ${t(lang, "vp.sell")}`}>
   741→          <span className="inline-block h-2.5 w-2.5 rounded-sm bg-emerald-500/40 ring-1 ring-emerald-500/50" />

   744→        </span>
   745→        {wyOn && (
   746→          <>
   747→            <span className="flex items-center gap-1.5">
   748→              <span className="inline-block h-2.5 w-2.5 rounded-sm bg-emerald-500/25 ring-1 ring-emerald-500/60" />
