   681→export function t(lang: Lang, key: string, vars?: Record<string, string | number>): string {
   682→  let s = DICTS[lang][key] ?? DICTS.vi[key] ?? key;
   683→  if (vars) {
   684→    for (const [k, v] of Object.entries(vars)) {
   685→      s = s.replace(`{${k}}`, String(v));
   686→    }
   687→  }
   688→  return s;
   689→}

   691→export function fmtPrice(n: number): string {
   692→  const abs = Math.abs(n);
   693→  // dynamic precision so sub-cent pairs (PEPE, SHIB…) stay readable
   694→  const max = abs >= 100 ? 2 : abs >= 1 ? 4 : abs >= 0.01 ? 5 : abs >= 0.0001 ? 6 : 8;
   695→  return new Intl.NumberFormat("en-US", {
   696→    minimumFractionDigits: 2,
   697→    maximumFractionDigits: max,
   698→  }).format(n);
   699→}

   702→export function symbolLabel(sym: string): string {
   703→  const base = sym.replace(/(USDT|USDC|FDUSD|TUSD|BUSD)$/i, "");
   704→  return base || sym;
   705→}
