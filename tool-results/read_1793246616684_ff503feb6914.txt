     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import type { Kline, Paper, Signal } from "@/lib/types";
     5→import type { PendingEntry } from "@/lib/papers";
     6→import type { WyckoffEvent, WyckoffResult } from "@/lib/wyckoff";
     7→import { volumeProfile, fmtCompact } from "@/lib/market";
     8→import { fmtPrice, fmtPct, t, type Lang } from "@/lib/i18n";
     9→import { equityFrom } from "@/lib/equity";
    10→import { Button } from "@/components/ui/button";
    11→import { Minus, Plus } from "lucide-react";
    12→
    13→const UP = "#10b981";
    14→const DOWN = "#f43f5e";
    15→const FLIP = "#f59e0b";
    16→const GRID = "#27272a";
    17→const TEXT = "#a1a1aa";
    18→// Wyckoff palette — mirrors the Pine original
    19→const WY_ACC = "#22c55e";
    20→const WY_DIST = "#ef4444";
    21→const WY_PHC = "#a855f7";
    22→const WY_ACC_BAR = "#84cc16"; // barcolor lime
    23→const WY_DIST_BAR = "#a855f7"; // barcolor purple
    24→const WY_PIVOT = "#f97316";
    25→const WY_EMA_F = "#14b8a6"; // teal
    26→const WY_EMA_S = "#ef4444"; // red
    27→// SMA band tolerance around the gate SMA (visual aid: how far price is from the gate)
    28→const BAND_PCT = 0.005;
    29→
    30→const wyEventColor = (e: WyckoffEvent) => (e.phaseC ? WY_PHC : e.dir === 1 ? WY_ACC : WY_DIST);
    31→
    32→interface Props {
    33→  klines: Kline[];
    34→  sma: (number | null)[];
    35→  signals: Signal[];
    36→  papers: Paper[];
    37→  openPaper?: Paper;
    38→  pending?: PendingEntry;
    39→  pinnedIndex: number | null;
    40→  onPin: (index: number | null) => void;
    41→  lang: Lang;
    42→  /** Wyckoff overlay (computed in page when enabled) */
    43→  wyckoff?: WyckoffResult | null;
    44→  wyckoffOn?: boolean;
    45→  /** UT Bot trailing stop + trend state (from signalsFrom) — "UT stop" overlay */
    46→  stop?: (number | null)[];
    47→  trend?: number[];
    48→  /** stretch to fill the parent height (terminal layout) */
    49→  fill?: boolean;
    50→  /** candle body style: "neon" = hollow up-bodies (default) · "normal" = filled green/red */
    51→  candleStyle?: "neon" | "normal";
    52→  /** style switch — omit to hide the toggle */
    53→  onCandleStyle?: (style: "neon" | "normal") => void;
    54→}
    55→
    56→const PAD_L = 8;
    57→const PAD_R = 64;
    58→const PAD_T = 12;
    59→const PAD_B = 26;
    60→
    61→const autoBars = (w: number) => (w < 520 ? 52 : w < 768 ? 76 : 110);
    62→
    63→export function PriceChart({
    64→  klines,
    65→  sma,
    66→  signals,
    67→  papers,
    68→  openPaper,
    69→  pending,
    70→  pinnedIndex,
    71→  onPin,
    72→  lang,
    73→  wyckoff,
    74→  wyckoffOn,
    75→  stop,
    76→  trend,
    77→  fill = false,
    78→  candleStyle = "neon",
    79→  onCandleStyle,
    80→}: Props) {
    81→  const wrapRef = useRef<HTMLDivElement>(null);
    82→  const plotRef = useRef<HTMLDivElement>(null);
    83→  const [width, setWidth] = useState(900);
    84→  const [measuredH, setMeasuredH] = useState(360);
    85→  const [viewEndRaw, setViewEndRaw] = useState<number | null>(null);
    86→  /** user zoom override — visible bar count (null = auto by width) */
    87→  const [bars, setBars] = useState<number | null>(null);
    88→  /** hover crosshair — candle index + cursor y inside the plot */
    89→  const [hover, setHover] = useState<{ i: number; py: number } | null>(null);
    90→  const dragRef = useRef<{ x: number; end: number } | null>(null);
    91→
    92→  useEffect(() => {
    93→    const el = wrapRef.current;
    94→    const plot = plotRef.current;
    95→    if (!el) return;
    96→    const ro = new ResizeObserver((entries) => {
    97→      for (const en of entries) {
    98→        if (en.target === el) {
    99→          setWidth(Math.max(320, Math.floor(en.contentRect.width)));
   100→        } else if (en.target === plot) {
   101→          // floor of 160px keeps the plot usable — going lower would overflow the card
   102→          setMeasuredH(Math.max(160, Math.floor(en.contentRect.height)));
   103→        }
   104→      }
   105→    });
   106→    ro.observe(el);
   107→    if (plot) ro.observe(plot);
   108→    return () => ro.disconnect();
   109→  }, []);
   110→
   111→  // wheel zoom — native listener (passive: false) so preventDefault works
   112→  useEffect(() => {
   113→    const el = wrapRef.current;
   114→    if (!el) return;
   115→    const onWheel = (e: WheelEvent) => {
   116→      if (!e.deltaY) return;
   117→      e.preventDefault();
   118→      setBars((prev) => {
   119→        const base = prev ?? autoBars(width);
   120→        const next = e.deltaY > 0 ? base * 1.18 : base / 1.18;
   121→        return Math.max(24, Math.min(340, Math.round(next)));
   122→      });
   123→    };
   124→    el.addEventListener("wheel", onWheel, { passive: false });
   125→    return () => el.removeEventListener("wheel", onWheel);
   126→  }, [width]);
   127→
   128→  const n = klines.length;
   129→  const winSize = Math.max(20, Math.min(bars ?? autoBars(width), n || 1));
   130→  const viewEnd = Math.min(viewEndRaw ?? n - 1, n - 1);
   131→  const viewStart = Math.max(0, viewEnd - winSize + 1);
   132→
   133→  const zoom = useCallback(
   134→    (dir: 1 | -1) =>
   135→      setBars((prev) => {
   136→        const base = prev ?? autoBars(width);
   137→        const next = dir === 1 ? base / 1.35 : base * 1.35;
   138→        return Math.max(24, Math.min(340, Math.round(next)));
   139→      }),
   140→    [width],
   141→  );
   142→
   143→  const jumpLatest = useCallback(() => setViewEndRaw(null), []);
   144→
   145→  const height = fill ? measuredH : width < 520 ? 280 : 360;
   146→  const plotW = width - PAD_L - PAD_R;
   147→  // ── Equity pane (Đường cong vốn) — a small strip carved out of the plot bottom
   148→  const eq = useMemo(() => equityFrom(papers), [papers]);
   149→  const eqOn = eq.points.length >= 2;
   150→  const EQ_PANE = 58;
   151→  const eqPaneH = eqOn ? EQ_PANE : 0;
   152→  const plotH = height - PAD_T - PAD_B - eqPaneH;
   153→  const cw = plotW / Math.max(1, viewEnd - viewStart + 1);
   154→
   155→  const { minP, maxP } = useMemo(() => {
   156→    let lo = Infinity;
   157→    let hi = -Infinity;
   158→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   159→      const k = klines[i];
   160→      if (k.l < lo) lo = k.l;
   161→      if (k.h > hi) hi = k.h;
   162→      const s = sma[i];
   163→      if (s != null && s < lo) lo = s;
   164→      if (s != null && s > hi) hi = s;
   165→      const st = stop?.[i];
   166→      if (st != null && st < lo) lo = st;
   167→      if (st != null && st > hi) hi = st;
   168→    }
   169→    // NOTE: open SL/TP are deliberately NOT part of the scale — a wide 6% SL
   170→    // would squash the candles. Levels are clipped to the plot instead.
   171→    if (openPaper && openPaper.entry >= lo * 0.98 && openPaper.entry <= hi * 1.02) {
   172→      if (openPaper.entry < lo) lo = openPaper.entry;
   173→      if (openPaper.entry > hi) hi = openPaper.entry;
   174→    }
   175→    if (!isFinite(lo) || !isFinite(hi)) return { minP: 0, maxP: 1 };
   176→    if (hi === lo) {
   177→      hi += hi * 0.001 || 1;
   178→      lo -= lo * 0.001 || 1;
   179→    }
   180→    const pad = (hi - lo) * 0.06;
   181→    return { minP: lo - pad, maxP: hi + pad };
   182→  }, [klines, sma, stop, viewStart, viewEnd, n, openPaper]);
   183→
   184→  const x = (i: number) => PAD_L + (i - viewStart) * cw + cw / 2;
   185→  const y = (p: number) => PAD_T + ((maxP - p) / (maxP - minP)) * plotH;
   186→  /** inverse of y() — price under the crosshair */
   187→  const priceAtY = (py: number) => minP + ((PAD_T + plotH - py) / plotH) * (maxP - minP);
   188→
   189→  const onPointerDown = (e: React.PointerEvent<SVGSVGElement>) => {
   190→    dragRef.current = { x: e.clientX, end: viewEnd };
   191→    (e.target as Element).setPointerCapture?.(e.pointerId);
   192→  };
   193→  const onPointerMove = (e: React.PointerEvent<SVGSVGElement>) => {
   194→    // hover crosshair (TradingView-style) — always tracked, pan continues on top
   195→    const rect = e.currentTarget.getBoundingClientRect();
   196→    const py = Math.max(PAD_T, Math.min(PAD_T + plotH, e.clientY - rect.top));
   197→    const i = Math.max(viewStart, Math.min(viewEnd, viewStart + Math.floor((e.clientX - rect.left - PAD_L) / cw)));
   198→    setHover((prev) => (prev && prev.i === i && Math.abs(prev.py - py) < 0.5 ? prev : { i, py }));
   199→    const d = dragRef.current;
   200→    if (!d) return;
   201→    const shift = Math.round((d.x - e.clientX) / cw);
   202→    const end = Math.min(n - 1, Math.max(winSize - 1, d.end + shift));
   203→    setViewEndRaw(end >= n - 1 ? null : end);
   204→  };
   205→  const onPointerUp = () => {
   206→    dragRef.current = null;
   207→  };
   208→  const onPointerLeave = () => {
   209→    dragRef.current = null;
   210→    setHover(null);
   211→  };
   212→
   213→  // ── Equity pane geometry (Đường cong vốn) ──────────────────────────────
   214→  const eqTop = PAD_T + plotH + 10;
   215→  const eqH = eqPaneH - 16;
   216→  const eqLo = Math.min(eq.trough, 0);
   217→  const eqHi = Math.max(eq.peak, eqLo + 0.5);
   218→  const eqY = (v: number) => eqTop + ((eqHi - v) / (eqHi - eqLo)) * eqH;
   219→  const eqX = (bar: number) => Math.max(PAD_L, Math.min(PAD_L + plotW, x(bar)));
   220→  // trades closed up to the viewed bar — the curve stays honest while panning back
   221→  const eqSeen = eqOn ? eq.points.filter((p) => p.bar <= viewEnd) : [];
   222→  const eqLine = (() => {
   223→    if (!eqOn || eqH <= 4) return null;
   224→    if (eqSeen.length === 0) {
   225→      const d = `M${PAD_L},${eqY(0).toFixed(1)} L${(PAD_L + plotW).toFixed(1)},${eqY(0).toFixed(1)}`;
   226→      return { d, area: `${d} Z`, endV: 0, endY: eqY(0) };
   227→    }
   228→    const pts = eqSeen;
   229→    let d = pts[0].bar > viewStart ? `M${PAD_L},${eqY(0).toFixed(1)} L${eqX(pts[0].bar).toFixed(1)},${eqY(pts[0].cum).toFixed(1)}` : `M${eqX(pts[0].bar).toFixed(1)},${eqY(pts[0].cum).toFixed(1)}`;
   230→    for (let k = 1; k < pts.length; k++) d += ` L${eqX(pts[k].bar).toFixed(1)},${eqY(pts[k].cum).toFixed(1)}`;
   231→    const endY = eqY(pts[pts.length - 1].cum);
   232→    d += ` L${(PAD_L + plotW).toFixed(1)},${endY.toFixed(1)}`; // flat extension to "now"
   233→    const area = `${d} L${(PAD_L + plotW).toFixed(1)},${eqY(0).toFixed(1)} L${PAD_L},${eqY(0).toFixed(1)} Z`;
   234→    return { d, area, endV: pts[pts.length - 1].cum, endY };
   235→  })();
   236→
   237→  const sigInView = signals.filter((s) => s.i >= viewStart && s.i <= viewEnd);
   238→  const eqColor = eqLine ? (eqLine.endV >= 0 ? UP : DOWN) : UP;
   239→  const papersInView = papers.filter(
   240→    (p) => p.exitIndex !== undefined ? p.exitIndex >= viewStart || p.entryIndex <= viewEnd : p.entryIndex <= viewEnd,
   241→  );
   242→
   243→  // ── Volume profile over the visible window (buy/sell split via taker data) ─
   244→  const vp = useMemo(() => volumeProfile(klines, viewStart, viewEnd, 26), [klines, viewStart, viewEnd]);
   245→  const vpW = plotW * 0.16;
   246→
   247→  // ── Whale delta histogram over the visible window ────────────────────────
   248→  const whale = useMemo(() => {
   249→    let maxD = 0;
   250→    const rows: { i: number; d: number }[] = [];
   251→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   252→      const k = klines[i];
   253→      if (k.tb == null || k.v <= 0) continue;
   254→      const d = 2 * k.tb - k.v;
   255→      rows.push({ i, d });
   256→      if (Math.abs(d) > maxD) maxD = Math.abs(d);
   257→    }
   258→    return { rows, maxD };
   259→  }, [klines, viewStart, viewEnd, n]);
   260→  const whaleH = plotH * 0.2;
   261→
   262→  // ── Wyckoff layers (only when enabled) ─────────────────────────────────
   263→  const wyOn = !!wyckoffOn && !!wyckoff;
   264→  const wyBoxes = wyOn
   265→    ? wyckoff!.boxes.filter((b) => !b.deleted && b.right >= viewStart && b.left <= viewEnd)
   266→    : [];
   267→  const wyEvents = wyOn
   268→    ? wyckoff!.events.filter((e) => !e.deleted && e.bar >= viewStart && e.bar <= viewEnd)
   269→    : [];
   270→  const wyPivots = wyOn
   271→    ? wyckoff!.pivots.filter((p) => !p.hidden && p.bar >= viewStart && p.bar <= viewEnd)
   272→    : [];
   273→  const tintOf = (i: number): 0 | 1 | -1 => (wyOn ? (wyckoff!.tints[i] as 0 | 1 | -1) : 0);
   274→
   275→  /** polyline path skipping nulls (EMA arrays) */
   276→  const linePath = (arr: (number | null)[]): string => {
   277→    let d = "";
   278→    let pen = false;
   279→    for (let i = viewStart; i <= viewEnd; i++) {
   280→      const v = arr[i];
   281→      if (v == null) {
   282→        pen = false;
   283→        continue;
   284→      }
   285→      d += `${pen ? "L" : "M"}${x(i)},${y(v)} `;
   286→      pen = true;
   287→    }
   288→    return d.trim();
   289→  };
   290→
   291→  /** SMA ± band filled area — one closed subpath per contiguous SMA run */
   292→  const bandPath = (): string => {
   293→    let d = "";
   294→    let top: string[] = [];
   295→    let bot: string[] = [];
   296→    const flush = () => {
   297→      if (top.length >= 2) {
   298→        d += `M${top.join(" L")} L${bot.reverse().join(" L")} Z `;
   299→      }
   300→      top = [];
   301→      bot = [];
   302→    };
   303→    for (let i = viewStart; i <= viewEnd; i++) {
   304→      const v = sma[i];
   305→      if (v == null) {
   306→        flush();
   307→        continue;
   308→      }
   309→      top.push(`${x(i)},${y(v * (1 + BAND_PCT))}`);
   310→      bot.push(`${x(i)},${y(v * (1 - BAND_PCT))}`);
   311→    }
   312→    flush();
   313→    return d.trim();
   314→  };
   315→
   316→  /** UT trailing-stop segments, colored per trend run */
   317→  const stopSegs = (): { d: string; up: boolean }[] => {
   318→    const segs: { d: string; up: boolean }[] = [];
   319→    let cur: { d: string; up: boolean } | null = null;
   320→    for (let i = viewStart; i <= viewEnd; i++) {
   321→      const v = stop?.[i];
   322→      const tr = trend?.[i] ?? 0;
   323→      if (v == null || tr === 0) {
   324→        cur = null;
   325→        continue;
   326→      }
   327→      const up = tr === 1;
   328→      if (cur && cur.up === up) cur.d += ` L${x(i)},${y(v)}`;
   329→      else {
   330→        cur = { d: `M${x(i)},${y(v)}`, up };
   331→        segs.push(cur);
   332→      }
   333→    }
   334→    return segs;
   335→  };
   336→
   337→  const timeLabel = (t: number) => {
   338→    const d = new Date(t);
   339→    const hh = String(d.getHours()).padStart(2, "0");
   340→    const mm = String(d.getMinutes()).padStart(2, "0");
   341→    return `${d.getDate()}/${d.getMonth() + 1} ${hh}:${mm}`;
   342→  };
   343→
   344→  if (n === 0) {
   345→    return (
   346→      <div
   347→        ref={wrapRef}
   348→        className={`flex items-center justify-center rounded-lg border border-zinc-800 bg-zinc-950 text-sm text-zinc-500 ${fill ? "h-full" : "h-64"}`}
   349→      >
   350→        {t(lang, "chart.empty")}
   351→      </div>
   352→    );
   353→  }
   354→
   355→  const gridPrices = Array.from({ length: 5 }, (_, k) => minP + ((maxP - minP) * k) / 4);
   356→  const openEntryLabel = openPaper?.flippedFrom ? t(lang, "chart.line.flipEntry") : t(lang, "chart.line.entry");
   357→
   358→  return (
   359→    <div ref={wrapRef} className={`relative flex select-none flex-col ${fill ? "h-full" : ""}`}>
   360→      <div ref={plotRef} className={`relative min-h-0 overflow-hidden ${fill ? "flex-1" : ""}`}>
   361→        <svg
   362→          width={width}
   363→          height={height}
   364→          className="touch-pan-y rounded-lg border border-zinc-800 bg-zinc-950"
   365→          onPointerDown={onPointerDown}
   366→          onPointerMove={onPointerMove}
   367→          onPointerUp={onPointerUp}
   368→          onPointerLeave={onPointerLeave}
   369→          role="img"
   370→          aria-label={t(lang, "chart.title")}
   371→        >
   372→          <defs>
   373→            <clipPath id="wy-plot-clip">
   374→              <rect x={PAD_L} y={PAD_T} width={plotW} height={plotH} />
   375→            </clipPath>
   376→          </defs>
   377→          {/* grid + price axis */}
   378→          {gridPrices.map((p, k) => (
   379→            <g key={k}>
   380→              <line x1={PAD_L} x2={PAD_L + plotW} y1={y(p)} y2={y(p)} stroke={GRID} strokeWidth={1} />
   381→              <text x={PAD_L + plotW + 6} y={y(p) + 4} fill={TEXT} fontSize={10} fontFamily="ui-monospace, monospace">
   382→                {fmtPrice(p)}
   383→              </text>
   384→            </g>
   385→          ))}
   386→          {/* time axis — edge labels anchor inward, greedy de-dupe avoids collisions */}
   387→          {(() => {
   388→            const last = width < 520 ? 3 : 5;
   389→            const est = (s: string) => s.length * 6.2; // ~px per char @ 10px mono
   390→            let prevRight = -Infinity;
   391→            const nodes: React.ReactNode[] = [];
   392→            for (let k = 0; k <= last; k++) {
   393→              const idx = Math.round(viewStart + ((viewEnd - viewStart) * k) / last);
   394→              const kk = klines[Math.min(idx, n - 1)];
   395→              const label = timeLabel(kk.t);
   396→              const w = est(label);
   397→              const anchor = k === 0 ? "start" : k === last ? "end" : "middle";
   398→              const cx = k === 0 ? PAD_L : k === last ? PAD_L + plotW : Math.min(x(idx), PAD_L + plotW - 10);
   399→              const left = anchor === "start" ? cx : anchor === "end" ? cx - w : cx - w / 2;
   400→              if (left < prevRight + 10) continue; // would collide with the previous label
   401→              prevRight = left + w;
   402→              nodes.push(
   403→                <text
   404→                  key={k}
   405→                  x={cx}
   406→                  y={height - 8}
   407→                  fill={TEXT}
   408→                  fontSize={10}
   409→                  textAnchor={anchor}
   410→                  fontFamily="ui-monospace, monospace"
   411→                >
   412→                  {label}
   413→                </text>,
   414→              );
   415→            }
   416→            return nodes;
   417→          })()}
   418→
   419→          {/* ── Whale taker-delta histogram (bottom of the plot) ───────── */}
   420→          {whale.maxD > 0 && (
   421→            <g clipPath="url(#wy-plot-clip)">
   422→              <line
   423→                x1={PAD_L}
   424→                x2={PAD_L + plotW}
   425→                y1={PAD_T + plotH}
   426→                y2={PAD_T + plotH}
   427→                stroke={GRID}
   428→                strokeWidth={1}
   429→              />
   430→              {whale.rows.map(({ i, d }) => {
   431→                const h = Math.max(1, (Math.abs(d) / whale.maxD) * whaleH);
   432→                const bw = Math.max(1.5, cw * 0.55);
   433→                return (
   434→                  <rect
   435→                    key={`wh-${i}`}
   436→                    x={x(i) - bw / 2}
   437→                    y={PAD_T + plotH - h}
   438→                    width={bw}
   439→                    height={h}
   440→                    fill={d >= 0 ? UP : DOWN}
   441→                    opacity={0.34}
   442→                  />
   443→                );
   444→              })}
   445→            </g>
   446→          )}
   447→
   448→          {/* ── Equity pane (Đường cong vốn) — cumulative ROE of closed trades ── */}
   449→          {eqOn && (
   450→            <g pointerEvents="none">
   451→              <defs>
   452→                <linearGradient id="eq-pane-fill" x1="0" y1="0" x2="0" y2="1">
   453→                  <stop offset="0%" stopColor={eqColor} stopOpacity={0.25} />
   454→                  <stop offset="100%" stopColor={eqColor} stopOpacity={0.02} />
   455→                </linearGradient>
   456→              </defs>
   457→              <line x1={PAD_L} x2={PAD_L + plotW} y1={PAD_T + plotH + 4} y2={PAD_T + plotH + 4} stroke={GRID} strokeWidth={1} />
   458→              <text x={PAD_L + 2} y={eqTop + 1} fill="#71717a" fontSize={9} fontFamily="ui-monospace, monospace">
   459→                {t(lang, "tabs.equity")} · {eq.count}
   460→              </text>
   461→              <line x1={PAD_L} x2={PAD_L + plotW} y1={eqY(0)} y2={eqY(0)} stroke="#3f3f46" strokeDasharray="4 4" strokeWidth={1} />
   462→              {eqLine && <path d={eqLine.area} fill="url(#eq-pane-fill)" />}
   463→              {eqLine && <path d={eqLine.d} fill="none" stroke={eqColor} strokeWidth={1.6} strokeLinejoin="round" />}
