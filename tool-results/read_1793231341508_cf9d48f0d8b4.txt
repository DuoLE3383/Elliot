    27→    27→const BAND_PCT = 0.005;
    28→    28→

    30→    30→
    31→    31→interface Props {
    32→    32→  klines: Kline[];
    33→    33→  sma: (number | null)[];
    34→    34→  signals: Signal[];
    35→    35→  papers: Paper[];
    36→    36→  openPaper?: Paper;
    37→    37→  pending?: PendingEntry;

    50→    50→
    51→    51→const PAD_L = 8;

    53→    53→const PAD_T = 12;

    57→    57→
    58→    58→export function PriceChart({
    59→    59→  klines,
    60→    60→  sma,
    61→    61→  signals,
    62→    62→  papers,
    63→    63→  openPaper,
    64→    64→  pending,
    65→    65→  pinnedIndex,
    66→    66→  onPin,
    67→    67→  lang,
    68→    68→  wyckoff,
    69→    69→  wyckoffOn,
    70→    70→  stop,
    71→    71→  trend,
    72→    72→  fill = false,
    73→    73→}: Props) {
    74→    74→  const wrapRef = useRef<HTMLDivElement>(null);
    75→    75→  const plotRef = useRef<HTMLDivElement>(null);
    76→    76→  const [width, setWidth] = useState(900);
    77→    77→  const [measuredH, setMeasuredH] = useState(360);
    78→    78→  const [viewEndRaw, setViewEndRaw] = useState<number | null>(null);
    79→    79→  /** user zoom override — visible bar count (null = auto by width) */
    80→    80→  const [bars, setBars] = useState<number | null>(null);
    81→    81→  const dragRef = useRef<{ x: number; end: number } | null>(null);
    82→    82→
    83→    83→  useEffect(() => {
    84→    84→    const el = wrapRef.current;
    85→    85→    const plot = plotRef.current;
    86→    86→    if (!el) return;
    87→    87→    const ro = new ResizeObserver((entries) => {
    88→    88→      for (const en of entries) {
    89→    89→        if (en.target === el) {
    90→    90→          setWidth(Math.max(320, Math.floor(en.contentRect.width)));
    91→    91→        } else if (en.target === plot) {
    92→    92→          // floor of 160px keeps the plot usable — going lower would overflow the card
    93→    93→          setMeasuredH(Math.max(160, Math.floor(en.contentRect.height)));
    94→    94→        }
    95→    95→      }
    96→    96→    });
    97→    97→    ro.observe(el);
    98→    98→    if (plot) ro.observe(plot);
    99→    99→    return () => ro.disconnect();
   100→   100→  }, []);
   101→   101→
   102→   102→  // wheel zoom — native listener (passive: false) so preventDefault works
   103→   103→  useEffect(() => {
   104→   104→    const el = wrapRef.current;
   105→   105→    if (!el) return;
   106→   106→    const onWheel = (e: WheelEvent) => {
   107→   107→      if (!e.deltaY) return;
   108→   108→      e.preventDefault();
   109→   109→      setBars((prev) => {
   110→   110→        const base = prev ?? autoBars(width);
   111→   111→        const next = e.deltaY > 0 ? base * 1.18 : base / 1.18;
   112→   112→        return Math.max(24, Math.min(340, Math.round(next)));
   113→   113→      });
   114→   114→    };
   115→   115→    el.addEventListener("wheel", onWheel, { passive: false });
   116→   116→    return () => el.removeEventListener("wheel", onWheel);
   117→   117→  }, [width]);
   118→   118→
   119→   119→  const n = klines.length;
   120→   120→  const winSize = Math.max(20, Math.min(bars ?? autoBars(width), n || 1));
   121→   121→  const viewEnd = Math.min(viewEndRaw ?? n - 1, n - 1);
   122→   122→  const viewStart = Math.max(0, viewEnd - winSize + 1);
   123→   123→
   124→   124→  const zoom = useCallback(
   125→   125→    (dir: 1 | -1) =>
   126→   126→      setBars((prev) => {
   127→   127→        const base = prev ?? autoBars(width);
   128→   128→        const next = dir === 1 ? base / 1.35 : base * 1.35;
   129→   129→        return Math.max(24, Math.min(340, Math.round(next)));
   130→   130→      }),
   131→   131→    [width],
   132→   132→  );
   133→   133→
   134→   134→  const jumpLatest = useCallback(() => setViewEndRaw(null), []);
   135→   135→
   136→   136→  const height = fill ? measuredH : width < 520 ? 280 : 360;
   137→   137→  const plotW = width - PAD_L - PAD_R;
   138→   138→  const plotH = height - PAD_T - PAD_B;
   139→   139→  const cw = plotW / Math.max(1, viewEnd - viewStart + 1);
   140→   140→
   141→   141→  const { minP, maxP } = useMemo(() => {
   142→   142→    let lo = Infinity;
   143→   143→    let hi = -Infinity;
   144→   144→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   145→   145→      const k = klines[i];
   146→   146→      if (k.l < lo) lo = k.l;
   147→   147→      if (k.h > hi) hi = k.h;
   148→   148→      const s = sma[i];
   149→   149→      if (s != null && s < lo) lo = s;
   150→   150→      if (s != null && s > hi) hi = s;
   151→   151→      const st = stop?.[i];
   152→   152→      if (st != null && st < lo) lo = st;
   153→   153→      if (st != null && st > hi) hi = st;
   154→   154→    }
   155→   155→    // NOTE: open SL/TP are deliberately NOT part of the scale — a wide 6% SL
   156→   156→    // would squash the candles. Levels are clipped to the plot instead.
   157→   157→    if (openPaper && openPaper.entry >= lo * 0.98 && openPaper.entry <= hi * 1.02) {
   158→   158→      if (openPaper.entry < lo) lo = openPaper.entry;
   159→   159→      if (openPaper.entry > hi) hi = openPaper.entry;
   160→   160→    }
   161→   161→    if (!isFinite(lo) || !isFinite(hi)) return { minP: 0, maxP: 1 };
   162→   162→    if (hi === lo) {
   163→   163→      hi += hi * 0.001 || 1;
   164→   164→      lo -= lo * 0.001 || 1;
   165→   165→    }
   166→   166→    const pad = (hi - lo) * 0.06;
   167→   167→    return { minP: lo - pad, maxP: hi + pad };
   168→   168→  }, [klines, sma, stop, viewStart, viewEnd, n, openPaper]);
   169→   169→
   170→   170→  const x = (i: number) => PAD_L + (i - viewStart) * cw + cw / 2;
   171→   171→  const y = (p: number) => PAD_T + ((maxP - p) / (maxP - minP)) * plotH;
   172→   172→
   173→   173→  const onPointerDown = (e: React.PointerEvent<SVGSVGElement>) => {
   174→   174→    dragRef.current = { x: e.clientX, end: viewEnd };
   175→   175→    (e.target as Element).setPointerCapture?.(e.pointerId);
   176→   176→  };
   177→   177→  const onPointerMove = (e: React.PointerEvent<SVGGElement>) => {
   178→   178→    const d = dragRef.current;
   179→   179→    if (!d) return;
   180→   180→    const shift = Math.round((d.x - e.clientX) / cw);
   181→   181→    const end = Math.min(n - 1, Math.max(winSize - 1, d.end + shift));
   182→   182→    setViewEndRaw(end >= n - 1 ? null : end);
   183→   183→  };
   184→   184→  const onPointerUp = () => {
   185→   185→    dragRef.current = null;
   186→   186→  };
   187→   187→
   188→   188→  const sigInView = signals.filter((s) => s.i >= viewStart && s.i <= viewEnd);
   189→   189→  const papersInView = papers.filter(
   190→   190→    (p) => p.exitIndex !== undefined ? p.exitIndex >= viewStart || p.entryIndex <= viewEnd : p.entryIndex <= viewEnd,
   191→   191→  );
   192→   192→
   193→   193→  // ── Volume profile over the visible window (buy/sell split via taker data) ─
   194→   194→  const vp = useMemo(() => volumeProfile(klines, viewStart, viewEnd, 26), [klines, viewStart, viewEnd]);
   195→   195→  const vpW = plotW * 0.16;
   196→   196→
   197→   197→  // ── Whale delta histogram over the visible window ────────────────────────
   198→   198→  const whale = useMemo(() => {
   199→   199→    let maxD = 0;
   200→   200→    const rows: { i: number; d: number }[] = [];
   201→   201→    for (let i = viewStart; i <= viewEnd && i < n; i++) {
   202→   202→      const k = klines[i];
   203→   203→      if (k.tb == null || k.v <= 0) continue;
   204→   204→      const d = 2 * k.tb - k.v;
   205→   205→      rows.push({ i, d });
   206→   206→      if (Math.abs(d) > maxD) maxD = Math.abs(d);
   207→   207→    }
   208→   208→    return { rows, maxD };
   209→   209→  }, [klines, viewStart, viewEnd, n]);
   210→   210→  const whaleH = plotH * 0.2;
   211→   211→
   212→   212→  // ── Wyckoff layers (only when enabled) ─────────────────────────────────
   213→   213→  const wyOn = !!wyckoffOn && !!wyckoff;
   214→   214→  const wyBoxes = wyOn
   215→   215→    ? wyckoff!.boxes.filter((b) => !b.deleted && b.right >= viewStart && b.left <= viewEnd)
   216→   216→    : [];
   217→   217→  const wyEvents = wyOn
   218→   218→    ? wyckoff!.events.filter((e) => !e.deleted && e.bar >= viewStart && e.bar <= viewEnd)
   219→   219→    : [];
   220→   220→  const wyPivots = wyOn
   221→   221→    ? wyckoff!.pivots.filter((p) => !p.hidden && p.bar >= viewStart && p.bar <= viewEnd)
   222→   222→    : [];
   223→   223→  const tintOf = (i: number): 0 | 1 | -1 => (wyOn ? (wyckoff!.tints[i] as 0 | 1 | -1) : 0);
   224→   224→
   225→   225→  /** polyline path skipping nulls (EMA arrays) */
   226→   226→  const linePath = (arr: (number | null)[]): string => {
   227→   227→    let d = "";
   228→   228→    let pen = false;
   229→   229→    for (let i = viewStart; i <= viewEnd; i++) {
   230→   230→      const v = arr[i];
   231→   231→      if (v == null) {
   232→   232→        pen = false;
   233→   233→        continue;
   234→   234→      }
   235→   235→      d += `${pen ? "L" : "M"}${x(i)},${y(v)} `;
   236→   236→      pen = true;
   237→   237→    }
   238→   238→    return d.trim();
   239→   239→  };
   240→   240→
   241→   241→  /** SMA ± band filled area — one closed subpath per contiguous SMA run */
   242→   242→  const bandPath = (): string => {
   243→   243→    let d = "";
   244→   244→    let top: string[] = [];
   245→   245→    let bot: string[] = [];
   246→   246→    const flush = () => {
   247→   247→      if (top.length >= 2) {
   248→   248→        d += `M${top.join(" L")} L${bot.reverse().join(" L")} Z `;
   249→   249→      }
   250→   250→      top = [];
   251→   251→      bot = [];
   252→   252→    };
   253→   253→    for (let i = viewStart; i <= viewEnd; i++) {
   254→   254→      const v = sma[i];
   255→   255→      if (v == null) {
   256→   256→        flush();
   257→   257→        continue;
   258→   258→      }
   259→   259→      top.push(`${x(i)},${y(v * (1 + BAND_PCT))}`);
   260→   260→      bot.push(`${x(i)},${y(v * (1 - BAND_PCT))}`);
   261→   261→    }
   262→   262→    flush();
   263→   263→    return d.trim();
   264→   264→  };
   265→   265→
   266→   266→  /** UT trailing-stop segments, colored per trend run */
   267→   267→  const stopSegs = (): { d: string; up: boolean }[] => {
   268→   268→    const segs: { d: string; up: boolean }[] = [];
   269→   269→    let cur: { d: string; up: boolean } | null = null;
   270→   270→    for (let i = viewStart; i <= viewEnd; i++) {
   271→   271→      const v = stop?.[i];
   272→   272→      const tr = trend?.[i] ?? 0;
   273→   273→      if (v == null || tr === 0) {
   274→   274→        cur = null;
   275→   275→        continue;
   276→   276→      }
   277→   277→      const up = tr === 1;
   278→   278→      if (cur && cur.up === up) cur.d += ` L${x(i)},${y(v)}`;
   279→   279→      else {
   280→   280→        cur = { d: `M${x(i)},${y(v)}`, up };
   281→   281→        segs.push(cur);
   282→   282→      }
   283→   283→    }
   284→   284→    return segs;
   285→   285→  };
   286→   286→
   287→   287→  const timeLabel = (t: number) => {
   288→   288→    const d = new Date(t);
   289→   289→    const hh = String(d.getHours()).padStart(2, "0");
   290→   290→    const mm = String(d.getMinutes()).padStart(2, "0");
   291→   291→    return `${d.getDate()}/${d.getMonth() + 1} ${hh}:${mm}`;
   292→   292→  };
   293→   293→
   294→   294→  if (n === 0) {
   295→   295→    return (
   296→   296→      <div
   297→   297→        ref={wrapRef}
   298→   298→        className={`flex items-center justify-center rounded-lg border border-zinc-800 bg-zinc-950 text-sm text-zinc-500 ${fill ? "h-full" : "h-64"}`}
   299→   299→      >
   300→   300→        {t(lang, "chart.empty")}
   301→   301→      </div>
   302→   302→    );
   303→   303→  }
   304→   304→
   305→   305→  const gridPrices = Array.from({ length: 5 }, (_, k) => minP + ((maxP - minP) * k) / 4);
   306→   306→  const openEntryLabel = openPaper?.flippedFrom ? t(lang, "chart.line.flipEntry") : t(lang, "chart.line.entry");
   307→   307→
   308→   308→  return (
   309→   309→    <div ref={wrapRef} className={`relative flex select-none flex-col ${fill ? "h-full" : ""}`}>
   310→   310→      <div ref={plotRef} className={`relative min-h-0 overflow-hidden ${fill ? "flex-1" : ""}`}>
   311→   311→        <svg
   312→   312→          width={width}
   313→   313→          height={height}
   314→   314→          className="touch-pan-y rounded-lg border border-zinc-800 bg-zinc-950"
   315→   315→          onPointerDown={onPointerDown}
   316→   316→          onPointerMove={onPointerMove}
   317→   317→          onPointerUp={onPointerUp}
   318→   318→          onPointerLeave={onPointerUp}
   319→   319→          role="img"
   320→   320→          aria-label={t(lang, "chart.title")}
   321→   321→        >
   322→   322→          <defs>
   323→   323→            <clipPath id="wy-plot-clip">
   324→   324→              <rect x={PAD_L} y={PAD_T} width={plotW} height={plotH} />
   325→   325→            </clipPath>
   326→   326→          </defs>
   327→   327→          {/* grid + price axis */}
   328→   328→          {gridPrices.map((p, k) => (
   329→   329→            <g key={k}>
   330→   330→              <line x1={PAD_L} x2={PAD_L + plotW} y1={y(p)} y2={y(p)} stroke={GRID} strokeWidth={1} />
   331→   331→              <text x={PAD_L + plotW + 6} y={y(p) + 4} fill={TEXT} fontSize={10} fontFamily="ui-monospace, monospace">
   332→   332→                {fmtPrice(p)}
   333→   333→              </text>
   334→   334→            </g>
   335→   335→          ))}
   336→   336→          {/* time axis — edge labels anchor inward, greedy de-dupe avoids collisions */}
   337→   337→          {(() => {
   338→   338→            const last = width < 520 ? 3 : 5;
   339→   339→            const est = (s: string) => s.length * 6.2; // ~px per char @ 10px mono
   340→   340→            let prevRight = -Infinity;
   341→   341→            const nodes: React.ReactNode[] = [];
   342→   342→            for (let k = 0; k <= last; k++) {
   343→   343→              const idx = Math.round(viewStart + ((viewEnd - viewStart) * k) / last);
   344→   344→              const kk = klines[Math.min(idx, n - 1)];
   345→   345→              const label = timeLabel(kk.t);
   346→   346→              const w = est(label);
   347→   347→              const anchor = k === 0 ? "start" : k === last ? "end" : "middle";
   348→   348→              const cx = k === 0 ? PAD_L : k === last ? PAD_L + plotW : Math.min(x(idx), PAD_L + plotW - 10);
   349→   349→              const left = anchor === "start" ? cx : anchor === "end" ? cx - w : cx - w / 2;
   350→   350→              if (left < prevRight + 10) continue; // would collide with the previous label
   351→   351→              prevRight = left + w;
   352→   352→              nodes.push(
   353→   353→                <text
   354→   354→                  key={k}
   355→   355→                  x={cx}
   356→   356→                  y={height - 8}
   357→   357→                  fill={TEXT}
   358→   358→                  fontSize={10}
   359→   359→                  textAnchor={anchor}
   360→   360→                  fontFamily="ui-monospace, monospace"
   361→   361→                >
   362→   362→                  {label}
   363→   363→                </text>,
   364→   364→              );
   365→   365→            }
   366→   366→            return nodes;
   367→   367→          })()}
   368→   368→
   369→   369→          {/* ── Whale taker-delta histogram (bottom of the plot) ───────── */}
   370→   370→          {whale.maxD > 0 && (
   371→   371→            <g clipPath="url(#wy-plot-clip)">
   372→   372→              <line
   373→   373→                x1={PAD_L}
   374→   374→                x2={PAD_L + plotW}
   375→   375→                y1={PAD_T + plotH}
   376→   376→                y2={PAD_T + plotH}
   377→   377→                stroke={GRID}
   378→   378→                strokeWidth={1}
   379→   379→              />
   380→   380→              {whale.rows.map(({ i, d }) => {
   381→   381→                const h = Math.max(1, (Math.abs(d) / whale.maxD) * whaleH);
   382→   382→                const bw = Math.max(1.5, cw * 0.55);
   383→   383→                return (
   384→   384→                  <rect
   385→   385→                    key={`wh-${i}`}
   386→   386→                    x={x(i) - bw / 2}
   387→   387→                    y={PAD_T + plotH - h}
   388→   388→                    width={bw}
   389→   389→                    height={h}
   390→   390→                    fill={d >= 0 ? UP : DOWN}
   391→   391→                    opacity={0.34}
   392→   392→                  />
   393→   393→                );
   394→   394→              })}
   395→   395→            </g>
   396→   396→          )}
   397→   397→
   398→   398→          {/* ── Volume profile (volume by price, right-anchored) ────────── */}
   399→   399→          {vp && (
   400→   400→            <g clipPath="url(#wy-plot-clip)">
   401→   401→              {vp.rows.map((r, k) => {
   402→   402→                const yTop = y(r.high);
   403→   403→                const yBot = y(r.low);
   404→   404→                const h = Math.max(1, yBot - yTop - 1);
   405→   405→                const tot = r.buy + r.sell;
   406→   406→                if (tot <= 0) return null;
   407→   407→                const w = (tot / vp.max) * vpW;
   408→   408→                const bw = (r.buy / tot) * w;
   409→   409→                const right = PAD_L + plotW;
   410→   410→                return (
   411→   411→                  <g key={`vp-${k}`}>
   412→   412→                    <rect x={right - bw} y={yTop} width={Math.max(0, bw)} height={h} fill={UP} opacity={0.2} />
   413→   413→                    <rect x={right - w} y={yTop} width={Math.max(0, w - bw)} height={h} fill={DOWN} opacity={0.18} />
   414→   414→                  </g>
   415→   415→                );
   416→   416→              })}
   417→   417→              <line
   418→   418→                x1={PAD_L}
   419→   419→                x2={PAD_L + plotW}
   420→   420→                y1={y(vp.poc)}
   421→   421→                y2={y(vp.poc)}
   422→   422→                stroke="#e4e4e7"
   423→   423→                strokeOpacity={0.22}
   424→   424→                strokeDasharray="3 5"
   425→   425→                strokeWidth={1}
   426→   426→              />
   427→   427→            </g>
   428→   428→          )}
