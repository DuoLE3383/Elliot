    49→// ─── Telegram message builders (channel language: Vietnamese) ────────────────
    50→
    51→const msgOpen = (r: AutoTradeRow, capital: number): string =>
    52→  [
    53→    "🤖 AUTO MỞ LỆNH (dry-run)",
    54→    `#${r.symbol} · ${r.interval} · ${r.side}`,
    55→    `Vào: ${fmtPrice(r.entryPrice)}`,
    56→    `SL −${r.slPct.toFixed(1)}% · TP +${r.tpPct.toFixed(1)}% · đòn bẩy ×${r.leverage % 1 === 0 ? r.leverage : r.leverage.toFixed(1)}`,
    57→    `Prob ${(r.prob != null ? r.prob * 100 : 0).toFixed(1)}% · AUC ${(r.auc ?? 0).toFixed(2)} · ACC ${(r.acc ?? 0).toFixed(1)}% · Điểm ${(r.score ?? 0).toFixed(1)}`,
    58→    `Vốn: ${capital.toFixed(2)} USDT`,
    59→  ].join("\n");
    60→
    61→const kindVi = (k: string | null): string => (k === "tp" ? "CHẠM TP ✅" : k === "sl" ? "CẮT LỖ 🔻" : "ĐÓNG THỦ CÔNG");
    62→
    63→const msgClose = (r: AutoTradeRow, closed: number, winRate: number, netPnlPct: number): string =>
    64→  [
    65→    `🤖 AUTO ĐÓNG LỆNH · ${kindVi(r.exitKind)}`,
    66→    `#${r.symbol} ${r.interval} ${r.side}`,
    67→    `Ra: ${fmtPrice(r.exitPrice ?? 0)} · ROE ${(r.pnlPct ?? 0) >= 0 ? "+" : ""}${(r.pnlPct ?? 0).toFixed(2)}%`,
    68→    `Vốn: ${(r.capitalBefore ?? 0).toFixed(2)} → ${(r.capitalAfter ?? 0).toFixed(2)} USDT`,
    69→    `Tổng: ${closed} lệnh · win ${winRate.toFixed(0)}% · net ROE ${netPnlPct >= 0 ? "+" : ""}${netPnlPct.toFixed(1)}%`,
    70→  ].join("\n");
    71→
    72→const msgScan = (rows: ScanRow[], nPairs: number, nTf: number): string => {
    73→  const top3 = rows
    74→    .slice(0, 3)
    75→    .map((r) => `${r.symbol} ${r.interval} (${r.score.toFixed(1)})`)
    76→    .join(" · ");
    77→  const best = rows[0];
    78→  return [
    79→    "📊 AUTO QUÉT XONG",
    80→    `${nPairs} pair × ${nTf} khung = ${nPairs * nTf} combo · ${rows.length} hợp lệ`,
    81→    best
    82→      ? `Tốt nhất: #${best.symbol} ${best.interval} ${best.bias} · AUC ${best.auc.toFixed(2)} · ACC ${best.acc.toFixed(1)}% · Điểm ${best.score.toFixed(1)}`
    83→      : "Không có combo hợp lệ",
    84→    top3 ? `Top 3: ${top3}` : "",
    85→  ]
    86→    .filter(Boolean)
    87→    .join("\n");
    88→};

    93→
    94→export function AutoPanel({ lang, pairs, leverage, tickers, settings, onPickPair, pushToast }: Props) {
    95→  const [cfg, setCfg] = useState<AutoConfig>(loadAutoConfigSafe());
    96→  const [rows, setRows] = useState<ScanRow[]>([]);
    97→  const [scanAt, setScanAt] = useState<number | null>(null);
    98→  const [scanning, setScanning] = useState<{ done: number; total: number; current: string } | null>(null);
    99→  const [auto, setAuto] = useState<AutoState | null>(null);
   100→  const [tgConfigured, setTgConfigured] = useState<boolean | null>(null);
   101→  const [tgState, setTgState] = useState<"idle" | "sending" | "ok" | "fail">("idle");
   102→  const [tgDetail, setTgDetail] = useState<string | null>(null);
   103→  const [status, setStatus] = useState<string>("");
   104→  const [countdown, setCountdown] = useState<number | null>(null);
   105→
   106→  // refs mirror props/state for the interval callbacks (no stale closures,
   107→  // and the cycle interval is NOT reset every time a prop changes)
   108→  const cfgRef = useRef(cfg);
   109→  const rowsRef = useRef(rows);
   110→  const scanAgeRef = useRef(scanAt);
   111→  const tickersRef = useRef(tickers);
   112→  const pairsRef = useRef(pairs);
   113→  const settingsRef = useRef(settings);
   114→  const leverageRef = useRef(leverage);
   115→  const langRef = useRef(lang);
   116→  const busyRef = useRef(false);
   117→  const scanBusyRef = useRef(false);
   118→  const mountedRef = useRef(true);
   119→  const nextAtRef = useRef<number>(0);
   120→
   121→  useEffect(() => {
   122→    cfgRef.current = cfg;
   123→  }, [cfg]);
   124→  useEffect(() => {
   125→    rowsRef.current = rows;
   126→  }, [rows]);
   127→  useEffect(() => {
   128→    scanAgeRef.current = scanAt;
   129→  }, [scanAt]);
   130→  useEffect(() => {
   131→    tickersRef.current = tickers;
   132→  }, [tickers]);
   133→  useEffect(() => {
   134→    pairsRef.current = pairs;
   135→  }, [pairs]);
   136→  useEffect(() => {
   137→    settingsRef.current = settings;
   138→  }, [settings]);
   139→  useEffect(() => {
   140→    leverageRef.current = leverage;
   141→  }, [leverage]);
   142→  useEffect(() => {
   143→    langRef.current = lang;
   144→  }, [lang]);
   145→  useEffect(() => {
   146→    mountedRef.current = true;
   147→    return () => {
   148→      mountedRef.current = false;
   149→    };
   150→  }, []);
   151→
   152→  const refreshAuto = useCallback(async (): Promise<AutoState | null> => {
   153→    try {
   154→      const st = await fetchAutoState();
   155→      if (mountedRef.current) setAuto(st);
   156→      return st;
   157→    } catch {
   158→      return null;
   159→    }
   160→  }, []);
   161→
   162→  // initial load — position book + telegram configuration probe
   163→  useEffect(() => {
   164→    void refreshAuto();
   165→    void telegramConfigured().then((ok) => mountedRef.current && setTgConfigured(ok));
   166→  }, [refreshAuto]);
   167→
   168→  const doScan = useCallback(async (): Promise<ScanRow[]> => {
   169→    if (scanBusyRef.current) return rowsRef.current;
   170→    scanBusyRef.current = true;
   171→    setScanning({ done: 0, total: pairsRef.current.length * 5, current: "" });
   172→    try {
   173→      const out = await scanAll({
   174→        pairs: pairsRef.current,
   175→        settings: settingsRef.current,
   176→        onProgress: (p) => mountedRef.current && setScanning(p),
   177→      });
   178→      if (mountedRef.current) {
   179→        setRows(out);
   180→        setScanAt(Date.now());
   181→      }
   182→      if (out.length > 0) {
   183→        void sendTelegram(msgScan(out, pairsRef.current.length, 5));
   184→      }
   185→      return out;
   186→    } finally {
   187→      scanBusyRef.current = false;
   188→      if (mountedRef.current) setScanning(null);
   189→    }
   190→  }, []);
   191→
   192→  // ── one auto cycle: manage the open position, then maybe enter the best pick
   193→  const runCycle = useCallback(async () => {
   194→    if (busyRef.current) return;
   195→    busyRef.current = true;
   196→    const c = cfgRef.current;
   197→    try {
   198→      const st = await refreshAuto();
   199→      if (!st) {
   200→        setStatus("DB error");
   201→        return;
   202→      }
   203→
   204→      // 1) manage the open position against the live price
   205→      if (st.open) {
   206→        const open = st.open;
   207→        const price = tickersRef.current[open.symbol]?.price ?? 0;
   208→        if (price > 0) {
   209→          const kind = slTpHit(open.side, price, open.entryPrice, open.slPct, open.tpPct);
   210→          const liq = liquidationHit(open.side, price, open.entryPrice, open.leverage);
   211→          if (kind || liq) {
   212→            const closed = await closeAutoTrade(open.id, price, liq ? "sl" : (kind ?? "sl"));
   213→            if (closed) {
   214→              pushToast({
   215→                kind: "flip",
   216→                text: t(langRef.current, "auto.closed", { pair: closed.symbol, tf: closed.interval, pnl: (closed.pnlPct ?? 0).toFixed(2) }),
   217→              });
   218→              void sendTelegram(msgClose(closed, st.stats.closed + 1, st.stats.winRate, st.stats.netPnlPct));
   219→              await refreshAuto();
   220→            }
   221→          } else {
   222→            const roe = roeOf(open.side, open.entryPrice, price, open.leverage);
   223→            setStatus(`${open.symbol} ${open.interval} ${open.side} · ROE ${roe >= 0 ? "+" : ""}${roe.toFixed(2)}%`);
   224→          }
   225→        }
   226→        return; // one position at a time — no new entry while one is open
   227→      }
   228→
   229→      // 2) no open position → rank the scan and enter the best tradeable row
   230→      let current = rowsRef.current;
   231→      if (current.length === 0 || Date.now() - (scanAgeRef.current ?? 0) > SCAN_TTL) {
   232→        setStatus(t(langRef.current, "auto.scan.busy", { done: 0, total: pairsRef.current.length * 5, current: "…" }));
   233→        current = await doScan();
   234→      }
   235→      if (current.length === 0) {
   236→        setStatus(t(langRef.current, "auto.scan.empty"));
   237→        return;
   238→      }
   239→      const best = bestTradeable(current, c.minAuc, c.minAcc);
   240→      if (!best) {
   241→        setStatus(t(langRef.current, "auto.rank.hint"));
   242→        return;
   243→      }
   244→      const price = tickersRef.current[best.symbol]?.price ?? 0;
   245→      if (price <= 0) return; // no live price for that pair — wait for the next cycle
   246→      const capitalNow = (await refreshAuto())?.capital ?? c.capital;
   247→      const trade = await openAutoTrade({
   248→        symbol: best.symbol,
   249→        interval: best.interval,
   250→        side: best.bias,
   251→        entryPrice: price,
   252→        slPct: best.slPct,
   253→        tpPct: best.tpPct,
   254→        leverage: leverageRef.current,
   255→        prob: best.prob,
   256→        acc: best.acc,
   257→        auc: best.auc,
   258→        score: best.score,
   259→        capitalBefore: capitalNow,
   260→      });
   261→      if (trade) {
   262→        pushToast({
   263→          kind: "entry",
   264→          text: t(langRef.current, "auto.opened", { side: trade.side, pair: trade.symbol, tf: trade.interval, price: fmtPrice(trade.entryPrice) }),
   265→        });
   266→        void sendTelegram(msgOpen(trade, capitalNow));
   267→        await refreshAuto();
   268→        setStatus(`${trade.symbol} ${trade.interval} ${trade.side} @ ${fmtPrice(trade.entryPrice)}`);
   269→      }
   270→    } finally {
   271→      busyRef.current = false;
   272→      nextAtRef.current = Date.now() + cfgRef.current.cycleMin * 60_000;
   273→    }
   274→  }, [doScan, pushToast, refreshAuto]);
   275→
   276→  // auto cycle — immediate tick on enable, then every cycleMin minutes
   277→  useEffect(() => {
   278→    if (!cfg.on) {
   279→      setCountdown(null);
   280→      return;
   281→    }
   282→    nextAtRef.current = Date.now() + 2_000;
   283→    void runCycle();
   284→    const id = window.setInterval(() => void runCycle(), cfg.cycleMin * 60_000);
   285→    return () => window.clearInterval(id);
   286→  }, [cfg.on, cfg.cycleMin, runCycle]);
   287→
   288→  // 1 s countdown to the next cycle
   289→  useEffect(() => {
   290→    if (!cfg.on) return;
   291→    const id = window.setInterval(() => {
   292→      const left = Math.max(0, Math.round((nextAtRef.current - Date.now()) / 1000));
   293→      setCountdown(left);
   294→    }, 1000);
   295→    return () => window.clearInterval(id);
   296→  }, [cfg.on]);
   297→
   298→  const updateCfg = (patch: Partial<AutoConfig>) => {
   299→    setCfg((prev) => {
   300→      const next = { ...prev, ...patch };
   301→      saveAutoConfig(next);
   302→      return next;
   303→    });
   304→  };
   305→
   306→  const checkTg = async () => {
   307→    setTgState("sending");
   308→    setTgDetail(null);
   309→    const r = await sendTelegram(TEST_MSG);
   310→    if (r.ok) {
   311→      setTgState("ok");
   312→      pushToast({ kind: "xgb", text: t(lang, "auto.tg.sent") });
   313→    } else {
   314→      setTgState("fail");
   315→      setTgDetail(r.detail ?? null);
   316→      pushToast({
   317→        kind: "error",
   318→        text: /chat not found/i.test(r.detail ?? "")
   319→          ? t(lang, "auto.err.chat")
   320→          : t(lang, "auto.tg.err", { detail: r.detail ?? "?" }),
   321→      });
   322→    }
   323→  };
   324→
   325→  const closeNow = async () => {
   326→    const open = auto?.open;
   327→    if (!open || busyRef.current) return;
   328→    const price = tickers[open.symbol]?.price ?? open.entryPrice;
   329→    const closed = await closeAutoTrade(open.id, price, "manual");
   330→    if (closed) {
   331→      pushToast({
   332→        kind: "flip",
   333→        text: t(lang, "auto.closed", { pair: closed.symbol, tf: closed.interval, pnl: (closed.pnlPct ?? 0).toFixed(2) }),
   334→      });
   335→      void sendTelegram(msgClose(closed, auto?.stats.closed ?? 0, auto?.stats.winRate ?? 0, auto?.stats.netPnlPct ?? 0));
   336→      await refreshAuto();
   337→    }
   338→  };
   339→
   340→  const capitalNow = auto?.capital ?? cfg.capital;
   341→  const open = auto?.open ?? null;
   342→  const best = rows.length > 0 ? rows[0] : null;
   343→  const closedHist = (auto?.trades ?? []).filter((x) => x.status === "closed").slice(0, 4);
   344→  const sideClass = (side: string) => (side === "LONG" ? "text-emerald-400" : "text-rose-400");
   345→
   346→  return (
   347→    <Card className="border-zinc-800 bg-zinc-900/60">
   348→      <CardHeader className="pb-2">
   349→        <div className="flex flex-wrap items-center justify-between gap-2">
   350→          <CardTitle className="flex items-center gap-1.5 text-base font-semibold text-zinc-100">
   351→            <Bot className="h-4 w-4 text-teal-400" />
   352→            {t(lang, "auto.title")}
   353→            {cfg.on && <span className="ml-1 inline-block h-2 w-2 animate-pulse rounded-full bg-emerald-400" aria-hidden />}
   354→          </CardTitle>
   355→          <div className="flex items-center gap-2">
   356→            {tgConfigured != null && (
   357→              <span
   358→                className={`h-2 w-2 rounded-full ${tgConfigured ? "bg-teal-400" : "bg-zinc-600"}`}
   359→                title={tgConfigured ? t(lang, "auto.tg.ok") : t(lang, "auto.tg.none")}
   360→                aria-hidden
   361→              />
   362→            )}
   363→            <Switch checked={cfg.on} onCheckedChange={(v) => updateCfg({ on: v })} aria-label={t(lang, "auto.on")} />
   364→          </div>
   365→        </div>
   366→        <p className="mt-0.5 text-[11px] leading-relaxed text-zinc-500">{t(lang, "auto.sub")}</p>
   367→      </CardHeader>
   368→
   369→      <CardContent className="space-y-3 pt-0">
   370→        {/* config */}
   371→        <div className="flex flex-wrap items-center gap-2">
   372→          <label className="flex items-center gap-1.5 text-[11px] text-zinc-500">
   373→            {t(lang, "auto.cfg.capital")}
   374→            <Input
   375→              type="number"
   376→              min={1}
   377→              value={cfg.capital}
   378→              onChange={(e) => updateCfg({ capital: Math.max(1, Number(e.target.value) || 0) })}
   379→              className="h-7 w-24 border-zinc-800 bg-zinc-950/60 px-2 font-mono text-xs text-zinc-100"
   380→            />
   381→          </label>
   382→          <div className="flex items-center gap-1" role="group" aria-label={t(lang, "auto.cfg.cycle")}>
   383→            <span className="mr-0.5 text-[11px] text-zinc-500">{t(lang, "auto.cfg.cycle")}</span>
   384→            {CYCLES.map((m) => (
   385→              <button
   386→                key={m}
   387→                type="button"
   388→                onClick={() => updateCfg({ cycleMin: m })}
   389→                className={`h-7 rounded-md border px-2 font-mono text-[11px] ${
   390→                  cfg.cycleMin === m
   391→                    ? "border-teal-700 bg-teal-950/40 text-teal-300"
   392→                    : "border-zinc-800 bg-zinc-950/40 text-zinc-400 hover:bg-zinc-800"
   393→                }`}
   394→              >
   395→                {m}
   396→              </button>
   397→            ))}
   398→            <span className="text-[11px] text-zinc-600">{t(lang, "auto.cfg.cycleUnit")}</span>
   399→          </div>
   400→        </div>
   401→        <p className="text-[11px] text-zinc-600">{t(lang, "auto.cfg.min", { auc: cfg.minAuc.toFixed(2), acc: cfg.minAcc })}</p>
   402→
   403→        {/* scan actions */}
   404→        <div className="flex flex-wrap items-center gap-2">
   405→          <Button
   406→            size="sm"
   407→            className="h-8 bg-teal-600 text-zinc-950 hover:bg-teal-500 disabled:opacity-50"
   408→            disabled={scanning != null || pairs.length === 0}
   409→            onClick={() => void doScan()}
   410→          >
   411→            <ScanLine className="mr-1.5 h-3.5 w-3.5" />
   412→            {t(lang, "auto.scan")} ({pairs.length}×5)
   413→          </Button>
   414→          <Button size="sm" variant="outline" className="h-8 border-zinc-700 text-[11px] text-zinc-300 hover:bg-zinc-800" disabled={tgState === "sending"} onClick={() => void checkTg()}>
   415→            <Send className="mr-1.5 h-3 w-3" />
   416→            {t(lang, "auto.tg.check")}
   417→          </Button>
   418→          {scanAt != null && !scanning && (
   419→            <span className="text-[11px] text-zinc-600">
   420→              {t(lang, "auto.scan.ago", { time: new Date(scanAt).toLocaleTimeString(lang === "vi" ? "vi-VN" : "en-GB", { hour: "2-digit", minute: "2-digit" }) })}
   421→            </span>
   422→          )}
   423→        </div>
   424→        {scanning && (
   425→          <div className="space-y-1">
   426→            <p className="font-mono text-[11px] text-teal-300">
   427→              {t(lang, "auto.scan.busy", { done: scanning.done, total: scanning.total, current: scanning.current })}
   428→            </p>
   429→            <div className="h-1.5 overflow-hidden rounded-full bg-zinc-800">
   430→              <div className="h-full rounded-full bg-teal-500 transition-all" style={{ width: `${scanning.total ? (scanning.done / scanning.total) * 100 : 0}%` }} />
   431→            </div>
   432→          </div>
   433→        )}
   434→        {cfg.on && countdown != null && (
   435→          <p className="text-[11px] text-zinc-500">
   436→            {t(lang, "auto.cycle.next", { s: countdown })} {status && <span className="font-mono text-zinc-400">· {status}</span>}
   437→          </p>
   438→        )}
   439→        {!cfg.on && status && <p className="font-mono text-[11px] text-zinc-500">{status}</p>}
   440→
   441→        {/* best pick + ranking */}
   442→        {best ? (
   443→          <div className="space-y-1.5">
   444→            <div className="flex items-center justify-between rounded-lg border border-teal-800/60 bg-teal-950/20 px-2.5 py-1.5">
   445→              <span className="flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wide text-teal-400">
   446→                {t(lang, "auto.best")}
   447→              </span>
   448→              <button
   449→                type="button"
   450→                className="font-mono text-xs font-bold text-zinc-100 hover:text-teal-300"
   451→                onClick={() => onPickPair(best.symbol, best.interval)}
   452→              >
   453→                {best.symbol} {best.interval} <span className={sideClass(best.bias)}>{best.bias}</span> · {best.score.toFixed(1)}
   454→              </button>
   455→            </div>
   456→            <div className="overflow-hidden rounded-lg border border-zinc-800">
   457→              <div className="grid grid-cols-[1fr_26px_44px_44px_38px_38px_40px] gap-x-1 border-b border-zinc-800 bg-zinc-950/60 px-2 py-1 text-[9px] font-semibold uppercase tracking-wide text-zinc-500">
   458→                <span>{t(lang, "auto.col.pair")}</span>
   459→                <span>{t(lang, "auto.col.tf")}</span>
   460→                <span>{t(lang, "auto.col.side")}</span>
   461→                <span className="text-right">{t(lang, "auto.col.prob")}</span>
   462→                <span className="text-right">{t(lang, "auto.col.auc")}</span>
   463→                <span className="text-right">{t(lang, "auto.col.acc")}</span>
   464→                <span className="text-right">{t(lang, "auto.col.score")}</span>
   465→              </div>
   466→              {rows.slice(0, TOP_N).map((r, i) => {
   467→                const tradeable = r.bias !== "NEUTRAL" && r.auc >= cfg.minAuc && r.acc >= cfg.minAcc;
   468→                return (
   469→                  <button
   470→                    key={`${r.symbol}-${r.interval}`}
   471→                    type="button"
   472→                    onClick={() => onPickPair(r.symbol, r.interval)}
   473→                    className={`grid w-full grid-cols-[1fr_26px_44px_44px_38px_38px_40px] items-center gap-x-1 px-2 py-1 text-left font-mono text-[10px] hover:bg-zinc-800/60 ${
   474→                      i === 0 ? "bg-teal-950/20" : ""
   475→                    }`}
   476→                  >
   477→                    <span className="truncate text-zinc-200">{r.symbol}</span>
   478→                    <span className="text-zinc-500">{r.interval}</span>
   479→                    <span className={sideClass(r.bias)}>{r.bias === "NEUTRAL" ? "—" : r.bias}</span>
   480→                    <span className="text-right text-zinc-300">{(r.prob * 100).toFixed(1)}%</span>
   481→                    <span className={`text-right ${r.auc >= cfg.minAuc ? "text-emerald-400" : "text-zinc-500"}`}>{r.auc.toFixed(2)}</span>
   482→                    <span className={`text-right ${r.acc >= cfg.minAcc ? "text-emerald-400" : "text-zinc-500"}`}>{r.acc.toFixed(0)}%</span>
   483→                    <span className={`text-right font-bold ${tradeable ? "text-teal-300" : "text-zinc-600"}`}>{r.score.toFixed(1)}</span>
   484→                  </button>
   485→                );
   486→              })}
   487→            </div>
   488→            <p className="text-[10px] leading-snug text-zinc-600">{t(lang, "auto.rank.hint")}</p>
   489→          </div>
   490→        ) : (
   491→          !scanning && <p className="text-[11px] text-zinc-600">{t(lang, "auto.scan.empty")}</p>
   492→        )}
   493→
   494→        {/* open position */}
   495→        {open ? (
   496→          <div className="rounded-lg border border-zinc-700 bg-zinc-900 p-2.5">
   497→            <div className="flex items-center justify-between">
   498→              <span className="text-[11px] font-semibold uppercase tracking-wide text-zinc-400">{t(lang, "auto.pos.open")}</span>
   499→              <Badge variant="outline" className={`border-zinc-700 font-mono ${sideClass(open.side)}`}>
   500→                {open.side}
   501→              </Badge>
   502→            </div>
   503→            <div className="mt-1.5 grid grid-cols-2 gap-x-3 gap-y-1 text-[11px]">
   504→              <span className="text-zinc-500">{t(lang, "auto.col.pair")}</span>
   505→              <span className="text-right font-mono text-zinc-200">
   506→                {open.symbol} · {open.interval}
   507→              </span>
   508→              <span className="text-zinc-500">{t(lang, "auto.pos.entry")}</span>
   509→              <span className="text-right font-mono text-zinc-200">{fmtPrice(open.entryPrice)}</span>
   510→              <span className="text-zinc-500">SL / TP</span>
   511→              <span className="text-right font-mono">
   512→                <span className="text-rose-300">−{open.slPct.toFixed(1)}%</span> / <span className="text-emerald-300">+{open.tpPct.toFixed(1)}%</span> · ×{open.leverage}
   513→              </span>
   514→              <span className="text-zinc-500">{t(lang, "auto.pos.uroe")}</span>
   515→              {(() => {
   516→                const live = tickers[open.symbol]?.price ?? open.entryPrice;
   517→                const roe = roeOf(open.side, open.entryPrice, live, open.leverage);
   518→                return (
   519→                  <span className={`text-right font-mono font-bold ${roe >= 0 ? "text-emerald-400" : "text-rose-400"}`}>
   520→                    {roe >= 0 ? "+" : ""}
   521→                    {roe.toFixed(2)}%
   522→                  </span>
   523→                );
   524→              })()}
   525→            </div>
   526→            <Button size="sm" variant="outline" className="mt-2 h-7 w-full border-zinc-700 text-[11px] text-zinc-300 hover:bg-zinc-800" onClick={() => void closeNow()}>
   527→              <Square className="mr-1.5 h-3 w-3" />
   528→              {t(lang, "auto.pos.close")}
   529→            </Button>
   530→          </div>
   531→        ) : (
   532→          rows.length > 0 && <p className="text-[11px] text-zinc-600">{t(lang, "auto.pos.none")}</p>
   533→        )}
   534→
   535→        {/* capital + stats */}
   536→        <div className="grid grid-cols-3 gap-2">
   537→          <div className="rounded-lg border border-zinc-800 bg-zinc-950/40 p-2">
   538→            <p className="truncate text-[10px] uppercase tracking-wide text-zinc-500">{t(lang, "auto.capital.now")}</p>
   539→            <p className="mt-0.5 truncate font-mono text-sm font-bold text-zinc-100">{capitalNow.toFixed(2)}</p>
   540→            <p className="truncate text-[9px] text-zinc-600">{t(lang, "auto.capital.start", { n: cfg.capital })}</p>
   541→          </div>
   542→          <div className="rounded-lg border border-zinc-800 bg-zinc-950/40 p-2">
   543→            <p className="truncate text-[10px] uppercase tracking-wide text-zinc-500">Net ROE</p>
   544→            <p className={`mt-0.5 truncate font-mono text-sm font-bold ${(auto?.stats.netPnlPct ?? 0) >= 0 ? "text-emerald-400" : "text-rose-400"}`}>
   545→              {(auto?.stats.netPnlPct ?? 0) >= 0 ? "+" : ""}
   546→              {(auto?.stats.netPnlPct ?? 0).toFixed(1)}%
   547→            </p>
   548→            <p className="truncate text-[9px] text-zinc-600">{t(lang, "auto.stats.trades", { n: auto?.stats.closed ?? 0 })}</p>
   549→          </div>
   550→          <div className="rounded-lg border border-zinc-800 bg-zinc-950/40 p-2">
   551→            <p className="truncate text-[10px] uppercase tracking-wide text-zinc-500">Win</p>
   552→            <p className="mt-0.5 truncate font-mono text-sm font-bold text-zinc-100">{(auto?.stats.winRate ?? 0).toFixed(0)}%</p>
   553→            <p className="truncate text-[9px] text-zinc-600">
   554→              {auto?.stats.wins ?? 0}/{auto?.stats.closed ?? 0}
   555→            </p>
   556→          </div>
   557→        </div>
   558→
   559→        {/* history */}
   560→        <div className="border-t border-zinc-800 pt-2">
   561→          <p className="text-[11px] font-medium uppercase tracking-wide text-zinc-500">{t(lang, "auto.hist")}</p>
   562→          {closedHist.length === 0 ? (
   563→            <p className="mt-1 text-[11px] text-zinc-600">{t(lang, "auto.hist.empty")}</p>
   564→          ) : (
   565→            <div className="rail-scroll mt-1.5 max-h-28 space-y-1 overflow-y-auto pr-0.5">
   566→              {closedHist.map((r) => (
   567→                <div key={r.id} className="flex items-center gap-2 rounded-md border border-zinc-800/80 bg-zinc-950/40 px-2 py-1 text-[10px]">
   568→                  <span className="shrink-0 text-zinc-500">
   569→                    {new Date(r.exitTime ?? r.createdAt).toLocaleTimeString(lang === "vi" ? "vi-VN" : "en-GB", { hour: "2-digit", minute: "2-digit", day: "2-digit", month: "2-digit" })}
   570→                  </span>
   571→                  <span className="truncate font-mono text-zinc-300">
   572→                    {r.symbol} {r.interval}
   573→                  </span>
   574→                  <span className={`shrink-0 font-mono ${sideClass(r.side)}`}>{r.side}</span>
   575→                  <span className={`ml-auto shrink-0 font-mono font-bold ${(r.pnlPct ?? 0) >= 0 ? "text-emerald-400" : "text-rose-400"}`}>
   576→                    {(r.pnlPct ?? 0) >= 0 ? "+" : ""}
   577→                    {(r.pnlPct ?? 0).toFixed(2)}%
   578→                  </span>
   579→                  <span className="shrink-0 font-mono text-zinc-600">{r.exitKind === "tp" ? "TP" : r.exitKind === "sl" ? "SL" : "…"}</span>
   580→                </div>
   581→              ))}
   582→            </div>
   583→          )}
   584→        </div>
   585→      </CardContent>
   586→    </Card>
   587→  );
   588→}
