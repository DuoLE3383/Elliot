    40→const QUOTE_RE = /(USDT|USDC|FDUSD|TUSD|BUSD|BTC|ETH)$/;

    42→/** "pepe" → "PEPEUSDT", "PEPEUSDT" → "PEPEUSDT" */
    43→const normalizePair = (raw: string): string => {
    44→  const s = raw.trim().toUpperCase().replace(/[^A-Z0-9]/g, "");
    45→  if (!s) return "";
    46→  return QUOTE_RE.test(s) ? s : `${s}USDT`;
    47→};
